```c++
//三种做法
#include <bits/stdc++.h>
#include <iostream>
using namespace std;

// 第一种：递归，会运行超时并且会超出int
int num(int n) {
    if(n == 1 || n == 2) {
        return 1;
    } else {
        return num(n - 1) + num(n - 2);
    }
}
int main() {
    int n;
    cin >> n;
    cout << num(n) << endl;
}

// 第二种：数组递推
long long a[100];
int main() {
    int n;
    cin >> n;
    a[1] = 1;
    a[2] = 1;
    for (int i = 3; i <= n; i++) {
        a[i] = a[i - 1] + a[i - 2];
    }
    cout << a[n] << endl;
}

// 第三种：for循环递推
long long x,y,z;
int main() {
    int n;
    cin >> n;
    x = 1;
    y = 1;
    for (int i = 3; i <= n; i++) {
        z = x + y;
        x = y;
        y = z;
    }
    if(n == 1 || n == 2) {
        cout << 1;
    } else {
        cout << z;
    }
}
```

二分查找

```c++
#include <iostream>
#include <bits/stdc++.h>
using namespace std;

int a[1000100],n,x;
int main(){
    int i,r = -1; //r 用来记录找到的数的位置，如果没找到就返回 -1
    int left,right,mid;
    scanf("%d",&n);
    for(i = 0; i < n; i++){
        scanf("%d",&a[i]);
    }
    scanf("%d",&x);
    // 二分查找
    left =  0;
    right = n - 1;
    while(left <= right){
        mid = (left + right) / 2;
        if(a[mid] == x){
            r = mid + 1;
            break;
        }else if(a[mid] > x){
            right = mid - 1;
        }else{
            left = mid + 1;
        }
    }
    printf("%d",r);
}
```





二分查找左边界

```c++
#include <iostream>
#include <bits/stdc++.h>
using namespace std;

const int N = 100100;
int a[N];
int n,q; // n个数，q次查询

int fun(int x) {
    int l = 1,r = n,mid;
    while(l <= r) {
        mid = (l + r) >> 1; // 右移运算符，相当于/2
        if(x < a[mid]) r = mid - 1;
        else if(x > a[mid]) l = mid + 1;
        else if(x == a[mid]) r = mid - 1;
    }
    if(a[l] == x) return l;
    else return -1;
}

int main() {
    scanf("%d", &n);
    for(int i = 1; i <= n; i++) {
        scanf("%d", &a[i]);
    }
    scanf("%d", &q);
    int x;
    for(int i = 1; i <= q; i++) {
        scanf("%d", &x);
        // 求x在数组中首次出现的位置
        // cout << fun(x) << " ";
        printf("%d ", fun(x));
    }
}
```





二分查找右边界

```c++
#include <iostream>
#include <bits/stdc++.h>
using namespace std;

const int N = 100100;
int a[N];
int n,q; // n个数，q次查询

int fun(int x) {
    int l = 1,r = n,mid;
    while(l <= r) {
        mid = (l + r) >> 1; // 右移运算符，相当于/2
        if(x < a[mid]) r = mid - 1;
        else if(x > a[mid]) l = mid + 1;
        else if(x == a[mid]) l = mid + 1; // 相等右移
    }
    // l就是数组中第一个大于x的数的位置
    if(a[l - 1] == x) return l - 1;
    else return -1;
}

int main() {
    scanf("%d", &n);
    for(int i = 1; i <= n; i++) {
        scanf("%d", &a[i]);
    }
    scanf("%d", &q);
    int x;
    for(int i = 1; i <= q; i++) {
        scanf("%d", &x);
        // 求x在数组中首次出现的位置
        // cout << fun(x) << " ";
        printf("%d ", fun(x));
    }
}
```

        for(j = 1; j <= i; j++) {

        }
    }

    for(i = n - 1; i >= 1; i--) {
        for(j = 1; j <= i; j++) {
            // if(a[i + 1][j] > a[i + 1][j + 1]) {

            //     a[i][j] += a[i + 1][j + 1];

            a[i][j] = a[i][j] + max(a[i + 1][j], a[i + 1][j + 1]);
        }
    }

}

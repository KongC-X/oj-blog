```c++
#include<iostream>
using namespace std;

int main() {
	int n,i,j;
    int a[11][11] = {0}; // 一定要赋值为0
	cin>>n;
	for(i = 0;i < n; i++) {
        for(j = 0;j <= i;j++) {
            if(j == 0) {
                a[i][j] = 1;
            } else {
                // 每个数是上一行的两数之和
            	a[i][j] = a[i-1][j-1] + a[i-1][j];
            }
        }
    }
    // 输出杨辉三角形
	for(i = 0;i < n;i++){
		for(j = 0;j <= i;j++){
			cout<<a[i][j]<<" ";
		}
		cout<<endl;
	}
}
```

// 加法
string add(string s1,string s2) {
    int a[250] = {0},b[250] = {0},c[250] = {0};
    int i,p = 0,len;
    // 逆序存入整数数组中
    for(i = s1.size() - 1;i >= 0;i--) {
        a[i] = s1[s1.size() - 1 - i] - '0';
    }
    for(i = s2.size() - 1;i >= 0;i--) {
        b[i] = s2[s2.size() - 1 - i] - '0';
    }
    // 加法的次数取决于较长的那个字符串
    len = s1.size() > s2.size() ? s1.size() : s2.size();
    // 逐位相加
    for(i = 0;i < len;i++) {
        c[i] = a[i] + b[i];
    }
    // 进位
    for(i = 0;i < len;i++) {
        if(c[i] >= 10) {
            c[i + 1] = c[i + 1] + c[i] / 10;
            c[i] = c[i] % 10;
        }
    }
    // 两个不超过len位的整数做加法，结果可能是len+1位
    if(c[len] != 0) {
        len++;
    }
    string s3 = "";
    // 逆序输出结果
    for(i = len - 1;i >= 0;i--){
        s3 = s3 + char(c[i] + '0');
    }
    return s3;
}

// 减法
string jian(string s1,string s2) {
    int a[250] = {0},b[250] = {0},c[250] = {0};
    int i = 0,p = 0,len = 0;
    string s3 = "";
    // 如果s1比s2对应的整数小，结果为负，交换s1和s2
    if(s1.size() < s2.size() || (s1.size() == s2.size() && s1 < s2)) {
        swap(s1,s2);
        s3 += '-'; // 结果为负，加一个负号
    }
    // 逆序存储，方便逐位相减
    for(i = s1.size() - 1;i >= 0;i--) {
        a[i] = s1[s1.size() - 1 - i] - '0';
    }
    for(i = s2.size() - 1;i >= 0;i--) {
        b[i] = s2[s2.size() - 1 - i] - '0';
    }
    // 逐位相减
    for(i = 0;i < s1.size();i++) {
        c[i] = a[i] - b[i];
    }
    // 处理借位
    for(i = 0;i < s1.size();i++) {
        if(c[i] < 0) {
            c[i] = c[i] + 10;
            c[i + 1] = c[i + 1] - 1;
        }
    }
    // 去除前导0
    for(i = s1.size() - 1;i >= 0;i--) {
        if(c[i] != 0) {
            p = i;
            break;
        }
    }
    // 逆序输出结果
    for(i = p;i >= 0;i--){
        s3 += to_string(c[i]);
    }
    return s3;
}

// 乘法
string cheng(string s1,string s2) {
    int a[5010] = {0},b[5010] = {0},c[10010] = {0};
    int i,j,p = 0;
    string s3 = "";
    // 逆序存储
    for(i = s1.size() - 1;i >= 0;i--) {
        a[i] = s1[s1.size() - 1 - i] - '0';
    }
    for(i = s2.size() - 1;i >= 0;i--) {
        b[i] = s2[s2.size() - 1 - i] - '0';
    }
    // 循环a数组的每一位，用a[i]去乘b数组的每一位b[i]
    // 结果错位加到c数组的c[i+j]这一位上
    for(i = 0;i < s1.size();i++) {
        for(j = 0;j < s2.size();j++) {
            c[i+j] += a[i] * b[j];
            // 进位
            if(c[i+j] >= 10) {
                c[i+j+1] += c[i+j] / 10;
                c[i+j] %= 10;
            }
        }
    }
    // 去除前导0
    for(i = s1.size() + s2.size() - 1;i >= 0;i--) {
        if(c[i] != 0) {
            p = i;
            break;
        }
    }
    // 逆序输出结果
    for(i = p;i >= 0;i--){
        s3 += to_string(c[i]);
    }
    return s3;
}

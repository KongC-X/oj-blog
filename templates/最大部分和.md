// 解法二：动态规划
#include <iostream>
#include <bits/stdc++.h>
using namespace std;

int a[110],n,i,j;
int dp[110]; // dp[i]表示以第i个数结尾的最大连续子序列和
int ma,s; // ma存放连续最大和，s存放从每个数开始连续数和

int main() {
    cin >> n;
    for(i = 1;i <= n;i++) {
        cin >> a[i];
    }

    // 状态转移方程：
    // dp[i] = max(dp[i-1] + a[i], a[i])
    // 边界：dp[1] = a[1]
    dp[1] = a[1];
    ma = a[1];
    for(i = 2;i <= n;i++) {
        dp[i] = max(dp[i-1] + a[i], a[i]);
        ma = max(ma, dp[i]);
    }
    cout << ma << endl;
}

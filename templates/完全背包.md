```c++
#include <bits/stdc++.h> // 万能头文件
#include <iostream>
using namespace std;

/*
    01背包问题
    二维数组：dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - w[i]] + v[i])
    一维数组滚动优化：从背包容量开始倒序遍历到当前物品的重量wi
    一维数组：dp[j] = max(dp[j], dp[j - w[i]] + v[i])

    完全背包问题：每种物品可以取无限次
    二维数组：dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - k * w[i]] + k* v[i])
    经过变换之后和01二维只有一个i和i - 1的区别：
    dp[i][j] = max(dp[i - 1][j], dp[i][j - w[i]] + v[i])
    一维数组滚动优化：从当前物品的重量wi开始正序遍历到背包容量c
    一维数组：dp[j] = max(dp[j], dp[j - w[i]] + v[i])

*/


// 我们先用二维数组求解完全背包，会发现超时了
int n,maxw;
int dp[10010][10010],i,j;
int w[10010],v[10010];

int main() {
    cin >> maxw >> n;
    for(i = 1; i <= n; i++) {
        cin >> w[i] >> v[i];
    }
    for(i = 1; i <= n; i++) {
        for(j = 1;j <= maxw; j++) {
            if(j >= w[i]) {
                dp[i][j] = max(dp[i - 1][j], dp[i][j - w[i]] + v[i]);
            } else {
                dp[i][j] = dp[i - 1][j];
            }
        }
    }
    cout << dp[n][maxw] << endl;
}
```





一维数组滚动优化

```c++
#include <bits/stdc++.h> // 万能头文件
#include <iostream>
using namespace std;

/*
    01背包问题
    二维数组：dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - w[i]] + v[i])
    一维数组滚动优化：从背包容量开始倒序遍历到当前物品的重量wi
    一维数组：dp[j] = max(dp[j], dp[j - w[i]] + v[i])

    完全背包问题：每种物品可以取无限次
    二维数组：dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - k * w[i]] + k* v[i])
    经过变换之后和01二维只有一个i和i - 1的区别：
    dp[i][j] = max(dp[i - 1][j], dp[i][j - w[i]] + v[i])
    一维数组滚动优化：从当前物品的重量wi开始正序遍历到背包容量c
    一维数组：dp[j] = max(dp[j], dp[j - w[i]] + v[i])

*/

int c; // 背包容量
int dp[100010];
int n,wi,vi;

int main() {
    cin >> c >> n;
    // 遍历每个物品
    for(int i = 1;i <= n;i++) {
        cin >> wi >> vi;
        // 从当前物品的重量wi开始正序遍历
        for(int j = wi;j <= c;j++) {
            dp[j] = max(dp[j], dp[j - wi] + vi);
        }
    }
    cout << dp[c] << endl;
}
```

## 栈

```c++
#include <iostream>
// 常量：栈的长度
#define MAXN 5
using namespace std;

int stack[MAXN]; // 数组模拟栈
int top = -1; // 栈顶指针

// 出栈
int pop() {
    if (top < 0) {
        cout << "栈为空" << endl;
        return -1;
    } else {
        return stack[top--];
    }
}

// 入栈
void push(int value) {
    if (top == MAXN - 1) {
        cout << "栈满" << endl;
    } else {
        top++;
        stack[top] = value;
    }
}

// 显示栈
void display() {
    for(int i = top; i >= 0; i--) {
        cout << stack[i] << " ";
    }
    cout << endl;
}

int main() {
    int order; // 指令
    int x,t;
    cout << "输入指令：";
    while(true) {
        cout << "1.入栈 2.出栈 3.显示栈 4.退出" << endl;
        cin >> order;
        switch(order) {
            case 1:
                cout << "输入入栈元素：";
                cin >> x;
                push(x);
                display();
                break;
            case 2:
                t = pop();
                if(t != -1) {
                    cout << "出栈元素：" << t << endl;
                    display();
                }
                break;
            case 3:
                display();
                break;
            case 4:
                return 0;
        }
    }
}
```





## 队列

```c++
#include <iostream>
#define MAXN 5
using namespace std;

// 队列
int queue[MAXN];
// 头指针
int front = 0;
// 尾指针
int rear = 0;

// 入队
void addqueue(int value) {
    if(rear >= MAXN) {
        cout << "队列已满" << endl;
        return;
    } else {
        queue[rear] = value;
        rear++;
    }
}

// 出队
int delqueue() {
    if(front == rear) {
        cout << "队列为空" << endl;
        return -1;
    } else {
        return queue[front++];
    }
}

// 显示队列
void display() {
    for(int i = front; i < rear; i++) {
        cout << queue[i] << " ";
    }
    cout << endl;
}

int main() {
    int order; // 操作指令
    int value,t; // 操作值
    cout << "输入指令：" << endl;
    while(true) {
        cout << "1.入队 2.出队 3.显示队列 4.退出" << endl;
        cin >> order;
        switch(order) {
            case 1:
                cout << "请输入入队值：";
                cin >> value;
                addqueue(value);
                display();
                break;
            case 2:
                t = delqueue();
                if(t != -1) {
                    cout << "出队值：" << t << endl;
                    display();
                }
                break;
            case 3: display(); break;
            case 4: return 0;
        }
    }
}
```





## 循环队列

```c++
#include <iostream>
#define MAXN 5
using namespace std;

// 队列
int queue[MAXN];
// 头指针
int front = 0;
// 尾指针
int rear = 0;

// 入队
void addqueue(int value) {
    if((rear + 1) % MAXN == front) {
        cout << "队列已满" << endl;
    } else {
        queue[rear] = value;
        rear = (rear + 1) % MAXN;
    }
}

// 出队
int delqueue() {
    int r;
    if(front == rear) {
        cout << "队列为空" << endl;
        r = -1;
    } else {
        r = queue[front];
        queue[front] = -1; // 出队标记
        front = (front + 1) % MAXN;
    }
    return r;
}

// 显示队列
void display() {
    if(front == rear) {
        cout << "队列为空" << endl;
    } else {
        int i = front;
        do{
            cout << queue[i] << " ";
            i = (i + 1) % MAXN;
        } while(i != rear);
        cout << endl;
    }
}

int main() {
    int order; // 操作指令
    int value,t; // 操作值
    cout << "输入指令：" << endl;
    while(true) {
        cout << "1.入队 2.出队 3.显示队列 4.退出" << endl;
        cin >> order;
        switch(order) {
            case 1:
                cout << "请输入入队值：";
                cin >> value;
                addqueue(value);
                display();
                break;
            case 2:
                t = delqueue();
                if(t != -1) {
                    cout << "出队值：" << t << endl;
                    display();
                }
                break;
            case 3: display(); break;
            case 4: return 0;
        }
    }
}
```

## 链表

```c++
#include <bits/stdc++.h>
#include <iostream>
using namespace std;

// 结点定义
struct Node {
    int data;
    struct Node *next;
};

// 头结点
Node *head = NULL;

// 在链表末尾追加
void add(int x) {
    // 如果有head的值
    if(head != NULL) {
        // 要追加的节点
        Node *a = new Node;
        a->data = x;
        a->next = NULL;
        Node *p = head;
        // 移动到最后一个节点
        while(p->next != NULL) {
            p = p->next;
        }
        // 追加节点
        p->next = a;
    } else {
        // 如果没有head的值，则创建一个节点
        head = new Node;
        head->data = x;
        head->next = NULL;
    }
}

// 在中间追加元素（在第n个元素的位置）
void insert(int n, int x) {
    // 新节点
    Node *d = new Node;
    d->data = x;
    d->next = NULL;
    // 如果是头结点
    if(n == 1) {
        d->next = head;
        head = d;
    } else {
        // 找到第n个节点
        Node *p = head;
        // 为什么是n - 2，因为n是从1开始的，而循环是从0开始的
        for(int i = 1; i <= n - 2; i++) {
            p = p->next;
            if(p == NULL) {
                break;
            }
        }
        if(p == NULL) {
            cout << "插入位置不合法" << endl;
        } else {
            d->next = p->next;
            p->next = d;
        }
    }
}

// 删除链表中第一个出现的data
void deldata(int data) {
    Node *p = head, *pre = NULL;
    while(p != NULL) {
        if(data == p->data) {
            if(p == head) {
                head = p->next;
            } else {
                pre->next = p->next;
            }
            delete p;
            break;
        }
        pre = p;
        p = p->next;
    }
}

// 删除某个位置的元素
void delpos(int n) {
    Node *p = head, *t;
    // 如果要删除头结点
    if(n == 1) {
        if(head != NULL) {
            head = head->next;
            delete p;
        } else {
            cout << "链表为空" << endl;
        }
    } else {
        for(int i = 1;i <= n - 2;i++) {
            p = p->next;
            if(p == NULL) {
                break;
            }
        }
        if(p == NULL || p->next == NULL) {
            cout << "删除位置不合法" << endl;
        } else {
            t = p->next;
            p->next = t->next;
            delete t;
        }
    }
}

// 输出链表
void display() {
    Node *p = head;
    while(p != NULL) {
        cout << p->data << " ";
        p = p->next;
    }
    cout << endl;
}

int main() {
    int order,x,p;
    cout << "输入指令：";
    while(true) {
        cout << "1:追加，2:插入，3:删除值，4:删除位置，5:显示，6:退出"<< endl;
        cin >> order;
        switch(order) {
            case 1:
                cout << "输入追加的值：";
                cin >> x;
                add(x);
                display();
                break;
            case 2:
                cout << "输入插入的位置和值：";
                cin >> p >> x;
                insert(p, x);
                display();
                break;
            case 3:
                cout << "输入要删除的值：";
                cin >> x;
                deldata(x);
                display();
                break;
            case 4:
                cout << "输入要删除的位置：";
                cin >> p;
                delpos(p);
                display();
                break;
            case 5:
                display();
                break;
            case 6:
                return 0;
        }
    }
}
```





## 链式栈

```c++
#include <bits/stdc++.h>
#include <iostream>
using namespace std;

// 栈元素
struct Stack {
    int data;
    Stack *next;
};

Stack *top = NULL; // 栈顶指针

// 入栈
void push(int x) {
    Stack *p = new Stack;
    p->data = x;
    p->next = top;
    top = p; // 栈顶指针指向新元素
}

// 出栈
void pop() {
    Stack *p = top;
    if(p != NULL) {
        cout << p->data << "出栈" << endl;
        p = p->next;
        delete top;
        top = p; // 栈顶指针指向下一个元素
    } else {
        cout << "栈为空" << endl;
    }
}

// 显示栈中元素
void display() {
    Stack *p = top;
    if(p != NULL) {
        cout << "栈中元素为：";
        while(p != NULL) {
            cout << p->data << " ";
            p = p->next;
        }
        cout << endl;
    } else {
        cout << "栈为空" << endl;
    }
}

// 获取栈长度
int getlen() {
    int len = 0;
    Stack *p = top;
    while(p != NULL) {
        len++;
        p = p->next;
    }
    return len;
}

int main() {
    int order,x,p;
    cout << "输入指令：" << endl;
    while(true) {
        cout << "1.入栈 2.出栈 3.显示栈中元素 4.获取栈长度 5.退出" << endl;
        cin >> order;
        switch(order) {
            case 1:
                cout << "请输入要入栈的元素：";
                cin >> x;
                push(x);
                display();
                break;
            case 2:
                pop();
                display();
                break;
            case 3:
                display();
                break;
            case 4:
                p = getlen();
                cout << "栈长度为：" << p << endl;
                break;
            case 5:
                return 0;
        }
    }
}
```





## 链式队列

```c++
#include <bits/stdc++.h>
#include <iostream>
using namespace std;

// 队列结点
struct Queue {
    int data;
    struct Queue *next;
};

Queue *front = NULL, *rear = NULL; // 队头和队尾指针

// 入队
void add(int value) {
    Queue *e = new Queue;
    e->data = value;
    e->next = NULL;
    // 队列第一个元素
    if (front == NULL) {
        front = e;
        rear = e;
    } else {
        rear->next = e;
        rear = e;
    }
}

// 出队
void del() {
    Queue *t;
    // 如果队列有元素
    if (front != NULL) {
        cout << front->data << "出队" << endl;
        t = front;
        front = front->next;
        if(front == NULL) rear = NULL;
        delete t;
    } else {
        cout << "队列为空" << endl;
    }
}

// 显示队列
void display() {
    Queue *p = front;
    while (p != NULL) {
        cout << p->data << " ";
        p = p->next;
    }
    cout << endl;
}

int main() {
    int order,x,p;
    cout << "输入指令：" << endl;
    while(true) {
        cout << "1.入队 2.出队 3.显示队列 4.退出" << endl;
        cin >> order;
        switch(order) {
            case 1:
                cout << "输入入队元素：";
                cin >> x;
                add(x);
                display();
                break;
            case 2:
                del();
                display();
                break;
            case 3:
                display();
                break;
            case 4:
                return 0;
        }
    }
}
```

鸡兔同笼

```c++
# include <iostream>
using namespace std;

int main() {
    int x,y; // x 是鸡的数量，y 是兔的数量
    for (x = 0; x <= 50; x++) {
        y = 50 - x;
        if (2*x + 4*y == 160) {
            cout << x << endl;
            cout << y << endl;
            break;
        }
    }
    return 0;
}

```



百钱百鸡

```c++
# include <iostream>
using namespace std;

int main() {
    int i,j,k;
    for(i = 1;i <= 100/5;i++){
        for(j = 1;j <= 100/3;j++){
            for(k = 1;k <= 300;k++) {
                if(i * 5 + j * 3 + k / 3 == 100 && k % 3 == 0 && i + j + k == 100) {
                    cout << i << " " << j << " " << k << endl;
                }
            }
        }
    }
    return 0;
}
```

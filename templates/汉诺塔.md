汉诺塔移动

```c++
#include<iostream>
using namespace std;

/*
汉诺塔移动问题
1、先将n-1个（除了最下面那个）盘子从A借助C移动到B，需要n-1次fun(n-1)次
2、将最下面那个盘子从A移动到C，需要1次fun(1)次
3、将n-1个（除了最下面那个）盘子从B借助A移动到C，需要n-1次fun(n-1)次
因此得到结论：fun(n) = fun(n-1) + 1 + fun(n-1) = 2*fun(n-1) + 1
* */

int fun(int n) {
    if(n == 1) {
        return 1;
    } else {
        return 2 * fun(n - 1) + 1;
    }
}

int main() {
    int n;
    cin >> n;
    cout << fun(n) << endl;
    return 0;
}
```





```c++
 #include<iostream>
using namespace std;

/*
汉诺塔移动问题
1、先将n-1个（除了最下面那个）盘子从A借助C移动到B，需要n-1次fun(n-1)次
2、将最下面那个盘子从A移动到C，需要1次fun(1)次
3、将n-1个（除了最下面那个）盘子从B借助A移动到C，需要n-1次fun(n-1)次
因此得到结论：fun(n) = fun(n-1) + 1 + fun(n-1) = 2*fun(n-1) + 1
* */

// 函数的作用：将n个盘子从p1位置借助p2位置移动到p3位置
// p1:起始位置，p2:借助位置，p3:目标位置
void move(int n, char p1, char p2, char p3) {
    if(n > 0) {
        // 第一步：将n-1个盘子从p1借助p3移动到p2
        move(n-1, p1, p3, p2);
        // 第二步：将第n个盘子从p1移动到p3
        cout << p1 << " To " << p3 << endl;
        // 第三步：将n-1个盘子从p2借助p1移动到p3
        move(n-1, p2, p1, p3);
    }
}

int main() {
    int n;
    cin >> n;
    move(n, 'A', 'B', 'C');
}
```

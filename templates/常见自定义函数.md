阶乘

```c++
// 求一个数的阶乘
int factorial(int a) {
    int result = 1;
    for (int i = 1; i <= a; i++) {
        result *= i;
    }
    return result;
}
```



数字之和

```c++
// 求一个数的各位数字之和
int numSum(int num) {
    int sum = 0;
    while(num != 0) {
        sum += num % 10;
        num = num / 10;
    }
    return sum;
}
```



素数

```c++
// 判断是否是素数
bool isPrime(int n) {
    if (n <= 1) {
        return false;
    }
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) {
            return false;
        }
    }
    return true;
}
```



因子和

```c++
// 求一个数的因子和
int sumOfFactors(int n) {
    int sum = 0;
    for (int i = 2; i < n; i++) {
        if (n % i == 0) {
            sum += i;
        }
    }
    return sum;
}
```



倒数

```c++
// 倒数
int reverse(int n) {
    int s = 0;
    while(n != 0) {
        s = s * 10 + n % 10;
        n = n / 10;
    }
    return s;
}
```



数位之和

```c++
// 求数位的和
int digitSum(int n) {
    int sum = 0;
    while (n > 0) {
        sum += n % 10;
        n /= 10;
    }
    return sum;
}
```

```c++
#include <bits/stdc++.h>
using namespace std;

/*
dp[i][j]：代表有i个物品，背包容量为j时，能够存储的最大价值
对于第i个物品：
情况一：背包容量j<物品重量w[i]，放不下，能够存放的价值dp[i-1][j]
情况二：背包容量j>=物品重量w[i]，放得下
能够得到的最大价值 = max(dp[i-1][j],v[i]+dp[i-1][j-w[i]])
*/ 
int dp[110][20010];
int n,w[110],v[110],maxw,i,j;

int main(){
	cin>>maxw>>n;
	//读入每个物品的重量和价值
	for(i = 1;i <= n;i++){
		cin>>w[i]>>v[i];
	} 
	// 递推求dp数组
	for(i = 1;i <= n;i++){
		//循环背包容量从1~maxw
		for(j = 1;j <= maxw;j++){
			//如果放不下:背包容量<物品重量 
			if(j < w[i]){
				dp[i][j] = dp[i-1][j]; 
			}else{
				//放得下，就看放进来价值高，还是不放价值高
				dp[i][j] = max(dp[i-1][j],v[i]+dp[i-1][j-w[i]]);
			} 
		} 
	} 	
	//输出n个物品，背包容量为maxw的情况下最大价值 
	cout<<dp[n][maxw];
}
```



一维数组滚动优化

```c++
#include <bits/stdc++.h> // 万能头文件
#include <iostream>
using namespace std;

/*
    二维数组：dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - w[i]] + v[i])
    一维数组滚动优化后的状态转移方程
    一维数组：dp[j] = max(dp[j], dp[j - w[i]] + v[i])
*/

int w; // 背包容量
int dp[20010];
int n,wi,vi;

int main() {
    cin >> w >> n;
    // 遍历每个物品
    for(int i = 1;i <= n;i++) {
        cin >> wi >> vi;
        // 倒过来循环
        for(int j = w;j >= wi;j--) {
            // 讨论每个物品选和不选的两个状态
            // 取能得到的价值的最大值
            dp[j] = max(dp[j], dp[j - wi] + vi);
        }
    }
    cout << dp[w] << endl;
}
```

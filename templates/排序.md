冒泡排序

```c++
# include <iostream>
using namespace std;

int main() {
    int a[20],i,n;
    cin >> n;
    for(i = 0; i < n; i++) {
        cin >> a[i];    
    }
    // 从小到大冒泡排序
    for(i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            if(a[j] > a[j + 1]) {
                int t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
            }
        }
    }
    for(i = 0; i < n; i++) {
        cout << a[i] << " ";
    }
}
```





归并排序

```c++
#include<bits/stdc++.h> 
#include<iostream> 
using namespace std;

// 合并函数，将两个已排序的子数组合并成一个有序数组
void merge(int arr[], int left, int mid, int right) {
	int i = left; // 左子数组的起始索引
	int j = mid + 1; // 右子数组的起始索引
	int k = 0; // 临时数组的索引
	int temp[right - left + 1]; // 临时数组用于存储合并后的结果
    // 区间内的元素数量 = 右边界 - 左边界 + 1

	// 当左右子数组都有元素时，比较它们的首元素，并将较小的元素放入临时数组
	while(i <= mid && j <= right) {
		if(arr[i] <= arr[j]) {
			temp[k++] = arr[i++];
		} else {
			temp[k++] = arr[j++];
		}
	}

	// 将剩余的左子数组元素放入临时数组
	while(i <= mid) {
		temp[k++] = arr[i++];
	}

	// 将剩余的右子数组元素放入临时数组
	while(j <= right) {
		temp[k++] = arr[j++];
	}

	// 将临时数组中的元素复制回原数组
	for(i = left, k = 0; i <= right; i++, k++) {
	    arr[i] = temp[k];
	}
	
}

// 归并排序函数，递归地将数组分成两半，然后调用合并函数合并
void mergeSort(int arr[], int left, int right) {
	if(left < right) {
		int mid = (left + right) / 2; // 中间索引
		// int mid = left + (right - left) / 2;  // 当left和right很大时防止整数溢出
		mergeSort(arr, left, mid); // 对左半部分进行归并排序
		mergeSort(arr, mid + 1, right); // 对右半部分进行归并排序
		merge(arr, left, mid, right); // 合并两个已排序的子数组
	}
}

int main() {
	int a[8] = {11,8,3,9,7,1,2,5};
	int n = sizeof(a)/sizeof(a[0]); // 计算数组长度通用方法
	mergeSort(a, 0, n-1); // 调用归并排序函数
	for(int i=0; i<n; i++) {
		cout << a[i] << " ";
	}
}
```



快速排序

```c++
#include<bits/stdc++.h> 
#include<iostream> 
using namespace std;

// 第一种方法：更符合快速排序的经典描述（分治 + 双指针交换），交换逻辑更直接，容易可视化
// 分区函数：将数组分为两部分，左边小于基准值，右边大于基准值
int partition(int arr[], int left, int right) {
    int pivot = arr[left]; // 选择第一个元素作为基准值
	int i = left; // 左指针
	int j = right; // 右指针
    while(i < j) {
        // 从右向左找到第一个小于基准值的元素
        while(i < j && arr[j] >= pivot) {
            j--;
        }
        // 从左向右找到第一个大于基准值的元素
        while(i < j && arr[i] <= pivot) {
            i++;
        }
        // 如果i < j, 交换两个元素
		if(i < j) {
			swap(arr[i], arr[j]);
		}
    }
    // 将基准值放到正确的位置，即排序后的最终位置
    swap(arr[left], arr[i]);
    return i; // 返回基准值的位置
}


// 第二种方法：单方向扫描，维护一个"小于基准"的边界，代码结构更简单，但是交换次数会变多
// 分区函数
int partition(int arr[], int left, int right) {
	int pivot = arr[right]; // 选择最右边的元素作为基准值
	int i = left - 1; // i指向小于基准值的最后一个元素
	for(int j = left; j < right; j++) {
		if(arr[j] <= pivot) { // 如果当前元素小于等于基准值
			i++; // i指针右移,扩大小于基准的区域
			swap(arr[i], arr[j]); // 把当前元素放到小于区域
		}
	}
	swap(arr[i + 1], arr[right]); // 将基准值放到正确的位置
	return i + 1; // 返回基准值的位置
}


// 快速排序函数，使用递归实现
void quickSort(int arr[],int left,int right) {
	if(left < right) { // 如果子数组长度大于1
		int pivotIndex = partition(arr, left, right); // 找到基准值的位置
		quickSort(arr, left, pivotIndex - 1); // 对基准值左边的子数组进行快速排序
		quickSort(arr, pivotIndex + 1, right); // 对基准值右边的子数组进行快速排序

	}
}

int main() {
	int a[9] = {52,144,42,134,63,186,18,143,167};
	int n = sizeof(a)/sizeof(a[0]); // 获取数组长度
	quickSort(a, 0, n - 1); // 调用快速排序函数
	for(int i=0; i<n; i++) {
		cout << a[i] << " ";
	}
}
```

P1835 素数密度

洛谷 | 普及| C++14 | 提交时间: 2026-05-22

算法标签：数学、NOI 导刊、枚举、素数判断

题目背景

UPD:

  • 2024.8.12：加入一组 Hack 数据。

题目描述

给定 L,RL,RL,R，请计算区间 [L,R][L,R][L,R] 中素数的个数。

1≤L≤R<2311\leq L\leq R < 2^{31}1≤L≤R<231，R−L≤106R-L\leq 10^6R−L≤106。

输入格式

第一行，两个正整数 LLL 和 RRR。

输出格式

一行，一个整数，表示区间中素数的个数。

样例

样例输入 #1

2 11```

**样例输出 #1**

5```

AC 代码

#include <iostream>
#include <bits/stdc++.h>   // 万能头，包含需要的所有库
using namespace std;

const int MAXLEN = 1000000;      // 区间最大长度 1e6
bool isP[46341];                 // 标记 sqrt(R) 以内的数是否为素数（false 表示素数）
int pr[5000], cnt;               // 存储 sqrt(R) 以内的素数（约 4792 个）
bool seg[MAXLEN + 5];            // 区间偏移数组，seg[idx]=false 表示 L+idx 可能是素数

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);                  // 加速输入输出

    long long L, R;
    cin >> L >> R;

    // ---------- 详细思路 ----------
    // 由于 L,R 可能很大（<2^31），但区间长度 R-L <= 1e6，
    // 不能直接开长度为 R 的数组筛素数。
    // 采用“区间筛”（分段筛）：
    // 1. 先用普通筛求出 2 ~ sqrt(R) 的所有素数（sqrt(R) ≤ 46340）。
    // 2. 对于每个小素数 p，将区间 [L,R] 内 p 的倍数标记为合数。
    //    标记时采用偏移下标：对于数 x，在 seg 中的位置为 x - L。
    // 3. 最后统计 seg 中未被标记的数的个数，注意 1 不是素数需跳过。

    // ---------- 第 1 步：筛出 sqrt(R) 以内素数 ----------
    // 使用埃氏筛，上限 U = 46340（46340^2 > 2^31）
    for (int i = 2; i * i <= 46340; i++) {
        if (!isP[i]) {                       // i 是素数
            for (int j = i * i; j <= 46340; j += i) {
                isP[j] = true;               // 标记 i 的倍数为合数
            }
        }
    }
    // 收集所有素数到 pr 数组，下标从 1 开始
    cnt = 0;
    for (int i = 2; i <= 46340; i++) {
        if (!isP[i]) {
            pr[++cnt] = i;                   // pr[1], pr[2], ..., pr[cnt] 存储素数
        }
    }

    // ---------- 第 2 步：用这些小素数筛区间 [L, R] ----------
    // 遍历每个小素数 p
    for (int i = 1; i <= cnt; i++) {
        int p = pr[i];                       // 当前素数

        // 计算区间 [L,R] 内第一个 p 的倍数，且至少为 2*p（避免把 p 本身筛掉）
        // (L + p - 1) / p 是 ceil(L / p) 的整数计算，再乘 p 得到第一个 >= L 的倍数
        // 与 2*p 取最大值，保证不会标记素数 p 自己
        long long start = max((long long)p * 2, ((L + p - 1) / p) * p);

        // 从 start 开始，每次增加 p，将区间内 p 的倍数都标记为合数
        for (long long j = start; j <= R; j += p) {
            // 通过偏移量 j - L 定位到 seg 数组
            seg[j - L] = true;               // true 表示该数是合数
        }
    }

    // ---------- 第 3 步：统计区间内素数的个数 ----------
    int ans = 0;                             // 答案计数器
    for (long long i = L; i <= R; i++) {
        // 注意：1 不是素数，需要跳过
        // seg[i - L] == false 表示没被筛掉，可能是素数
        if (i >= 2 && !seg[i - L]) {
            ans++;
        }
    }

    // 输出结果
    cout << ans << '\n';                     // 用 '\n' 换行比 endl 更快

    return 0;
}

P1080 [NOIP 2012 提高组] 国王游戏

洛谷 | 提高| C++14 | 提交时间: 2026-02-08

算法标签：贪心、高精度、2012、NOIP 提高组、排序

题目描述

恰逢 H 国国庆，国王邀请 nnn 位大臣来玩一个有奖游戏。首先，他让每个大臣在左、右手上面分别写下一个整数，国王自己也在左、右手上各写一个整数。然后，让这 nnn 位大臣排成一排，国王站在队伍的最前面。排好队后，所有的大臣都会获得国王奖赏的若干金币，每位大臣获得的金币数分别是：排在该大臣前面的所有人的左手上的数的乘积除以他自己右手上的数，然后向下取整得到的结果。

国王不希望某一个大臣获得特别多的奖赏，所以他想请你帮他重新安排一下队伍的顺序，使得获得奖赏最多的大臣，所获奖赏尽可能的少。注意，国王的位置始终在队伍的最前面。

输入格式

第一行包含一个整数 nnn，表示大臣的人数。

第二行包含两个整数 aaa 和 bbb，之间用一个空格隔开，分别表示国王左手和右手上的整数。

接下来 nnn 行，每行包含两个整数 aaa 和 bbb，之间用一个空格隔开，分别表示每个大臣左手和右手上的整数。

输出格式

一个整数，表示重新排列后的队伍中获奖赏最多的大臣所获得的金币数。

样例

样例输入 #1

3 
1 1 
2 3 
7 4 
4 6 ```

**样例输出 #1**

2```

AC 代码

#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

// 大臣结构体
struct P {
    int l;  // 左手上的数字
    int r;  // 右手上的数字
    int v;  // l × r 的乘积（排序依据）
} a[1010];

// 比较函数：按 l×r 升序排序
bool cmp(P x, P y) {
    return x.v < y.v;
}

// 高精度乘法：大整数 x 乘以小整数 b
// x 是低位在前的向量（如123存储为[3,2,1]）
vector<int> mul(vector<int> x, int b) {
    vector<int> res;  // 存储结果
    int c = 0;        // 进位值
    
    // 逐位相乘并处理进位
    for (int i = 0; i < x.size(); i++) {
        c += x[i] * b;           // 当前位乘积加上进位
        res.push_back(c % 10);   // 取个位数存入结果
        c /= 10;                 // 计算新的进位
    }
    
    // 处理最后的进位
    while (c) {
        res.push_back(c % 10);
        c /= 10;
    }
    
    return res;
}

// 高精度除法：大整数 x 除以小整数 b（整除）
// x 是低位在前的向量，返回商（也是低位在前）
vector<int> div(vector<int> x, int b) {
    vector<int> res;  // 存储商
    int r = 0;        // 余数
    
    // 从高位向低位除（x中存储是低位在前，所以从最后一位开始）
    for (int i = x.size() - 1; i >= 0; i--) {
        r = r * 10 + x[i];       // 当前位的值加上上一位的余数×10
        res.push_back(r / b);    // 计算商的当前位
        r %= b;                  // 计算新的余数
    }
    
    // 此时res是高位在前，需要反转变成低位在前
    reverse(res.begin(), res.end());
    
    // 去除前导零（至少保留一位）
    while (res.size() > 1 && res.back() == 0) {
        res.pop_back();
    }
    
    return res;
}

// 比较两个高精度数的大小：返回 true 表示 x > y
bool gt(vector<int> x, vector<int> y) {
    // 先比较位数
    if (x.size() != y.size()) {
        return x.size() > y.size();
    }
    
    // 位数相同，从高位向低位比较
    for (int i = x.size() - 1; i >= 0; i--) {
        if (x[i] != y[i]) {
            return x[i] > y[i];
        }
    }
    
    return false;  // 两个数相等
}

int main() {
    int n;        // 大臣人数
    int kl, kr;   // 国王左手和右手上的数字
    
    // 输入
    cin >> n >> kl >> kr;
    
    // 输入每个大臣的左右手数字
    for (int i = 0; i < n; i++) {
        cin >> a[i].l >> a[i].r;
        a[i].v = a[i].l * a[i].r;  // 计算乘积用于排序
    }
    
    // 按 l×r 升序排序大臣
    sort(a, a + n, cmp);
    
    // 初始化前缀积 s：存储前面所有人左手数字的乘积
    // 从国王的左手数字开始
    vector<int> s;
    int t = kl;
    while (t) {
        s.push_back(t % 10);  // 转为高精度，低位在前
        t /= 10;
    }
    // 如果国王左手是0，则初始化为[0]
    if (s.empty()) {
        s.push_back(0);
    }
    
    // 记录最大金币数，初始为0
    vector<int> ans(1, 0);
    
    // 遍历排序后的大臣
    for (int i = 0; i < n; i++) {
        // 计算当前大臣的金币数：前面所有人的左手乘积 ÷ 当前大臣右手数字
        vector<int> cur = div(s, a[i].r);
        
        // 更新最大金币数
        if (gt(cur, ans)) {
            ans = cur;
        }
        
        // 更新前缀积：乘上当前大臣的左手数字
        s = mul(s, a[i].l);
    }
    
    // 输出结果：高精度数从高位向低位输出
    for (int i = ans.size() - 1; i >= 0; i--) {
        cout << ans[i];
    }
    cout << endl;
    
    return 0;
}

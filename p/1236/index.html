1236 - 二分查找

题目描述

请在一个有序递增数组中（不存在相同元素），采用二分查找，找出值 xxx 的位置，如果 xxx 在数组中不存在，请输出 -1 ！

输入

第一行，一个整数 nnn ，代表数组元素个数（n≤106n \le 10^6n≤106）

第二行，nnn 个数，代表数组的 nnn 个递增元素（1≤1 \le1≤数组元素值≤108\le 10^8≤108）

第三行，一个整数 xxx ，代表要查找的数（0≤x≤1080 \le x \le 10^80≤x≤108）

输出

xxx 在数组中的位置，或者 -1。

AC代码

AC代码

#include <iostream>
#include <bits/stdc++.h>
using namespace std;

int a[1000010],n,x;
int main() {
    int i,r = -1; // r用来记录找到的数的位置，如果没找到就返回-1
    int left,right,mid;
    // scanf("%d",&n);
    // ios::sync_with_stdio(false);
    // cin.tie(0);
    cin >> n;
    for(i = 0;i < n;i++) {
        scanf("%d",&a[i]);
        // cin >> a[i];
    }
    cin >> x;
    // 二分查找
    left = 0,right = n - 1;
    while(left <= right) {
        mid = (left + right) / 2;
        if(a[mid] == x) {
            r = mid + 1;
            break; // 找到就退出循环
        } else if(a[mid] < x) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    cout << r;
}

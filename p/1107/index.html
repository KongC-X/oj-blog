1107 - 求英文句子中的最长单词

题目描述

一个英文句子（长度不超过 $255$ ），只含有字母和空格，输出最长的一个单词。如有多个长度相同的单词，则输出最前面的一个。

输入

一个字符串。

输出

输出最长的单词。

AC代码

AC代码

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

// 字符数组相关函数
// sizeof(a) 求字符数组a所占内存空间大小
// strlen(a) 求字符串长度
// gets(a) 从键盘输入字符串
// puts(a) 输出字符串
// strcmp(a,b) 比较字符串大小,s1的字典码⼤返回正数差值；字典码相等返回0；s2的字
// 典码⼤返回负数差值；
// 什么是字典码？两个字符数组从第⼀个字符开始逐位⽐较ascii码的⼤⼩，如果第⼀位有结果，则⽐较结束，如果
// 第⼀位相同则顺序⽐较第⼆位。

// strcat(a,b) 将字符串b拼接到字符串a的末尾, 注意，s1要有⾜够的空间存

// strcpy(a,b) 将字符串b复制到字符串a中  strcpy(max,s); == max = s;

int main() {
    char s[300],a[300],b[300] = {'\0'};
    int i ,k = 0;
    gets(s);
    for(i = 0;i < strlen(s);i++) {
        //如果s[i]不是空格，那么就是单词的⼀部分，存到a中
        if(s[i] != ' ') {
            a[k] = s[i];
            k++;
            // 如果当前单词结束了
            if(s[i + 1] == ' ' || s[i + 1] == '\0') {
                a[k] = '\0';
                if(strlen(a) > strlen(b)) {
                    strcpy(b,a);  
                }
                // if(a[i] > max) {
                //     max = a[i];
                // }
                k = 0;
            }
        }
    }
    cout << b << endl;
}

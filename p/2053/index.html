2053 - 图的bfs遍历

题目描述

一个有 nnn 个结点的无向连通图，这些结点以编号：1,2…n1,2 \dots n1,2…n 进行编号，现给出结点间的连接关系。

请以结点 111 为起点，按广度优先搜索（bfs）、优先访问小编号结点的顺序遍历并输出该图。

输入

第一行为两整数，nnn 和 eee ，表示 nnn 个顶点，eee 条边。（ 2≤n,e≤102 \le n,e \le 102≤n,e≤10 ）

以下 eee 行每行两个数，表示两个结点是连通的。

输出

只有一行，为节点按照广度优先、小编号结点优先访问的结果。

AC代码

#include<iostream>
#include <bits/stdc++.h>
using namespace std;
 
int n,e;
int a[20][20];//邻接矩阵 
bool f[20];//节点是否访问 
int q[20];//存储走过的点
int head,tail; 
 
//访问x节点 
void bfs(){
    head = 1;
    tail = 1;
    q[1] = 1;//从1的点开始访问
    f[1] = true;//标记走过
    cout<<1<<" ";
    while(head <= tail){
        //尝试head能访问到的所有的点
        for(int i = 1;i <= n;i++){
            if(a[q[head]][i]==1&&!f[i]){
                //入队 
                tail++;
                f[i] = true;
                q[tail] = i;
                cout<<i<<" ";
            }
        } 
         
        head++;
    } 
}
 
int main(){
    cin>>n>>e;
    int x,y;
    //存储邻接矩阵
    for(int i = 1;i <= e;i++){
        cin>>x>>y;
        a[x][y] = 1;
        a[y][x] = 1;
    } 
     
    //从节点1开始访问 
    bfs(); 
}

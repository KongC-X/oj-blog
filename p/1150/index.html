1150 - 求完全数的个数

题目描述

一个大于 111 的正整数若等于全部因子的和，则称此数为完全数。例如： 666 有因子 1,2,31,2,31,2,3 。同时 6=1+2+36=1+2+36=1+2+3 ，所以 666 是完全数。

输入

一个正整数 NNN ( 10≤N≤10000010 \le N \le 10000010≤N≤100000 )

输出

小于等于 NNN 的完全数的个数。

解题思路

AC代码

AC代码

#include <iostream>
#include <cmath>
using namespace std;

// 求一个数的因子和
int sumOfFactors(int n) {
    int sum = 0;
    for (int i = 1; i <= sqrt(n); i++) {
        if (n % i == 0) {
            sum = sum + i + n / i;
        }
    }
    sum = sum - n; // 减去n本身
    return sum;
}

int main() {
    int n,i,s = 0;
    cin >> n;
    for(i=2;i<=n;i++) {
        if(sumOfFactors(i)==i) {
            s++;
        }
    }
    cout << s << endl;
    // cout << sumOfFactors(6);
}

6061 - 星际通信系统

题目描述

在星际通信系统中，传输的数字会因宇宙射线干扰而倒序，请编写程序对接收到的 nnn 个干扰数字进行修复，输出每个数字原始的正确值。

修复规则：将每个数字的各位数完全逆转（例如：123123123 修复为 321321321 ， 120120120 修复为 212121）

输入

第一行，一个整数 nnn。

第二行 nnn个正整数 aia_iai​，表示接收到的干扰数字。

输出

一行 nnn 个整数，表示修复后的正确数字，用空格分隔。

AC代码

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n;                        // n: 数字个数
    cin >> n;

    for (int i = 0; i < n; i++) {
        int x;                    // x: 当前接收到的干扰数字
        cin >> x;

        // 反转数字：不断取末尾数字加到结果中
        int y = 0;                // y: 修复后的数字
        while (x > 0) {
            y = y * 10 + x % 10;
            x /= 10;
        }
        // 若原数就是0，则反转后仍为0（while不进，y保持0）
        if (i == 0)
            cout << y;
        else
            cout << " " << y;
    }
    cout << endl;

    return 0;
}

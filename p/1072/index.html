1072 - 字符图形8-数字三角

题目描述

输入一个整数打印字符图形。

输入

一个整数（ $ 0 \lt n \lt 10$ ）。

输出

一个字符图形。

AC代码

AC代码

# include <iostream>
using namespace std;

int main() {
    int i,j,n;
    cin >> n;
    for(i = 1;i <= n;i++) {
        // 每行输出n-i个空格
        for(j = 1;j <= n - i;j++) {
            cout << " ";
        }
        // 每行输出2*i-1个数，数字为j
        for(j = 1;j <= 2 * i - 1;j++) {
            cout << j;
        }
        cout << endl;
    }
}

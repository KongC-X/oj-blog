1195 - 斜角IV

题目描述

输入整数 $N$ ，输出相应方阵。

输入

一个整数 $N$ 。（ $0 \lt N \le 10$ )

输出

一个方阵，每个数字的场宽为 $3$ 。

解题思路

AC代码

AC代码

#include <bits/stdc++.h>
using namespace std;
int a[20][20],n;
int main(){
    int i,j;
    cin>>n;
    for(i=1; i<=n; i++){
        for(j=1; j<=n-i+1; j++){
            a[i][n-j+1] = i+j-1; 
            a[n-j+1][i] = i+j-1;
            //cout<<i<<" "<<n-j+1<<" "<<i+j-1<<endl;
        }
    }
    
    for(i=1; i<=n;i++){
        for(j=1; j<=n; j++){
            cout<<setw(3)<<a[i][j];
        } 
        cout<<endl;
    }
    return 0;
}

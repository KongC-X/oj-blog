2355 - 进制转换

题目描述

【具体要求】

输入一个二进制数（位数最长不超过 15 位），将其转换成八进制后输出。

输入

输入一个二进制数。

输出

输出一个八进制数。

AC代码

AC代码

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

// 将三位的二进制转换为1位的八进制
int num(string s) {
    // 110，从最低位开始展开，先转换为10进制
    // 再转换为八进制
    int n,x = 0;
    for(int i = s.size() - 1; i >= 0; i--) {
        n = (s[i] - '0') * pow(2, s.size() - 1 - i);
        x += n;
    }
    return x;
}

int main(){
    string s;
    cin >> s;
    // 补0
    if(s.size() % 3 == 1) {
        s = "00" + s;
    } else if(s.size() % 3 == 2) {
        s = "0" + s;
    }
    // 截取输出对应的进制
    for(int i = 0;i < s.size();i += 3) {
        cout << num(s.substr(i, 3));
    }
}

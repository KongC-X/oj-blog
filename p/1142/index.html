1142 - 素数回文数

题目描述

如果一个数从左边读和右边读都是同一个数，就称为回文数，例如 686686686 就是一个回文数。

编程求 101010 到 100010001000 内所有的既是回文数同时又是素数的自然数。

输入

无

输出

若干个数，每行一个

解题思路

AC代码

AC代码

#include<iostream>
#include<cmath>
using namespace std;
int a[1010];
bool sushu(int n){  //判断素数函数 
    bool f = true;
    for(int i = 2;i <= sqrt(n);i++){
        if(n % i == 0){
            f = false;
            break;
        }
    } 
    //判断特殊情况
    if(n <= 1) f = false; 
    return f;
} 

int main() {
    int k=0;
    for(int i=10;i<=1000;i++){
        if(sushu(i)==true){
            a[k]=i;
            k++;
        }
    }
    for(int i=0;i<k;i++){
        if(a[i]>=10&&a[i]<=99){
            if(a[i]/10==a[i]%10){
                cout<<a[i]<<endl;
            }
        }else{
            if(a[i]/100==a[i]%10){
                cout<<a[i]<<endl;
            }	
        }
    }
    
}

1166 - 数的排序

题目描述

输入 nnn 个不超过 300003000030000 的整数（n≤10n \le 10n≤10 ）。然后求出每个数的数字和，再按每个数的数字和由小到大排列输出。

输入

第一行为整数 nnn；

第二行为 nnn 个整数。

输出

由小到大排列的每个数的数字和（每个数之间保留一个空格）。

AC代码

AC代码

# include <iostream>
using namespace std;

int main() {
    int a[20],i,n,s,b[20];
    cin >> n;
    for(i = 0; i < n; i++) {
        cin >> a[i];
        s = 0;
        while(a[i] != 0) {
            s += a[i] % 10;
            a[i] /= 10;
        }
        b[i] = s;
    }
    // 从小到大冒泡排序
    for(i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            if(b[j] > b[j + 1]) {
                int t = b[j];
                b[j] = b[j + 1];
                b[j + 1] = t;
            }
        }
    }
    for(i = 0; i < n; i++) {
        cout << b[i] << " ";
    }
}

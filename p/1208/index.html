1208 - 螺旋方阵

题目描述

输出如样例所示的 n×nn \times nn×n 的螺旋方阵。

输入

一个整数 nnn ( 0<n<100 \lt n \lt 100<n<10 )。

输出

一个 nnn 行方阵，每行 nnn 个数，每个数场宽为 333。

解题思路

AC代码

AC代码

#include <iostream>
#include <iomanip>
using namespace std;

int a[20][20],n,x = 1;
void fun(int start,int len) {
    if(start <= n * n) {
        int i,j;
        for(j = start;j <= start + len - 1;j++) {
            a[start][j] = x;
            x++;
        }
        for(i = start + 1;i <= start + len - 1;i++) {
            a[i][start + len - 1] = x;
            x++;
        }
        for(j = start + len - 2;j >= start;j--) {
            a[start + len - 1][j] = x;
            x++;
        }
        for(i = start + len - 2;i >= start + 1;i--) {
            a[i][start] = x;
            x++;
        }
        fun(start + 1,len - 2);
    }
}


int main() {
    cin >> n;
    fun(1,n);
    for(int i = 1;i <= n;i++) {
        for(int j = 1;j <= n;j++) {
            cout << setw(3) << a[i][j];
        }
        cout << endl;
    }
}

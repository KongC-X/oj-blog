1442 - 走出迷宫的最短路径

题目描述

有 n×mn \times mn×m 的迷宫，该迷宫有一个入口，一个出口。编写一程序打印一条从迷宫入口到出口的最短路径，黑色方块的单元表示走不通（用 111 表示），白色方块的内容表示走的通（用 000 表示）。

只能往上下左右四个方向走，如果有最短路径，保证最短路径一定是唯一的，如果没有路径可以到达，则输出“no way”。

输入

第一行输入 222 个整数 nnn 和 mmm ( nnn 和 mmm 都是 10∼15010 \sim 15010∼150 之间的整数)，代表迷宫的行数和列数；

接下来 nnn 行，每行有 mmm 个整数，111 代表不可走的点，000 代表可走的点；

接下来一行，有 222 个整数 s1s_1s1​ 和 s2s_2s2​ 代表入口的坐标；

接下来一行，有 222 个整数 e1e_1e1​ 和 e2e_2e2​ 代表出口的坐标；

本题数据上保证出发点和终点的值一定为 000，也就是不存在出发点和终点不能走的情况。

输出

输出从入口到出口的最短路径，如果没有路径可达输出“no way”。

// 思路：记录遍历过程中的每个点，以及每个点的来源下标，到终点时递归寻找每个点的来源,并打印下标
#include<iostream>
#include<bits/stdc++.h>
using namespace std;

// a存储矩阵，q存储队列（遍历过的点）
int a[200][200];
int q[40000][4];
int n,m,i,j,tx,ty;
int s1,s2,e1,e2; // 起始点
int head = 1,tail = 1; // head：头指针，tail：尾指针
// 四个方向,右下左上
int fx[5] = {0,0,1,0,-1};
int fy[5] = {0,1,0,-1,0};

// 打印路径
void print(int k) {
    // 如果前驱节点不为0，则递归找前驱节点
    if(q[k][3] != 0) {
        print(q[k][3]);
    }
    cout << "(" << q[k][1] << "," << q[k][2] << ")";
    if(k != tail) {
        cout << "->";
    }
}

int main() {
    cin >> n >> m;
    for(i = 1;i <= n;i++) {
        for(j = 1;j <= m;j++) {
            cin >> a[i][j];
        }
    }
    cin >> s1 >> s2 >> e1 >> e2;

    // 1,1点存入队列，以1，1为head访问四个方向
    q[1][1] = s1;
    q[1][2] = s2;
    q[1][3] = 0; // 前驱点，初始为0
    // 当head <= tail时，说明队列还有点没有尝试标记过四个方向
    while(head <= tail) {
        // 尝试访问head的四个方向
        for(i = 1;i <= 4;i++) {
            // 从head对应的点加上四个方向的变化量得到新的点
            tx = q[head][1] + fx[i];
            ty = q[head][2] + fy[i];
            // 如果新的点在矩阵范围内，且没有被标记过，则标记
            if(tx >= 1 && tx <= n && ty >= 1 && ty <= m && a[tx][ty] == 0) {
                a[tx][ty] = 1; // 标记代表走过了
                // 新的点存入队列，以新点为head访问四个方向
                tail++;
                q[tail][1] = tx;
                q[tail][2] = ty;
                q[tail][3] = head; // 记录该点的前驱点的行下标

                if(tx == e1 && ty == e2) { // 如果到达终点，输出步数
                    print(tail);
                    return 0;
                }
            }
        }
        head++; // head指向队列头的下一个点
    }
    cout << "no way" << endl; // 没有路径
}

1242 - 第K大与第K小数

题目描述

给定一个长度为 NNN ( 0<n≤100000 \lt n \le 100000<n≤10000 )的序列，保证每一个序列中的数字 a[i]a[i]a[i] 是正整数且唯一 ，编程要求求出整个序列中第 kkk 大的数字减去第 kkk 小的数字的值 mmm ，并判断 mmm 是否为质数。(0<k≤n0 \lt k \le n0<k≤n)

输入

输入格式：

第一行为 222 个数 n，kn，kn，k（含义如上题）；

第二行为 nnn 个数，表示这个序列。

输出

输出格式：

如果 mmm 为质数,则第一行为YES,第二行为这个数 mmm;

否则 ，第一行为NO ，第二行为这个数 mmm。

解题思路

AC代码

AC代码

#include <iostream>
using namespace std;
int a[10010];
int main() {
    int n,i,j,t,k,m;
    cin >> n >> k;
    for(i=0; i<n; i++) {
        cin>>a[i];
    }
    for(i = 0; i < n - 1; i++) {
        for(j = 0; j < n - i - 1; j++) {
            if(a[j] > a[j+1]) {
                t=a[j];
                a[j]=a[j+1];
                a[j+1]=t;
            }
        }
    }
    m = a[n-k] - a[k-1];
    // 判断m是否是质数
    bool flag = true;
    for(int i = 2; i <= m / 2; i++) {
        if(m % i == 0) {
            flag = false;
            break;
        }
    }
    if(m <= 1) {
        flag = false;  // 有可能m是负数
    }
    if(flag) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
    cout << m << endl;
}

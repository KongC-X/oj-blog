1242 - 第K大与第K小数

题目描述

给定一个长度为 $N$ ( $0 \lt n \le 10000$ )的序列，保证每一个序列中的数字 $a[i]$ 是正整数且唯一 ，编程要求求出整个序列中第 $k$ 大的数字减去第 $k$ 小的数字的值 $m$ ，并判断 $m$ 是否为质数。($0 \lt k \le n$)

输入

输入格式：

第一行为 $2$ 个数 $n，k$（含义如上题）；

第二行为 $n$ 个数，表示这个序列。

输出

输出格式：

如果 $m$ 为质数,则第一行为YES,第二行为这个数 $m$;

否则 ，第一行为NO ，第二行为这个数 $m$。

解题思路

AC代码

AC代码

#include <iostream>
using namespace std;
int a[10010];
int main() {
    int n,i,j,t,k,m;
    cin >> n >> k;
    for(i=0; i<n; i++) {
        cin>>a[i];
    }
    for(i = 0; i < n - 1; i++) {
        for(j = 0; j < n - i - 1; j++) {
            if(a[j] > a[j+1]) {
                t=a[j];
                a[j]=a[j+1];
                a[j+1]=t;
            }
        }
    }
    m = a[n-k] - a[k-1];
    // 判断m是否是质数
    bool flag = true;
    for(int i = 2; i <= m / 2; i++) {
        if(m % i == 0) {
            flag = false;
            break;
        }
    }
    if(m <= 1) {
        flag = false;  // 有可能m是负数
    }
    if(flag) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
    cout << m << endl;
}

1010 - 数组元素的排序

题目描述

对数组的元素按从小到大进行排序。

输入

第一行有一个整数 nnn ( 5≤n≤105 \le n \le 105≤n≤10 )；

第二行有 nnn 个整数，每个整数的值在 [0,109][0,10^9][0,109] 的范围内。

输出

输出排序后的数组。

AC代码

AC代码

# include <iostream>
using namespace std;

int main() {
    int a[20],i,n;
    cin >> n;
    for(i = 0; i < n; i++) {
        cin >> a[i];    
    }
    // 从小到大冒泡排序
    for(i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            if(a[j] > a[j + 1]) {
                int t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
            }
        }
    }
    for(i = 0; i < n; i++) {
        cout << a[i] << " ";
    }
}

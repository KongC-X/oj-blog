P1045 [NOIP 2003 普及组] 麦森数

洛谷 | 普及| C++14 | 提交时间: 2026-02-10

算法标签：数学、高精度、2003、NOIP 普及组

题目描述

形如 $2^{P}-1$ 的素数称为麦森数，这时 $P$ 一定也是个素数。但反过来不一定，即如果 $P$ 是个素数，$2^{P}-1$ 不一定也是素数。到 1998 年底，人们已找到了 37 个麦森数。最大的一个是 $P=3021377$，它有 909526 位。麦森数有许多重要应用，它与完全数密切相关。

任务：输入 $P(1000<P<3100000)$，计算 $2^{P}-1$ 的位数和最后 $500$ 位数字（用十进制高精度数表示）

输入格式

文件中只包含一个整数 $P(1000<P<3100000)$

输出格式

第一行：十进制高精度数 $2^{P}-1$ 的位数。

第 $2\sim 11$ 行：十进制高精度数 $2^{P}-1$ 的最后 $500$ 位数字。（每行输出 $50$ 位，共输出 $10$ 行，不足 $500$ 位时高位补 $0$）

不必验证 $2^{P}-1$ 与 $P$ 是否为素数。

1279

AC 代码

#include <iostream>
#include <bits/stdc++.h>
using namespace std;

const int N = 500;

int main() {
    int p;
    cin >> p;
    
    // 1. 位数
    cout << (int)(p * log10(2)) + 1 << endl;
    
    // 2. 计算2^p
    int ans[N] = {0};
    int base[N] = {0};
    ans[0] = 1;      // 结果初始为1
    base[0] = 2;     // 底数为2
    
    // 快速幂
    while (p > 0) {
        if (p & 1) {  // 如果当前位为1
            // ans = ans * base
            int temp[N] = {0};
            for (int i = 0; i < N; i++) {
                if (ans[i] == 0) continue;
                for (int j = 0; j < N && i + j < N; j++) {
                    temp[i + j] += ans[i] * base[j];
                }
            }
            // 进位
            for (int i = 0; i < N; i++) {
                if (temp[i] >= 10) {
                    if (i + 1 < N) temp[i + 1] += temp[i] / 10;
                    temp[i] %= 10;
                }
                ans[i] = temp[i];
            }
        }
        
        // base = base * base
        int temp[N] = {0};
        for (int i = 0; i < N; i++) {
            if (base[i] == 0) continue;
            for (int j = 0; j < N && i + j < N; j++) {
                temp[i + j] += base[i] * base[j];
            }
        }
        // 进位
        for (int i = 0; i < N; i++) {
            if (temp[i] >= 10) {
                if (i + 1 < N) temp[i + 1] += temp[i] / 10;
                temp[i] %= 10;
            }
            base[i] = temp[i];
        }
        
        p >>= 1;  // 右移一位
    }
    
    // 3. 减1
    ans[0] -= 1;
    for (int i = 0; i < N; i++) {
        if (ans[i] < 0) {
            ans[i] += 10;
            ans[i + 1] -= 1;
        }
    }
    
    // 4. 输出
    for (int i = 9; i >= 0; i--) {
        for (int j = 49; j >= 0; j--) {
            cout << ans[i * 50 + j];
        }
        cout << endl;
    }
    
    return 0;
}

1183 - 去除重复数字

题目描述

给你 NNN 个数（N≤100N≤100N≤100），每个数都在（0∼10000 \sim 10000∼1000）之间，其中由很多重复的数字，请将重复的数字只保留一个，并将剩下的数由小到大排序并输出。

输入

输入有 222 行；

第 111 行为 111 个正整数，表示数的个数：NNN；

第 222 行有 NNN 个用空格隔开的整数。

输出

第 111 行为 111 个正整数 MMM，表示不相同数的个数。

接下来的 MMM 行，每行一个整数，表示从小到大排好序的不相同的数。

解题思路

AC代码

AC代码

#include<iostream>
using namespace std;
int a[1010];
int main() {
    int n,i,m,k = 0;
    cin >> n;
    for(i = 0; i < n; i++) {
        cin >> m;
        bool flag = false; // 标记元素是否出现过
        for(int j = 0; j < k; j++) {
            if(a[j] == m) {
                flag = true;
                break;
            }    
        }
        if(flag == false) {
            a[k] = m;
            k++;
        }
    }
    cout << k << endl;
    // 排序
    for(i = 0;i < k - 1; i++) {
        for(int j = 0; j < k - i - 1; j++) {
            if(a[j] > a[j + 1]) {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }
    for(i = 0;i < k; i++) {
        cout << a[i] << endl;
    }
}

1587 - 连续非素数的最大长度

题目描述

给出一个正整数 nnn （ 2≤n≤10000002 \le n \le 10000002≤n≤1000000 ），例如 n=30n=30n=30 ，在 1，2，3，…301，2，3，\dots 301，2，3，…30 中，连续的非素数有:

444 长度为 111

666 长度为 111

888 999 101010 长度为 333

121212 长度为 111

141414 151515 161616 长度为 333

181818 长度为 111

202020 212121 222222 长度为 333

242424 252525 262626 272727 282828 长度为555

303030 长度为 111

其中，最大长度为 555 ，即有连续的 555 个非素数。

输入

一个整数 nnn 。

输出

一个整数，即连续非素数最大长度。

AC代码

AC代码

#include <iostream>
#include <cmath>
using namespace std;

const int MAXN = 5000005;  // 最大数据范围
bool isPrime[MAXN];        // true 表示是素数，false 表示不是素数
int n;

int main() {
    cin >> n;

    // 1. 埃氏筛法：标记素数
    // 初始化：假设所有数都是素数
    for (int i = 2; i <= n; i++) {
        isPrime[i] = true;
    }
    // 筛掉非素数
    for (int i = 2; i * i <= n; i++) {
        if (isPrime[i]) {
            for (int j = i * i; j <= n; j += i) {
                isPrime[j] = false;
            }
        }
    }

    // 2. 统计最长连续非素数长度
    int maxLen = 0;  // 记录最大长度
    int curLen = 0;  // 当前连续非素数长度

    for (int i = 2; i <= n; i++) {
        if (!isPrime[i]) {       // 如果是非素数
            curLen++;
            if (curLen > maxLen) {
                maxLen = curLen;
            }
        } else {                 // 如果是素数，中断连续
            curLen = 0;
        }
    }

    cout << maxLen << endl;

    return 0;
}

1273 - 哥德巴赫猜想的所有解

题目描述

求出哥德巴赫猜想的所有解（将一个大于 999 的奇数拆分成三个素数之和），并按从小到大的顺序写出。

输入

一行，一个大于 999 的奇数。

输出

第一行，一个整数 NNN（ N≤1200N \le 1200N≤1200 ），表示解的总数。

第 2−N+12-N+12−N+1 行，每行一个解。

解题思路

AC代码

AC代码

#include<bits/stdc++.h>
using namespace std;
//判断素数
bool sushu(int n) {
    bool f = true;//假设是素数
    //循环n的因子范围
    for(int i = 2; i <= sqrt(n); i++) {
        //如果找到n的因子，n就不是素数
        if(n % i == 0) {
            f = false;
            break;
        }
    }

    //判断特殊情况
    if(n <= 1) f = false;
    return f;
}

//存储所有素数和的拆分方案 
int a[100000][3];
int n,k = 0;//k代表数组下标 
int main() {
    cin>>n;
    //找第1个素数
    for(int i = 1; i <= n / 3; i++) {
        if(sushu(i)==true) {
            //找第2个素数
            for(int j = i; j <= (n - i) / 2; j++) {
                //如果j和n-i-j是素数 
                if(sushu(j)==true&&sushu(n-i-j)==true) {
                    //cout<<n<<"="<<i<<"+"<<j<<"+"<<n-i-j<<endl;
                    a[k][0] = i;
                    a[k][1] = j;
                    a[k][2] = n - i - j;
                    k++;//下标自增，存下一个方案 
                }

            }
        }
    }
    
    cout<<k<<endl;
    //输出所有的方案
    for(int i = 0;i < k;i++){
        cout<<n<<"="<<a[i][0]<<"+"<<a[i][1]<<"+"<<a[i][2]<<endl;
    } 
    return 0;
}

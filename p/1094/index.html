1094 字符图形10-字母三角

1094 - 字符图形10-字母三角

题目描述

输入一个整数打印字符图形。

输入

一个整数。（ 0<n<100 \lt n \lt 100<n<10）

输出

一个字符图形。

AC代码

AC代码

#include<iostream>
using namespace std;

int main() {
    // 我们先来打印数字三角
    //     1
    //    222
    //   33333
    // n = 3
    // 第1行 2个空格 1个数字
    // 第2行 1个空格 3个数字
    // 第3行 0个空格 5个数字
    // 第i行 n-i个空格 2*i-1个数字
    int i,j,n;
    char c;
    cin >> n;
    // 循环控制行数
    for(i = 1;i <= n;i++) {
        // 打印每一行的空格数
        for(j = 1;j <= n - i;j++) {
            cout << " ";
        }
        // 打印每一行的数字
        for(j = 1;j <= 2 * i - 1;j++) {
            // cout << i;
            // 1->'A' 2->'B' 3->'C'
            c = i + 64;
            cout << c;
        }
        // 打印每一行的换行符
        cout << endl;
    }
}

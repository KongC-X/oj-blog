1358 - 素数环

题目描述

从 $1 \sim n$ 这 $n$ 个数，摆成一个环，要求相邻的两个数的和是素数，按照由小到大请输出所有可能的摆放形式。

比如：$n = 4$，输出形式如下；

`1:1 2 3 4
2:1 4 3 2
3:2 1 4 3
4:2 3 4 1
5:3 2 1 4
6:3 4 1 2
7:4 1 2 3
8:4 3 2 1
total:8
`

比如：$n = 6$，输出形式如下；

`1:1 4 3 2 5 6
2:1 6 5 2 3 4
3:2 3 4 1 6 5
4:2 5 6 1 4 3
5:3 2 5 6 1 4
6:3 4 1 6 5 2
7:4 1 6 5 2 3
8:4 3 2 5 6 1
9:5 2 3 4 1 6
10:5 6 1 4 3 2
11:6 1 4 3 2 5
12:6 5 2 3 4 1
total:12
`

输入

一个整数 $n$ ；（$2 \le n \le 10$）

输出

前若干行，每行输出一个素数环的解，最后一行，输出解的总数。

解题思路

AC代码

AC代码

#include<iostream> 
using namespace std;

int n,c;
int a[20];
bool f[20];

bool isPrime(int x) {
    if(x < 2) return false;
    for(int i=2; i*i<=x; i++) {
        if(x%i==0) return false;
    }
    return true;
}

void print() {
    c++;
    cout << c << ":";
    for(int i = 1; i < n; i++) {
        cout << a[i] << " ";
    }
    cout << a[n] << endl;
}


void dfs(int k) {
    for(int i = 1;i <= n;i++) {
        if(f[i] == false && (k == 1 || isPrime(a[k-1] + i))) {
            a[k] = i;
            f[i] = true;
            if(k == n && isPrime(a[n] + a[1])) {
                print();
            } else {
                dfs(k+1);
            }
            f[i] = false;
        }
    }
}

int main() {
    cin >> n;
    dfs(1);
    cout << "total:" << c << endl;
}

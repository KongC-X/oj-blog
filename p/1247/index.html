1247 - 打印n行的完整的蝴蝶结

题目描述

请从键盘读入一个整数 nnn（nnn 是 1∼101 \sim 101∼10 的范围内的奇数），打印出如下图所示的 nnn 行的完整的蝴蝶结！

比如，n=5n=5n=5，则打印图形如下：

n=7n=7n=7:

输入

一个整数 nnn ，代表图形的行数！

输出

nnn 行的图形！

解题思路

AC代码

AC代码

#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int n;
    cin>>n;
    int i,j;
    //上半部分循环 n/2+1行 
    for(i=1;i<=n/2+1;i++){
        //先遍历输出* 
        for(j=1;j<=i;j++){
            cout<<"*";
        }
        //遍历输出空格 
        for(j=1;j<=n-i*2+1;j++){
            cout<<" ";
        }
        //遍历输出* 
        for(j=1;j<=i;j++){
            cout<<"*";
        }
        cout<<endl;
    }
    //上半部分的逆序 
    for(i=n/2;i>=1;i--){
        for(j=1;j<=i;j++){
            cout<<"*";
        }
        for(j=1;j<=n-i*2+1;j++){
            cout<<" ";
        }
        for(j=1;j<=i;j++){
            cout<<"*";
        }
        cout<<endl;
    }
    return 0;
}

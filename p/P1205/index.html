P1205 [USACO1.2] Transformations

洛谷 | 普及| C++14 | 提交时间: 2025-08-02

算法标签：模拟、搜索、USACO

题目描述

A square pattern of size N×NN \times NN×N (1≤N≤101 \le N \le 101≤N≤10) black and white square tiles is transformed into another square pattern. Write a program that will recognize the minimum transformation that has been applied to the original pattern given the following list of possible transformations:

  • #1: 90 Degree Rotation: The pattern was rotated clockwise 909090 degrees.
  • #2: 180 Degree Rotation: The pattern was rotated clockwise 180180180 degrees.
  • #3: 270 Degree Rotation: The pattern was rotated clockwise 270270270 degrees.
  • #4: Reflection: The pattern was reflected horizontally (turned into a mirror image of itself by reflecting around a vertical line in the middle of the image).
  • #5: Combination: The pattern was reflected horizontally and then subjected to one of the rotations (#1-#3).
  • #6: No Change: The original pattern was not changed.
  • #7: Invalid Transformation: The new pattern was not obtained by any of the above methods.

In the case that more than one transform could have been used, choose the one with the minimum number above.

输入格式

  • Line 1: A single integer, NNN.
  • Line 2..N+1N+1N+1: NNN lines of NNN characters (each either @ or -); this is the square before transformation.
  • Line N+2N+2N+2..2N+12N+12N+1: NNN lines of NNN characters (each either @ or -); this is the square after transformation.

输出格式

A single line containing the number from 111 through 777 described above that categorizes the transformation required to change from the "before" representation to the "after" representation.

样例

样例输入 #1

3
@-@
---
@@-
@-@
@--
--@

样例输出 #1

1

#include<bits/stdc++.h>
#include<iostream>
using namespace std;

int n;
char a[15][15],b[15][15],c[15][15];

bool work0(){//判断是否相等
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            if(b[i][j]!=c[i][j])
              return false;
          }    
    }
    return true;
}

bool work1(){
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)
        b[j][n-i+1]=a[i][j];
    }
    if(work0())return true;
    return false;
}
bool work2(){
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)
        b[n-i+1][n-j+1]=a[i][j];
    }
    if(work0())return true;
    return false;
}
bool work3(){
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)
        b[n-j+1][i]=a[i][j];
    }
    if(work0())return true;
    return false;
}
bool work4(){
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)
        b[i][n-j+1]=a[i][j];
    }
    if(work0())return true;
    return false;
}
bool work5(){
    work4();
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            a[i][j]=b[i][j]; 
        }
    }   
    if(work1())return 1;
       
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)
        a[i][j]=b[i][j]; 
    }
    if(work2())return 1;
    
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++)
        a[i][j]=b[i][j]; 
    }
    if(work3())return 1;
    
    return false;
}

signed main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            cin>>a[i][j];	
        }
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            cin>>c[i][j];
        }
    }
    if(work1()){
        cout<<1;
        return 0;
    }
    if(work2()){
        cout<<2;
        return 0;
    }
    if(work3()){
        cout<<3;
        return 0;
    }
    if(work4()){
        cout<<4;
        return 0;
    }
    if(work5()){
        cout<<5;
        return 0;
    }
    if(work0()){
        cout<<6;
        return 0;
    }
    cout<<7;
    return 0;
}

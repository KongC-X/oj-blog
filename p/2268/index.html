2268 - 求逆序对数

题目描述

对于一个长度为 NNN 的整数序列 AAA，满足 且 Ai>AjA_i > A_jAi​>Aj​ 的数对 (i,j)(i,j)(i,j) 称为整数序列 AAA 的一个逆序对；

请求出整数序列 AAA 的所有逆序对个数。

输入

输入包含多组测试数据，每组测试数据有两行。

第一行为整数 NNN(1≤N≤200001 \le N \le 200001≤N≤20000)，当输入 000 时结束。

第二行为 NNN 个整数，表示长为 NNN 的整数序列。（这 NNN 个数在 [−109,109][-10^9, 10^9][−109,109] 的范围内）

输出

每组数据对应一行，输出逆序对的个数。

AC代码

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 20005;
int a[MAXN];                  // 原始序列
int tmp[MAXN];                // 归并临时数组
long long ans;                // 逆序对总数（可能较大）

// 合并两个有序子数组，并统计逆序对
void merge(int l, int mid, int r) {
    int i = l, j = mid + 1, k = l;
    while (i <= mid && j <= r) {
        if (a[i] <= a[j]) {
            tmp[k++] = a[i++];
        } else {
            // a[i] > a[j]，那么 a[i..mid] 都 > a[j]，形成逆序对
            ans += (mid - i + 1);
            tmp[k++] = a[j++];
        }
    }
    while (i <= mid) tmp[k++] = a[i++];
    while (j <= r)   tmp[k++] = a[j++];
    for (int p = l; p <= r; p++) a[p] = tmp[p];
}

// 归并排序递归函数
void mergeSort(int l, int r) {
    if (l >= r) return;
    int mid = (l + r) / 2;
    mergeSort(l, mid);
    mergeSort(mid + 1, r);
    merge(l, mid, r);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n;                     // 序列长度
    while (cin >> n && n != 0) {
        for (int i = 1; i <= n; i++) cin >> a[i];

        ans = 0;
        mergeSort(1, n);
        cout << ans << '\n';
    }
    return 0;
}

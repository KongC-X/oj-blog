1209 - 回形方阵

题目描述

输入 $n$ 打印回形方阵。

输入

一个整数 $n$ （ $0 \lt n \lt 10$）

输出

一个方阵，每个数字的场宽为 $2$

AC代码

AC代码

#include<iostream>
#include<iomanip>
using namespace std;

int a[30][30];
int n;

void fun(int start,int len,int x) {
    if(len > 0) {
        int i,j;
        // 向右
        for(j = start;j <= start + len - 1;j++) {
            a[start][j] = x;
        }
        // 向下
        for(i = start + 1;i <= start + len - 1;i++) {
            a[i][start + len - 1] = x;
        }
        // 向左
        for(j = start + len - 2;j >= start;j--) {
            a[start + len - 1][j] = x;
        }
        // 向上
        for(i = start + len - 2;i >= start + 1;i--) {
            a[i][start] = x;
        }
        fun(start + 1,len - 2,--x);
    }
}

int main() {
    cin >> n;
    fun(1, 2 * n + 1, n);
    for(int i = 1;i <= 2 * n + 1;i++) {
        for(int j = 1;j <= 2 * n + 1;j++) {
            cout << setw(2) << a[i][j];
        }
        cout << endl;
    }
}

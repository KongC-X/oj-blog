#include<bits/stdc++.h>
using namespace std;

int main() {
    //定义变量a,b,c,d代表四门成绩 
    int a,b,c,d;
    //输入四门成绩 
    cin>>a>>b>>c>>d;
    //只要有一门低于60就输出 "Poor LanYangYang"
    if(a<60||b<60||c<60||d<60){
        cout<<"Poor LanYangYang";
    }else{
        //计数大于等于90的学科的数量 
        int num = 0;
        if(a>=90){
            num++;
        }
        if(b>=90){
            num++;
        }
        if(c>=90){
            num++;
        }
        if(d>=90){
            num++;
        }
        
        //根据A等级学科的数量输出对应的加分 
        if(num==4){
            cout<<5;
        }else{
            cout<<num;
        } 
    }
    return 0;
}

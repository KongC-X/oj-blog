B4319 [语言月赛 202504] 礼堂预约

洛谷 | 普及| C++14 | 提交时间: 2026-08-04

算法标签：模拟、2025、排序、数组、语言月赛

题目描述

小 M 所在的学校有一个礼堂，礼堂里至多能同时进行一个活动，因此需要预约。每个活动占据某一天的上午、下午或者晚上。

礼堂负责人按照时间顺序接收到了 nnn 个活动预约，其中第 iii 个活动预约描述了活动如下信息：

  • 活动类型 typeitype_itypei​：一个大写字母，O 表示学校官方活动，C 表示社团活动，P 表示个人活动。
  • 活动日期 dateidate_idatei​：使用 YYYYMMDD 格式表示，例如 20250411 表示 202520252025 年 444 月 111111 日。
  • 活动时间 timeitime_itimei​：一个大写字母，M 表示上午，A 表示下午，E 表示晚上。

负责人每收到一个新的活动预约，会直接加入日程，接下来重复执行如下流程，直到日程里没有冲突：

  • 我们规定学校官方活动优先于社团活动优先于个人活动。如果冲突的两个活动预约类型不同，那么把优先程度更低的活动向后挪一天（活动时间不变）。
  • 如果冲突的两个活动类型相同，那么遵循先到先得的原则，把较后预约的活动向后挪一天。
  • 注意向后挪动一天可能引发新的冲突，此时对于新冲突的两个活动，仍然按照上述方案调整。

给出这 nnn 个活动预约的信息，请输出最终每个活动在日程里的活动日期，按照活动预约的提交顺序输出。

输入格式

输入的第一行有一个正整数 nnn，表示活动个数。

之后 nnn 行，每行依次输入一个大写字母 typeitype_itypei​、一个八位数日期 dateidate_idatei​ 和一个大写字母 timeitime_itimei​，表示一个活动预约，具体含义同题目描述。

输出格式

输出 nnn 行，每行一个八位数日期，其中第 iii 行表示的是第 iii 个提交的活动预约最终的日期。

#include <bits/stdc++.h>
using namespace std;

/*
 * 【题目类型】模拟 + 冲突解决（队列化插入）
 * 【核心思路】
 * 1. 按提交顺序逐个处理活动，但用队列来管理“待插入”的活动。
 * 2. 用 map<int,int> occ 记录当前已占用的（日期+时间段）→ 活动编号。
 * 3. 每次从队列取出一个活动 cur，尝试插入：
 *    - 若位置空闲，直接插入并继续处理队列中下一个。
 *    - 若位置已被活动 j 占用，则比较 cur 和 j 的优先级：
 *      * 类型不同时，优先级低的（数字小）被挪动；
 *      * 类型相同时，后提交的（id大）被挪动。
 *    - 被挪动的活动日期+1，并重新加入队列（因为它需要重新找位置）。
 *    - 另一个活动（未挪动的）也需重新加入队列，因为原位置可能已空。
 * 4. 重复直到队列为空，所有活动都安置完毕。
 * 5. 最后按原始提交顺序输出每个活动的最终日期。
 *
 * 【时间复杂度】每个活动被挪动的次数有限，总操作数远小于 O(n²)，n=5000 通过。
 */

struct Date {
    int y, m, d;   // 年、月、日
};

struct Act {
    int id;         // 提交顺序编号（从0开始）
    int type;       // 优先级：O=3, C=2, P=1
    Date date;      // 当前日期
    int time;       // 时间段：M=0, A=1, E=2
};

int n;              // 活动总数
Act a[5005];        // a[i] 存储第 i 个活动（按提交顺序）

// 闰年判断
bool isLeap(int y) {
    return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
}

// 某年某月的天数
int daysInMonth(int y, int m) {
    if (m == 2) return isLeap(y) ? 29 : 28;
    if (m == 4 || m == 6 || m == 9 || m == 11) return 30;
    return 31;
}

// 日期加一天
Date nextDay(Date d) {
    d.d++;
    if (d.d > daysInMonth(d.y, d.m)) {
        d.d = 1;
        d.m++;
        if (d.m > 12) {
            d.m = 1;
            d.y++;
        }
    }
    return d;
}

// 将日期和时间段转换为唯一整数 key（用于 map）
int getKey(Date d, int t) {
    int dateInt = d.y * 10000 + d.m * 100 + d.d;  // 如 20250419
    return dateInt * 3 + t;    // *3 区分 M(0), A(1), E(2)
}

// 决定冲突双方谁该挪动，返回需要挪动的活动编号
int decide(int i, int j) {
    // 类型不同：优先级低的（数字小）挪动
    if (a[i].type != a[j].type) {
        return (a[i].type < a[j].type) ? i : j;
    }
    // 类型相同：后提交的（id大）挪动
    return (a[i].id > a[j].id) ? i : j;
}

int main() {
    // 读入活动数量
    cin >> n;

    // 读入每个活动的原始信息
    for (int i = 0; i < n; i++) {
        char typeChar, timeChar;
        string dateStr;
        cin >> typeChar >> dateStr >> timeChar;

        a[i].id = i;

        // 类型转数字
        if (typeChar == 'O') a[i].type = 3;
        else if (typeChar == 'C') a[i].type = 2;
        else a[i].type = 1;   // 'P'

        // 解析日期
        a[i].date.y = stoi(dateStr.substr(0, 4));
        a[i].date.m = stoi(dateStr.substr(4, 2));
        a[i].date.d = stoi(dateStr.substr(6, 2));

        // 时间段转数字
        if (timeChar == 'M') a[i].time = 0;
        else if (timeChar == 'A') a[i].time = 1;
        else a[i].time = 2;   // 'E'
    }

    map<int, int> occ;   // 当前已占用的 (key → 活动编号)
    queue<int> q;        // 待插入的活动队列

    // 按提交顺序将每个活动加入队列
    for (int i = 0; i < n; i++) {
        q.push(i);

        // 处理队列中的所有待插入活动（包括因冲突被挪动的）
        while (!q.empty()) {
            int cur = q.front();
            q.pop();

            int key = getKey(a[cur].date, a[cur].time);

            if (occ.find(key) == occ.end()) {
                // 无冲突，直接插入
                occ[key] = cur;
            } else {
                // 冲突，取出已占用的活动编号
                int j = occ[key];
                int moveId = decide(cur, j);   // 决定谁挪动

                if (moveId == cur) {
                    // 挪动的是当前活动 cur（尚未插入）
                    a[cur].date = nextDay(a[cur].date);
                    q.push(cur);   // 重新尝试插入
                    // 注意：occ 中仍然保留 j，因为 cur 被挪走，位置不变
                } else {
                    // 挪动的是已存在的活动 j
                    occ.erase(key);                // 先删除原占用
                    a[j].date = nextDay(a[j].date);
                    q.push(j);                    // j 需要重新找位置
                    q.push(cur);                  // cur 也要重新尝试插入
                    // 因为原位置空了，cur 可能可以插入，所以一起加回队列
                }
            }
        }
    }

    // 按原始提交顺序输出最终日期
    for (int i = 0; i < n; i++) {
        cout << setw(4) << setfill('0') << a[i].date.y;
        cout << setw(2) << setfill('0') << a[i].date.m;
        cout << setw(2) << setfill('0') << a[i].date.d << endl;
    }

    return 0;
}

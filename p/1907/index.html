1907 - 有多少细胞

题目描述

一矩形阵列由数字 000 到 999 组成，数字 111 到 999 代表细胞，细胞的定义为沿细胞数字上下左右若还是细胞数字则为同一细胞,求给定矩形阵列的细胞个数。（1≤m,n≤1001≤m,n≤1001≤m,n≤100）?

输入

整数 m,nm,nm,n(mmm 行，nnn 列)；

接下来 mmm 行，每行有 nnn 个 000~999 之间的整数；

输出

细胞的个数。

AC代码

AC代码

#include<bits/stdc++.h>
using namespace std;
int n, m;
char a[110][110];
int  r[100100][3];
int fx[] = {0, 1, 0, -1};
int fy[] = {1, 0, -1, 0};

void bfs(int x, int y)
{
    memset(r, 0, sizeof(r));
    int head = 0, tail = 0;
    r[head][0] = x;
    r[head][1] = y;
    a[x][y] = '0';
    while(head <= tail)
    {
        for(int i = 0; i < 4; i++)
        {
            int dx = r[head][0] + fx[i];
            int dy = r[head][1] + fy[i];
            if(a[dx][dy] > '0' && a[dx][dy] <= '9')
            {
                tail++;
                r[tail][0] = dx;
                r[tail][1] = dy;
                a[dx][dy] = '0';
            }
        }
        head++;
    }

}

int main()
{
    cin >> n >> m;
    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < m; j++)
        {
            cin >> a[i][j];
        }
    }
    int cnt = 0;
    for(int i = 0; i < n; i++)
    {
        for(int j = 0; j < m; j++)
        {
            if(a[i][j] > '0' && a[i][j] <= '9')
            {
                cnt++;
                bfs(i, j);
            }
        }
    }
    cout << cnt;
    return 0;
}

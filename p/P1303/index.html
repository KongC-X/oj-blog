P1303 A*B Problem

洛谷 | 普及- | C++14 | 提交时间: 2025-08-11

算法标签：模拟、高精度

题目描述

给出两个非负整数，求它们的乘积。

输入格式

输入共两行，每行一个非负整数。

输出格式

输出一个非负整数表示乘积。

样例

样例输入 #1

1 
2```

**样例输出 #1**

2```

AC 代码

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

string s1,s2;
int a[5010],b[5010],c[10010];
int i,j,p;
int main() {
    cin >> s1 >> s2;
    // 逆序存储
    for(i = s1.size() - 1;i >= 0;i--) {
        a[i] = s1[s1.size() - 1 - i] - '0';
    }
    for(i = s2.size() - 1;i >= 0;i--) {
        b[i] = s2[s2.size() - 1 - i] - '0';
    }

    // 循环a数组的每一位，用a[i]去乘b数组的每一位b[i]
    // 结果错位加到c数组的c[i+j]这一位上
    for(i = 0;i < s1.size();i++) {
        for(j = 0;j < s2.size();j++) {
            c[i+j] += a[i] * b[j];
            // 进位
            if(c[i+j] >= 10) {
                c[i+j+1] += c[i+j] / 10;
                c[i+j] %= 10;
            }
        }
    }

    // 去除前导0
    for(i = s1.size() + s2.size() - 1;i >= 0;i--) {
        if(c[i] != 0) {
            p = i;
            break;
        }
    }
    // 逆序输出结果
    for(i = p;i >= 0;i--){
        cout << c[i];
    }
}

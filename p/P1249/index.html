P1249 最大乘积

洛谷 | 普及| C++14 | 提交时间: 2026-02-10

算法标签：贪心、高精度

题目描述

一个正整数一般可以分为几个互不相同的自然数的和，如 3=1+23=1+23=1+2，4=1+34=1+34=1+3，5=1+4=2+35=1+4=2+35=1+4=2+3，6=1+5=2+46=1+5=2+46=1+5=2+4。

现在你的任务是将指定的正整数 nnn 分解成若干个互不相同的自然数（也可以不分解，就是这个数字本身）的和，且使这些自然数的乘积最大。

输入格式

只有一个正整数 nnn，（3≤n≤100003 \leq n \leq 100003≤n≤10000）。

输出格式

第一行是分解方案，相邻的数之间用一个空格分开，并且按由小到大的顺序。

第二行是最大的乘积。

样例

样例输入 #1

10

样例输出 #1

2 3 5
30```

## AC 代码

```cpp
#include <iostream>
#include <string>
#include <cstring>
using namespace std;

// 高精度乘法函数
string cheng(string s1, string s2) {
    int a[5010] = {0}, b[5010] = {0}, c[10010] = {0};
    int i, j, p = 0;
    string s3 = "";
    
    // 逆序存储
    for(i = s1.size() - 1; i >= 0; i--) {
        a[i] = s1[s1.size() - 1 - i] - '0';
    }
    for(i = s2.size() - 1; i >= 0; i--) {
        b[i] = s2[s2.size() - 1 - i] - '0';
    }

    // 循环a数组的每一位，用a[i]去乘b数组的每一位b[j]
    // 结果错位加到c数组的c[i+j]这一位上
    for(i = 0; i < s1.size(); i++) {
        for(j = 0; j < s2.size(); j++) {
            c[i + j] += a[i] * b[j];
            // 进位
            if(c[i + j] >= 10) {
                c[i + j + 1] += c[i + j] / 10;
                c[i + j] %= 10;
            }
        }
    }
    
    // 去除前导0
    for(i = s1.size() + s2.size() - 1; i >= 0; i--) {
        if(c[i] != 0) {
            p = i;
            break;
        }
    }
    
    // 逆序输出结果
    for(i = p; i >= 0; i--) {
        s3 += to_string(c[i]);
    }
    return s3;
}

int main() {
    int n;
    cin >> n;
    
    // 特殊情况：n <= 4
    if (n <= 4) {
        cout << n << endl;
        cout << n << endl;
        return 0;
    }
    
    // 贪心分解：从2开始连续取数
    int a[200];  // 存储分解的数，n最大10000，分解的数不会超过200个
    int count = 0;  // 分解的数的个数
    int sum = 0;    // 当前和
    
    // 从2开始连续取数，直到不能再取
    for (int i = 2; sum + i <= n; ++i) {
        a[count] = i;
        sum += i;
        count++;
    }
    
    // 剩余的数
    int r = n - sum;
    
    // 从后往前分配剩余的数
    int index = count - 1;  // 从最后一个数开始
    while (r > 0) {
        a[index] += 1;
        r--;
        index--;
        
        // 如果已经加到第一个数，但还有剩余，回到最后一个数继续
        if (index < 0 && r > 0) {
            index = count - 1;
        }
    }
    
    // 输出分解方案
    for (int i = 0; i < count; ++i) {
        if (i > 0) cout << " ";
        cout << a[i];
    }
    cout << endl;
    
    // 计算最大乘积（高精度）
    string p = "1";  // 初始化为1
    
    // 逐个乘以分解的数
    for (int i = 0; i < count; ++i) {
        // 将整数转换为字符串
        string s1 = to_string(a[i]);
        p = cheng(p, s1);
    }
    
    // 输出乘积
    cout << p << endl;
    
    return 0;
}

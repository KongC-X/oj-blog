2207 - 树的中心

题目描述

给定一棵树，树中有 nnn 个结点（结点编号为 1∼n1 \sim n1∼n ），请求出该树的中心结点的编号。树的中心指的是，该结点离树中的其他结点，最远距离最近。

比如，按照树的中心的定义，下图中的结点 111 和结点 333 ，就是该树的中心。

输入

第 111 行包括一个整数 nnn（n≤105n≤10^5n≤105），代表树中结点的数量。

接下来 n−1n-1n−1 行，每行有 222 个整数 xxx 和 yyy，代表结点 xxx 到结点 yyy 之间有一条边。（不确定结点之间的父子关系）

输出

请输出树的中心结点的编号，如果有多个中心结点，请按照从小到大的顺序输出所有中心结点的编号。

AC代码

#include <iostream>
#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;

// ---------- 链式前向星（邻接表） ----------
int pre[N];          // pre[x]：结点 x 的"最后一条出边"在 a 数组中的下标
struct Edge {
    int to;          // 这条边指向的目标结点
    int next;        // 上一条以相同起点出发的边的下标（链表中的 next）
} a[N * 2];          // 存边的数组，无向图每条边存两次 → 2*(n-1) 条边

int k = 0;           // 当前已使用的 a 数组下标（从 1 开始计）
int n;              // 结点数

// 加一条从 u 到 v 的有向边
void add(int u, int v) {
    k++;                // 分配一个新下标
    a[k].to = v;        // 这条边指向 v
    a[k].next = pre[u]; // 新边的 next 指向 u 原来的"最后一条边"
    pre[u] = k;         // 更新 u 的"最后一条边"为当前新边
}
// -------------------------------------------------

int fa[N];   // fa[x]：结点 x 的父结点（在 DFS 树中）
int dep[N];  // dep[x]：结点 x 在 DFS 树中的深度

/**
 * DFS 遍历树
 * @param x 当前结点
 * @param f 当前结点的父结点
 */
void dfs(int x, int f) {
    fa[x] = f;                 // 记录父结点
    dep[x] = dep[f] + 1;       // 深度 = 父结点深度 + 1

    // 遍历 x 的所有邻接点（即与 x 有边相连的结点）
    // i = pre[x]：从 x 的"最后一条边"开始
    // i != 0：0 表示链表的末尾（没有更多边了）
    // i = a[i].next：跳到上一条以 x 为起点的边
    for (int i = pre[x]; i != 0; i = a[i].next) {
        int v = a[i].to;       // v 是 x 的邻居
        if (v != f) {          // 不能走回父结点
            dfs(v, x);          // 递归遍历 v
        }
    }
}

int main() {
    cin >> n;

    // 读入 n-1 条边，建图
    for (int i = 1; i <= n - 1; i++) {
        int x, y;
        cin >> x >> y;
        add(x, y);   // 加一条 x→y 的边
        add(y, x);   // 加一条 y→x 的边（无向图双向建边）
    }

    // ========== 第一步：找直径的第一个端点 ==========
    // 从 1 号点开始 DFS，求出每个结点的深度
    dfs(1, 0);                     // 1 的父为 0（不存在）
    
    int x = 1;
    for (int i = 2; i <= n; i++) {
        if (dep[i] > dep[x]) {     // 找深度最大的结点
            x = i;                 // x 就是离 1 最远的点 → 直径的一个端点
        }
    }

    // ========== 第二步：找直径的第二个端点 ==========
    // 以 x 为根（父为 0）重新 DFS，求出每个结点到 x 的深度
    dfs(x, 0);
    
    int y = 1;
    for (int i = 2; i <= n; i++) {
        if (dep[i] > dep[y]) {     // 找深度最大的结点
            y = i;                 // y 是离 x 最远的点 → 直径的另一个端点
        }
    }
    // 此时，x 到 y 的路径就是树的直径
    // dep[y] 就是直径的长度（边数 + 1）

    // ========== 第三步：在直径上找中心 ==========
    int d = dep[y];                // 直径的长度（结点数）
    
    if (d % 2 != 0) {
        // 直径长度为奇数 → 只有一个中心
        // 从 y 向上爬 d/2 步（向下取整），到达中点
        // 例如直径长度为 5（结点数），中心在第 3 个结点
        // 从 y 向上爬 5/2 = 2 步
        for (int i = 1; i <= d / 2; i++) {
            y = fa[y];             // 向上移动到父结点
        }
        cout << y;                 // 输出唯一的中心
    } else {
        // 直径长度为偶数 → 有两个中心
        // 从 y 向上爬 d/2 - 1 步，到达两个中心中较远的那个
        // 另一个是它的父结点
        // 例如直径长度为 4（结点数），中心在第 2 和第 3 个结点
        // 从 y 向上爬 4/2 - 1 = 1 步
        for (int i = 1; i <= d / 2 - 1; i++) {
            y = fa[y];
        }
        // y 和 fa[y] 就是两个中心，按从小到大输出
        if (y < fa[y]) {
            cout << y << " " << fa[y];
        } else {
            cout << fa[y] << " " << y;
        }
    }

    return 0;
}

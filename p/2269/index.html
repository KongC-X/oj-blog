2269 - 最小新整数

题目描述

给定一个十进制正整数n(0 < n < 1000000000)，每个数位上数字均不为0。n的位数为m。 现在从m位中删除k位(0 < k < m)，求生成的新整数最小为多少？
例如: n = 9128456, k = 2, 则生成的新整数最小为12456

输入

第一行t, 表示有t组数据； 接下来t行，每一行表示一组测试数据，每组测试数据包含两个数字n, k。

输出

t行，每行一个数字，表示从n中删除k位后得到的最小整数。

AC代码

#include <iostream>
#include <string>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int t;                        // 测试组数
    cin >> t;
    while (t--) {
        string s;
        int k;                    // 要删除的位数
        cin >> s >> k;

        int len = s.size();
        int a[15] = {0};          // 最多9位，开15足够
        for (int i = 0; i < len; i++) {
            a[i] = s[i] - '0';
        }

        // 删除 k 位
        for (int del = 1; del <= k; del++) {
            bool removed = false;
            for (int i = 0; i < len - 1; i++) {   // 注意只检查到倒数第二位
                if (a[i] > a[i + 1]) {             // 找到下降点
                    // 删除 a[i]，后面的数往前移
                    for (int j = i; j < len - 1; j++) {
                        a[j] = a[j + 1];
                    }
                    len--;
                    removed = true;
                    break;
                }
            }
            // 如果没有下降点，说明序列非递减，删除最后一个数
            if (!removed) {
                len--;
            }
        }

        // 输出结果（题目保证没有前导零）
        for (int i = 0; i < len; i++) {
            cout << a[i];
        }
        cout << '\n';
    }
    return 0;
}

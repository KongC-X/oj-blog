2217 - 找数字的位置

题目描述

给定一个 nnn 个数的数组 aaa （保证 nnn 个数互不相同），再给定 mmm 次询问，每次询问一个整数 xxx 在数组 aaa 中出现的位置，如果存在输出该数在 aaa 数组中出现的位置，不存在输出 000 。

输入

第 111 行是一个整数 nnn（5≤n≤1055≤n≤10^55≤n≤105）。

第 222 行有 nnn 个整数，数与数之间用空格隔开。

第 333 行有一个整数 mmm（5≤m≤1055≤m≤10^55≤m≤105）。

接下来 mmm 行，每行有一个整数。

输出

输出 mmm 行，每行输出每次查询的整数在 aaa 数组中的位置，如果该数不存在，输出 000。

解题思路

AC代码

AC代码

#include<bits/stdc++.h>
using namespace std;
int n,m,x;
struct Num
{
    int num,loc;//num是当前数字,loc是该数字的位置
} a[100010];
bool cmp(Num a,Num b)//sort判断函数
{
    if(a.num<b.num)return true;
    else return false;
}
int find(int x)//二分找数函数
{
    int l=1,r=n,mid;
    while(l<=r)
    {
        mid=l+r>>1;
        if(a[mid].num<x)//通过分治来缩小二分左右边界
        {
            l=mid+1;
        }
        else if(a[mid].num>x)
        {
            r=mid-1;
        }
        else
        {
            return a[mid].loc;//找到了直接返回它的位置
        }
    }
    return 0;//否则返回0
}
int main()
{
    scanf("%d",&n);//scanf运行效率较高,cin可能会超时
    for(int i = 1; i <= n; i++)
    {
        scanf("%d",&a[i].num);
        a[i].loc=i;
    }
    sort(a+1,a+n+1,cmp);//自定义排序
    scanf("%d",&m);
    for(int i = 1; i <= m; i++)
    {
        scanf("%d",&x);
        printf("%d\n",find(x));
    }
    return 0;
}

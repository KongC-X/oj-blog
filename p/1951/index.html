1951 - 子串统计

题目描述

给定一个长度为 nnn 的字符串 SSS，还有一个数字 LLL，统计长度大于等于 LLL 的出现次数最多的子串（不同的出现可以相交），如果有多个，输出最长的，如果仍然有多个，输出第一次出现最早的。

输入

第一行一个数字 LLL。

第二行是字符串 SSS。

LLL 大于 000 ，且不超过 SSS 的长度。

输出

一行，题目要求的字符串。

AC代码

AC代码

#include <bits/stdc++.h>
using namespace std;

/*
 从长度为len开始找每个长度的可能的子串出现的次数
4
bbaabbaaaaa 
从长度为4的子串开始求起，求出所有可能的子串每个子串出现的次数；
然后求长度为5、6……
*/
string s,t;//t表示子字符串
string r;//表示出现次数最多的结果字符串
int ma;//表示出现次数最多的字符串的出现次数 
int len; 

//计算字符串s中t子串出现的次数
int fun(string s,string t){
    int c = 0;
    
    for(int i = 0;i <= s.size() - t.size();i++){
        if(s.substr(i,t.size()) == t){
            c++;
        }
    }
    return c;
} 

int main(){
    cin>>len>>s;
    //先循环所有可能的子串的长度
    for(int i = len;i <= s.size();i++){
        //求在长度为i的情况下，所有可能出现的子串
        //每个子串出现的次数
        for(int j = 0;j <= s.size() - i;j++){
            t = s.substr(j,i);//截出长度为i的子串
            //计算t字符串在s字符串中出现的次数
            //cout<<t<<endl; 
            int c = fun(s,t);
            //如果出现次数比ma多
            //或者一样多，但更长 
            if(c>ma||(c==ma&&t.size()>r.size())){
                r = t;
                ma = c;
            }
        } 
    } 
    
    cout<<r;
}

1852 - 集合运算

题目描述

给出两个整数集合 AAA 、 BBB ，求出他们的交集、并集以及 BBB 在 AAA 中的余集。

输入

第一行为一个整数 nnn ，表示集合 AAA 中的元素个数。

第二行有 nnn 个互不相同的用空格隔开的整数，表示集合 AAA 中的元素。

第三行为一个整数 mmm ，表示集合 BBB 中的元素个数。

第四行有 mmm个互不相同的用空格隔开的整数，表示集合 BBB 中的元素。

集合中的所有元素均为 intintint 范围内的整数，nnn 、m≤1000m \le 1000m≤1000。

输出

第一行按从小到大的顺序输出 AAA 、BBB 交集中的所有元素。

第二行按从小到大的顺序输出 AAA 、BBB 并集中的所有元素。

第三行按从小到大的顺序输出 BBB 在 AAA 中的余集中的所有元素。

解题思路

AC代码

#include <bits/stdc++.h>
using namespace std;
 
/*
c交集：a有b也有
d并集：a和b合并去重复
e余集：a有b没有 
*/

int a[1010],b[1010],c[1010],d[2010],e[1010];
//n代表a有几个数，m代表b有几个数
//k1~k3代表：c d e数组的下标 
int n,m,k1,k2,k3; 

//长度为n的arr数组中是否有元素x 
bool have(int arr[],int n,int x){
    for(int i = 1;i <= n;i++){
        if(arr[i] == x) return true; 
    } 
    
    return false;
}

//输出
void print(int arr[],int len){
    for(int i = 1;i <= len;i++){
        cout<<arr[i]<<" ";
    }
    cout<<endl;
} 

int main()
{
    cin>>n;
    for(int i = 1;i <= n;i++){
        cin>>a[i];
        //a数组每个数都是并集的一部分
        k2++;
        d[k2] = a[i]; 
    } 
    
    cin>>m;
    for(int i = 1;i <= m;i++){
        cin>>b[i];
        //交集：b[i]如果在a中出现了，存入c的数组 
        if(have(a,n,b[i])){
            k1++;
            c[k1] = b[i];
        }else{
            //b[i]在a中没有出现，存入并集
            k2++;
            d[k2] = b[i]; 
        } 
    }
    
    //输出交集
    sort(c+1,c+k1+1);
    print(c,k1); 
    
    //并集
    sort(d+1,d+k2+1);
    print(d,k2); 
    
    //求余集：a有b没有
    for(int i = 1;i <= n;i++){
        if(have(b,m,a[i])==false){
            k3++;
            e[k3] = a[i];
        }
    } 
    
    //余集
    sort(e+1,e+k3+1);
    print(e,k3); 
    return 0;
}

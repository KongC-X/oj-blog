2265 - 酒鬼

题目描述

Santo刚刚与房东打赌赢得了一间在New Clondike 的大客厅。今天，他来到这个大客厅欣赏他的奖品。房东摆出了一行瓶子在酒吧上。瓶子里都装有不同体积的酒。令Santo高兴的是，瓶子中的酒都有不同的味道。

房东说道：“你可以喝尽可能多的酒，但是一旦打开酒盖你就必须把它喝完，喝完一瓶后把它放回原处。还有一件最重要的事，你必须从左至右依次喝，并且不能连续超过两瓶，不然会给你带来坏运气。”

现在可怜的Santo站在酒吧前努力的想着，他到底应该喝哪几瓶才能使喝的酒最多呢？请帮助他找出他应该喝的酒瓶号，因为思考让他感到不安。

输入

第一行一个整数 NNN，有 NNN 个酒瓶。（N≤700N \le 700N≤700）

接下有 NNN 行，第 i+1i+1i+1 行的数字代表酒瓶 iii 中酒的体积。

输出

一个数字，喝的酒的最大总体积。遵守以上规则，使得三个连续瓶子中至少一个瓶子是满的。（本题测试数据保证计算结果 ≤109≤10^9≤109）

AC代码

#include <iostream>
#include <algorithm>
using namespace std;

const int MAXN = 705;
int a[MAXN];
int dp[MAXN][3][2];   // dp[i][j][k]: 前i瓶，以连续j个状态k结尾的最大总体积
                      // j=1或2，k=0(不喝)或1(喝)

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    // 初始化所有状态为极小值
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= 2; j++) {
            dp[i][j][0] = dp[i][j][1] = -1e9;
        }
    }

    // 第1瓶
    dp[1][1][0] = 0;           // 不喝第1瓶
    dp[1][1][1] = a[1];        // 喝第1瓶

    for (int i = 1; i < n; i++) {
        for (int j = 1; j <= 2; j++) {
            for (int k = 0; k <= 1; k++) {
                if (dp[i][j][k] < 0) continue;
                int cur = dp[i][j][k];

                // 下一瓶不喝：c = 0
                if (k == 0) {
                    if (j + 1 <= 2) {  // 连续不喝长度+1，不能超过2
                        dp[i+1][j+1][0] = max(dp[i+1][j+1][0], cur);
                    }
                } else {
                    // 上一瓶喝了，这瓶不喝，重新从1开始
                    dp[i+1][1][0] = max(dp[i+1][1][0], cur);
                }

                // 下一瓶喝：c = 1
                if (k == 1) {
                    if (j + 1 <= 2) {
                        dp[i+1][j+1][1] = max(dp[i+1][j+1][1], cur + a[i+1]);
                    }
                } else {
                    dp[i+1][1][1] = max(dp[i+1][1][1], cur + a[i+1]);
                }
            }
        }
    }

    int ans = 0;
    for (int j = 1; j <= 2; j++) {
        ans = max(ans, max(dp[n][j][0], dp[n][j][1]));
    }

    cout << ans << '\n';
    return 0;
}

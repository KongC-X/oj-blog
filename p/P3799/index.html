P3799 小 Y 拼木棒

洛谷 | 普及| C++14 | 提交时间: 2026-02-10

算法标签：洛谷原创、枚举、组合数学、分类讨论、洛谷比赛

题目背景

上道题中，小 Y 斩了一地的木棒，现在她想要将木棒拼起来。

题目描述

有 nnn 根木棒，现在从中选 444 根，想要组成一个正三角形，问有几种选法？

答案对 109+710^9+7109+7 取模。

输入格式

第一行一个整数 nnn。

第二行往下 nnn 行，每行 111 个整数，第 iii 个整数 aia_iai​ 代表第 iii 根木棒的长度。

输出格式

一行一个整数代表答案。

样例

样例输入 #1

4 
1
1
2
2```

**样例输出 #1**

1```

AC 代码

#include <iostream>
using namespace std;

const int MOD = 1e9+7;
int cnt[5050] = {0};

// 计算组合数C(n,2) = n*(n-1)/2
long long C2(long long x) {
    if (x < 2) return 0;
    return x * (x-1) / 2 % MOD;   // 直接除2取模，因为x*(x-1)一定是偶数
}

int main() {
    int n;
    cin >> n;
    int l = 0;
    
    // 统计每种长度的木棒数量
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        cnt[x]++;
        if (x > l) l = x;
    }
    
    long long ans = 0;
    
    // 枚举正三角形的边长a
    for (int a = 2; a <= l; a++) {
        // 需要至少两根长度为a的木棒
        if (cnt[a] < 2) continue;
        
        // 从长度为a的木棒中选2根的方案数
        long long base = C2(cnt[a]);
        
        // 计算另外两根木棒的长度组合(b,c)，满足b+c=a
        long long sum = 0;
        for (int b = 1; b <= a/2; b++) {
            int c = a - b;
            
            if (b == c) {
                // b=c时，需要从长度为b的木棒中选2根
                sum = (sum + C2(cnt[b])) % MOD;
            } else {
                // b≠c时，需要从长度为b和c的木棒中各选1根
                sum = (sum + (long long)cnt[b] * cnt[c] % MOD) % MOD;
            }
        }
        
        // 累加当前a的方案数
        ans = (ans + base * sum % MOD) % MOD;
    }
    
    cout << ans << endl;
    return 0;
}

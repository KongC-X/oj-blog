1280 - 求2的n次方

题目描述

求 222 的 nnn 次方。（ 0≤n≤1000 \le n \le 1000≤n≤100）

输入

从键盘读入一个整数 nnn；

输出

请输出 222 的 nnn 次方；

解题思路

AC代码

AC代码

#include <iostream> // 头文件
using namespace std; // 使用标准的命名空间

int a[1000] = {1};
int n,i,j,k = 1;
// k代表a数组的元素个数，代表2的n次方高精度的整数位数
int main() {
    cin >> n;
    // 第一层循环是这个数要乘多少次2
    for(i = 1;i <= n;i++) {
        // 第二层循环每次有几位要乘2
        for(j = 0; j < k;j++) {
            a[j] = a[j] * 2;
        }
        // 处理进位
        for(j = 0;j < k;j++) {
            if(a[j] >= 10) {
                a[j + 1] += a[j] / 10;
                a[j] = a[j] % 10;
            }
        }
        // 判断a数组是否多出一位
        if(a[k] != 0) {
            k++;
        }
    }
    // 输出结果
    for(i = k - 1;i >= 0;i--) {
        cout << a[i];
    }
}

P2884 [USACO07MAR] Monthly Expense S

洛谷 | 普及| C++14 | 提交时间: 2026-07-24

算法标签：2007、USACO

题目描述

给定一个 $n$ 项的正整数序列 ${a_n}$，请找出满足以下条件的最小的正整数 $S$：存在一种将 ${a_n}$ 划分成 $m$ 段的方案，使得每一段中所有数字的和都不大于 $S$。

输入格式

第一行两个正整数 $n,m$。

下面 $n$ 行，每行一个正整数 $a_i$。

输出格式

一行一个正整数表示答案。

样例

样例输入 #1

7 5
100
400
300
100
500
101
400```

**样例输出 #1**

500```

AC 代码

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100010;          // n 最大 1e5，多开10
int a[MAXN];                      // 存储每天的花费
int n, m;                         // n:天数, m:段数

// 判断当每段上限为 S 时，能否在 m 段内完成划分
bool check(int S) {
    int cnt = 1;                  // 当前段数，至少1段
    long long sum = 0;            // 当前段的和
    for (int i = 0; i < n; i++) {
        if (sum + a[i] > S) {     // 当前段装不下，另起一段
            cnt++;
            sum = a[i];
            if (cnt > m) return false;  // 段数超标，不可行
        } else {
            sum += a[i];
        }
    }
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    
    /*
      思路：二分答案 + 贪心检验
      要求最大段和的最小值，这是典型的二分答案模型。
      对答案 S 进行二分：
        - 下界 L = max(a[i])，因为至少要把最大的元素单独放一段
        - 上界 R = sum(a[i])，全部放一段的极端情况
      每次取 mid = (L+R)/2，用贪心检验能否在 m 段内使每段和 <= mid
      贪心策略：从头累加，超过 mid 就新开一段。
      若段数 <= m，说明 mid 可行，尝试更小的（R = mid - 1），并记录答案；
      否则 L = mid + 1。
      最后输出答案 ans。
      时间复杂度 O(n log sum(a))，n=1e5 可行。
    */
    
    cin >> n >> m;
    int L = 0;                    // 二分下界：最大单个元素
    int R = 0;                    // 二分上界：总和
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] > L) L = a[i];
        R += a[i];
    }
    
    int ans = R;                  // 答案初始化为上界
    while (L <= R) {
        int mid = L + (R - L) / 2;
        if (check(mid)) {
            ans = mid;            // 可行，记录答案，尝试更小
            R = mid - 1;
        } else {
            L = mid + 1;          // 不可行，需要更大
        }
    }
    
    cout << ans << endl;
    return 0;
}

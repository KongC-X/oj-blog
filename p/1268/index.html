1268 - 高精度加法

题目描述

计算 $a+b$ 的值，$a,b$ 皆为不超过 $240$ 位的非负整数。

输入

两个正整数，每行一个。

输出

一个数，代表两个整数的和。

AC代码

AC代码

#include <iostream>
#include <bits/stdc++.h>
using namespace std;
/*
第⼀步：⽤string读⼊⾼精度整数
第⼆步：将两个⾼精度整数逆序存⼊ab两个整数数组
第三步：从左向右，逐位求和，结果存⼊c数组从左向右，逐位进位
第四步：逆序输出结果
*/
string s1,s2; //⾼精度整数
int a[250],b[250],c[250]; //存储⾼精度整数的数组
int main() {
    cin >> s1 >> s2; //读⼊⾼精度整数
    // 逆序存放到数组中
    for(int i = s1.size() - 1;i >= 0;i--) {
        a[i] = s1[s1.size() - 1 - i] - '0';
    }
    for(int i = s2.size() - 1;i >= 0;i--) {
        b[i] = s2[s2.size() - 1 - i] - '0';
    }
    //验证
    // for(int i = 0;i < s1.size();i++) {
    // 	cout << a[i];
    // }
    // 找到更长的一个数
    int len = s1.size() > s2.size() ? s1.size() : s2.size();
    // int len = max(s1.size(),s2.size());
    // 诸位相加
    for(int i = 0;i < len;i++) {
        c[i] = a[i] + b[i];
    }
    // 判断是否需要进位
    for(int i = 0;i < len;i++) {
        if(c[i] >= 10) {
            c[i + 1]++; //进位
            c[i] = c[i] - 10; //当前位
        }
    }
    // 两个不超过len位的整数做加法，结果可能是len+1位
    if(c[len] != 0) {
        len++; //如果最后一位不为0，长度加1
    }
    // 逆序输出结果
    for(int i = len - 1;i >= 0;i--) {
        cout << c[i];
    }
    return 0;
}

1739 - 迷宫的所有路径

题目描述

已知一 $N \times N$ 的迷宫，允许往上、下、左、右四个方向行走，且迷宫中没有任何障碍，所有的点都可以走。

现请你按照右、下、左、上顺序进行搜索，找出从左上角到右下角的所有路径。

输入

输入一个整数 $N$（$N≤5$）代表迷宫的大小。

输出

按右、下、左、上搜索顺序探索迷宫，输出从左上角 $(1,1)$ 点走到右下角 $(N,N)$ 点的所有可能的路径。

AC代码

#include<iostream> 
using namespace std;

int n,c;
int r[30][3];
bool f[10][10];

// 右下左上
int fx[5] = {0,0,1,0,-1};
int fy[5] = {0,1,0,-1,0};

void print(int k) {
    c++;
    cout << c << ":";
    for(int i = 1;i < k;i++) {
        cout << r[i][1] << "," << r[i][2] << "->";
    }
    cout << r[k][1] << "," << r[k][2] << endl;
}

void dfs(int x,int y ,int k) {
    r[k][1] = x;
    r[k][2] = y;
    if(x == n & y == n) {
        print(k);
        return;
    } else {
        int tx,ty;
        for(int i = 1;i <= 4;i++) {
            tx = x + fx[i];
            ty = y + fy[i];
            if(tx >= 1 && tx <= n && ty >=1 && ty <= n && f[tx][ty] == false) {
                f[tx][ty] = true; // 标记为已走
                dfs(tx,ty,k+1); // dfs
                f[tx][ty] = false; // 回溯,撤销标记
                
            }
        }
    }
}

int main() {
    cin >> n;
    f[1][1] = true;
    dfs(1,1,1);
}

B4263 [GESP202503 四级] 荒地开垦

洛谷 | 普及- | C++14 | 提交时间: 2025-03-28

算法标签：模拟、2025、枚举、GESP

题目背景

对应的选择、判断题：https://ti.luogu.com.cn/problemset/1176

题目描述

小杨有一大片荒地，可以表示为一个 $n$ 行 $m$ 列的网格图。

小杨想要开垦这块荒地，但荒地中一些位置存在杂物，对于一块不存在杂物的荒地，该荒地可以开垦当且仅当其上下左右四个方向相邻的格子均不存在杂物。

小杨可以选择至多一个位置，清除该位置的杂物，移除杂物后该位置变为荒地。小杨想知道在清除至多一个位置的杂物的情况下，最多能够开垦多少块荒地。

输入格式

第一行包含两个正整数 $n, m$，含义如题面所示。

之后 $n$ 行，每行包含一个长度为 $m$ 且仅包含字符 . 和 # 的字符串。如果为 .，代表该位置为荒地；如果为 #，代表该位置为杂物。

输出格式

输出一个整数，代表在清除至多一个位置的杂物的情况下，最多能够开垦的荒地块数。

样例

样例输入 #1

3 5
.....
.#..#
.....```

**样例输出 #1**

11```

AC 代码

#include <iostream>
#include <string>
using namespace std;

// 我们有一个n行m列的网格，每个格子要么是荒地（.），要么是杂物（#）。
// 荒地可以开垦的条件是它的上下左右四个相邻格子都没有杂物。
// 我们可以选择清除最多一个杂物（将其变为荒地），然后计算在这种情况下最多能开垦多少块荒地。
// 解决思路：
// 初始状态：首先计算在不移除任何杂物的情况下，有多少荒地是可以开垦的。
// 移除杂物的影响：对于每一个杂物，假设我们移除它，看看这样能新增多少可开垦的荒地。
// 最优解：在所有可能的杂物移除中，找到能带来最大新增可开垦荒地的情况，结合初始状态得到最终答案。

int main() {
    int n, m;
    cin >> n >> m;
    string grid[1010];

    for (int i = 0; i < n; ++i) {
        cin >> grid[i];
    }

    // 初始可开垦的荒地数
    int initial = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (grid[i][j] == '.') {
                bool can_cultivate = true;
                // 检查四个方向
                if (i > 0 && grid[i-1][j] == '#') can_cultivate = false;
                if (i < n-1 && grid[i+1][j] == '#') can_cultivate = false;
                if (j > 0 && grid[i][j-1] == '#') can_cultivate = false;
                if (j < m-1 && grid[i][j+1] == '#') can_cultivate = false;
                if (can_cultivate) {
                    initial++;
                }
            }
        }
    }

    int max_additional = 0;
    // 尝试清除每一个杂物
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (grid[i][j] == '#') {
                int additional = 0;
                // 检查清除该杂物后，四个方向的荒地是否变为可开垦
                // 上
                if (i > 0 && grid[i-1][j] == '.') {
                    bool can_now = true;
                    if (i-1 > 0 && grid[i-2][j] == '#') can_now = false;
                    if (j > 0 && grid[i-1][j-1] == '#') can_now = false;
                    if (j < m-1 && grid[i-1][j+1] == '#') can_now = false;
                    if (can_now) additional++;
                }
                // 下
                if (i < n-1 && grid[i+1][j] == '.') {
                    bool can_now = true;
                    if (i+1 < n-1 && grid[i+2][j] == '#') can_now = false;
                    if (j > 0 && grid[i+1][j-1] == '#') can_now = false;
                    if (j < m-1 && grid[i+1][j+1] == '#') can_now = false;
                    if (can_now) additional++;
                }
                // 左
                if (j > 0 && grid[i][j-1] == '.') {
                    bool can_now = true;
                    if (j-1 > 0 && grid[i][j-2] == '#') can_now = false;
                    if (i > 0 && grid[i-1][j-1] == '#') can_now = false;
                    if (i < n-1 && grid[i+1][j-1] == '#') can_now = false;
                    if (can_now) additional++;
                }
                // 右
                if (j < m-1 && grid[i][j+1] == '.') {
                    bool can_now = true;
                    if (j+1 < m-1 && grid[i][j+2] == '#') can_now = false;
                    if (i > 0 && grid[i-1][j+1] == '#') can_now = false;
                    if (i < n-1 && grid[i+1][j+1] == '#') can_now = false;
                    if (can_now) additional++;
                }
                // 当前格子本身是否变为可开垦
                bool can_current = true;
                if (i > 0 && grid[i-1][j] == '#') can_current = false;
                if (i < n-1 && grid[i+1][j] == '#') can_current = false;
                if (j > 0 && grid[i][j-1] == '#') can_current = false;
                if (j < m-1 && grid[i][j+1] == '#') can_current = false;
                if (can_current) additional++;

                if (additional > max_additional) {
                    max_additional = additional;
                }
            }
        }
    }

    cout << initial + max_additional << endl;
    return 0;
}

1205 - 有趣的数字图形II

题目描述

输入一个整数 nnn （ n≤12n \le 12n≤12），打印出如下要求的方阵：左上到右下对角线上的数与行数相同，右上半个区域中每个元素等于左边的和下面的元素之和。每个元素场宽为 555 。左下半个区域为空。

输入

一个整数 nnn （ n≤12n \le 12n≤12 ）

输出

n×nn \times nn×n 的方阵（场宽为 555 ）。

AC代码

AC代码

#include <iostream>
//#include <bits/stdc++.h>
#include <cstdlib>
#include <ctime>
#include <cstring>
using namespace std;
int data[120][120];
int main(){
    int n,temp;
    cin>>n;
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            if(i==j){
                data[i][j] = i + 1;
            }
        }
    }

    for(int i=0;i<n-1;i++){
        for(int j=0;j<n-1-i;j++){
            data[j][j+1+i] = data[j][j+i] + data[j+1][j+1+i];
        }
    }

    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            if(data[i][j] > 0){
                printf("%5d",data[i][j]);
            }else{
                printf("%5s","");
            }
        }
        printf("\n");
    }
}

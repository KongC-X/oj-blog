4678 - 时间跨越

题目描述

假设现在是 $y$ 年 $m$ 月 $d$ 日 $h$ 时而 $k$ 小时后是 $y'$ 年 $m'$ 月 $d'$ 日 $h'$ 时，对于给定的 $y$，$m$，$d$，$h$，$k$，小杨想请你帮他计算出对应的 $y'$，$m'$，$d'$，$h'$ 是多少。

输入

输入包含五行，每行一个正整数，分别代表 $y$，$m$，$d$，$h$，$k$。

输出

输出四个正整数，代表 $y'$，$m'$，$d'$，$h'$ 。

AC代码

AC代码

#include <iostream>
using namespace std;

int main() {
    int y,m,d,h,k;
    cin >> y >> m >> d >> h >> k;
    h += k;
    if(h >= 24) {
        h -= 24;
        d++;
        int days = 0;
        if(m == 1 || m == 3 || m == 5 || m == 7 || m == 8 || m == 10 || m == 12) {
            days = 31;
        } else if(m == 4 || m == 6 || m == 9 || m == 11) {
            days = 30;
        } else if(m == 2) {
            if(y % 4 == 0 && (y % 100 != 0 || y % 400 == 0)) {
                days = 29;
            } else {
                days = 28;
            }
        }
        if(d > days) {
            d -= days;
            m++;
            if(m > 12) {
                m = 1;
                y++;
            }
        }
    }
    cout << y << " " << m << " " << d << " " << h << endl;
}

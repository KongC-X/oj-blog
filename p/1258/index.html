1258 - 求一个三位数

题目描述

求这样一个三位数，该三位数等于其每位数字的阶乘之和， 即 abc=a!+b!+c!abc = a! + b! + c!abc=a!+b!+c!。

(n!n!n! 表示 nnn 的阶乘, n!=1×2×3×⋯×nn!=1 \times 2 \times 3 \times \dots \times nn!=1×2×3×⋯×n，如：5!=1×2×3×4×55!=1 \times 2 \times 3 \times 4 \times 55!=1×2×3×4×5)

输入

无

输出

输出这个数

解题思路

AC代码

AC代码

#include <iostream>
using namespace std;

int factorial(int a) {
    int result = 1;
    for (int i = 1; i <= a; i++) {
        result *= i;
    }
    return result;
}

int main() {
    int n,a,b,c,i;
    for(i = 100;i <= 999;i++) {
        a = i / 100;
        b = (i / 10) % 10;
        c = i % 10;
        if(i == factorial(a) + factorial(b) + factorial(c)) {
            cout << i << endl;
        }
    }


    
}

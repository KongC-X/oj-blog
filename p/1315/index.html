1315 - 遥控飞机争夺赛

题目描述

红太阳杯遥控飞机大赛拉开帷幕。比赛规则为，每位选手让自己的飞机从起点到终点飞行 555 次，组委会记录 555 次的飞行的成绩之后去掉一个最大成绩、一个最小成绩后计算剩余 333 个成绩的平值（平均分保留 333 位小数）作为该选手的最终成绩。

有 nnn 名选手参加了比赛，从键盘读入每位选手的编号以及他们的 555 次飞行的成绩。

请根据 nnn 名选手的比赛成绩，编程计算出冠军、亚军、季军的编号以及组委会计算出的成绩。（假设不存在多名选手成绩正好一样）（ 4.1.514.1.514.1.51 ）

输入

第一行为一个整数 nnn ，代表参加比赛的选手数量（ 4≤n≤1004 \le n \le 1004≤n≤100）

后面的 nnn 行，每行有 666 个数，第一个数是选手的编号，后 555 个数为选手的 555 次飞行的成绩。

输出

333 行：

第一行输出冠军的编号及飞行成绩（保留 333 位小数）用空格隔开 222 个数；

第二行输出亚军的编号及飞行成绩；

第三行输出季军的编号及飞行成绩。

解题思路

#include<bits/stdc++.h>

using namespace std;

struct Xuanshou{

    int id;

    int fenshu;

    double zongfen;

};

bool cmp(Xuanshou a,Xuanshou b){

    if(a.zongfen>b.zongfen){

        return true;

    }

    else if(a.zongfen<b.zongfen){

        return false;

    }

}

int main(){

    int n;

    cin>>n;

    Xuanshou xs[10010];

    int zf=0;

    double pj=0;

    int ma=0,mi=100;

    for(int i=0;i<n;i++){

        int ma=0,mi=100;

        cin>>xs[i].id;

        zf=0;

        for(int j=1;j<6;j++){

            cin>>xs[i].fenshu;

            zf=zf+xs[i].fenshu;

            ma=max(ma,xs[i].fenshu);

            mi=min(mi,xs[i].fenshu);

        }

        zf=zf-ma-mi;

        pj=zf/3.0;

        xs[i].zongfen=pj;

    }

    sort(xs,xs+n,cmp);

    for(int i=0;i<3;i++){

        cout<<xs[i].id<<" ";

        printf("%.3f",xs[i].zongfen);

        cout<<endl;

    }

}

1362 - 马的遍历

题目描述

中国象棋半张棋盘如图（a）所示。马自左下角往右上角跳。

今规定只许往右跳，不许往左跳，且要求马跳的方式按照（b）图顺时针深度优先递归。比如图（a）中所示为一种跳行路线。如果马要从 0,00,00,0 点，跳到 4,84,84,8 点，前 666 种跳法的打印格式如下，请参考前 666 种跳的方式，输出马从 0,00,00,0 点到 4,84,84,8 点所有可能的跳的路线。 

`1:0,0->2,1->4,2->3,4->4,6->2,7->4,8
2:0,0->2,1->4,2->3,4->1,5->3,6->4,8
3:0,0->2,1->4,2->3,4->1,5->2,7->4,8
4:0,0->2,1->4,2->2,3->4,4->3,6->4,8
5:0,0->2,1->4,2->2,3->4,4->2,5->4,6->2,7->4,8
6:0,0->2,1->4,2->2,3->4,4->2,5->0,6->2,7->4,8
`

输入

无

输出

按要求输出路径。

解题思路

AC代码

AC代码

#include <iostream>
#include <iomanip>
#include <bits/stdc++.h>
using namespace std;
int n,m,c; 
int r[20][3]; // 记录路径

int fx[5] = {0,2,1,-1,-2};
int fy[5] = {0,1,2,2,1};
// 打印函数，k是路径数组的下标
void print(int k) {
    c++;
    cout << c << ":";
    for(int i = 1;i < k;i++) {
        cout << r[i][1] << "," << r[i][2] << "->";
    }
    // 最后到n，m点，单独打印输出
    cout << 4 << "," << 8 << endl;
}
// 深度优先搜索，k是路径数组的下标
void dfs(int x,int y,int k) {
    // 记录探索到的点的坐标
    r[k][1] = x;
    r[k][2] = y;
    // 如果xy是终点了，就打印路径
    if(x == 4 && y == 8) {
        print(k);
        // 停止递归函数，到了终点打印，就不需要递归了
        return;
    }
    int tx,ty;
    for(int i = 1;i <= 4;i++) {
        tx = x + fx[i];
        ty = y + fy[i];
        if(tx >= 0 && tx <= 4 && ty >= 0 && ty <= 8) {
            dfs(tx,ty,k+1);
        }
    }
}
int main(){
    dfs(0,0,1);
}

1206 - 有趣的数字图形III

题目描述

读入一个整数 nnn （n<100n \lt 100n<100 ）,输出一个 n×nn \times nn×n 的方阵。 对角线是 111 ，对角线向右下逐渐递增，左上为空格，每个数场宽为 555 。

输入

一个整数 nnn 。

输出

n×nn \times nn×n 的方阵

AC代码

AC代码

#include <bits/stdc++.h>
using namespace std;

int main(){
    int a[110][110]={0};
    int n;
    
    cin >> n;

    // 赋值次数 
    for (int i=1; i<=n; i++){
        // j代表第i次，赋值元素的个数
        for(int j=1; j<=n-i+1; j++){
            a[j+i-1][n-j+1] = i;
        }
    }

    // 输出数组 
    for(int i=1; i<=n; i++){
        for(int j=1; j<=n; j++){
            if(a[i][j]!= 0){
                cout << setw(5) << a[i][j];
        }else{
            cout << setw(5) << ' ';
        }
        }
        cout << endl;
    }
}

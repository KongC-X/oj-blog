AC代码

#include<bits/stdc++.h>
using namespace std;

/*
  01背包
  f[i,j] = max(f[i-1,j],f[i-1,j-v[i]]+w[i])
  分组背包：三重循环
  f[i,j] = max(f[i-1,j],f[i-1,j-v[i,k]]+w[i,k])
*/

int f[110],v[110][110],w[110][110],s[110];
int main() {
    int n,m;
    cin>>n>>m;
    for(int i = 1; i <= n; i++) {
        cin>>s[i];
        for(int j = 0; j < s[i]; j++) {
            cin>>v[i][j]>>w[i][j];
        }
    }

    for(int i = 1; i <= n; i++) {
        for(int j = m; j >= 0; j--) {
            for(int k = 0; k < s[i]; k++) {
                if(v[i][k] <= j) {
                    f[j] = max(f[j],f[j-v[i][k]]+w[i][k]);
                }
            }
        }
    }

    cout<<f[m];
    return 0;
}

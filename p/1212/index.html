1212 - 移动数组元素

题目描述

在一个不重复的数组中，请将这个数组的最小数和数组第一个数交换，最大数和数组最后一个数交换。

输入

第一行有一个整数 $n$ ( $5 \le n \le 100$ ）；

第二行有 $n$ 个不重复的整数。

输出

移动位置后的数组。

解题思路

AC代码

AC代码

#include <bits/stdc++.h>
using namespace std;

int a[210],n,ma,mi,t;

int main(){
    cin>>n;
    for(int i = 0;i < n;i++){
        cin>>a[i];
    }
    
    //找最小数下标和最大数下标 
    mi = 0;
    ma = 0;
    for(int i = 1;i < n;i++){
        if(a[i] < a[mi]){
            mi = i;
        }
        
        if(a[i] > a[ma]){
            ma = i;
        }
    } 
    
    //对调下标为min的数和下标为0的数 
    t = a[mi];
    a[mi] = a[0];
    a[0] = t;
    
    //对调下标为max的数和下标为n-1的数
    t = a[ma];
    a[ma] = a[n-1];
    a[n-1] = t; 
    
    //输出
    for(int i = 0;i < n;i++){
        cout<<a[i]<<" ";	
    } 
}

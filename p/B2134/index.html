B2134 质数的和与积

洛谷 | 普及- | C++14 | 提交时间: 2026-07-14

算法标签：函数与递归

题目描述

两个质数的和是 SSS，它们的积最大是多少？

输入格式

一个不大于 100001000010000 的正整数 SSS，为两个质数的和。

输出格式

一个整数，为两个质数的最大乘积。数据保证有解。

样例

样例输入 #1

50```

**样例输出 #1**

589```

AC 代码

#include <bits/stdc++.h>
using namespace std;

// 判断是否为素数
bool isPrime(int x) {
    if (x < 2) return false;
    for (int i = 2; i * i <= x; i++) {
        if (x % i == 0) return false;
    }
    return true;
}

int main() {
    int S;                         // S: 两个质数的和
    cin >> S;

    int maxP = 0;                  // maxP: 最大乘积
    for (int i = 2; i <= S / 2; i++) {
        int j = S - i;             // 另一个数
        if (isPrime(i) && isPrime(j)) {
            int prod = i * j;      // 乘积
            if (prod > maxP) maxP = prod;
        }
    }

    cout << maxP << endl;
    return 0;
}

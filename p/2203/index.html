2203 - 二叉树遍历

题目描述

小明学完了树的基本知识，想做二叉树遍历的程序设计练习。

根据二叉树的深度从数据文件中读入 nnn 个整数生成顺序存储的二叉树，做各种遍历的输出。小明发现输出的数据中有不少重复的，为此他先将重复的数据过滤掉，然后再生成二叉树的结构。

现在要求你从数据文件中读取若干个整数，生成一个满二叉树，然后输出一个二叉树遍历的序列。

输入

输入包含 222 行；

第一行是两个整数 hhh 和 ppp ，hhh 是二叉树的深度， h≤8h≤8h≤8 。p=1p=1p=1时先序输出，p=2p=2p=2 时中序输出，p=3p=3p=3 时后序输出。

第二行若干个整数（数字小于等于 100010001000 ），读入数据前，先要计算二叉树的结点个数 nnn ，然后在数据中顺序读取 nnn 个不重复的整数。

例如深度为 333 的二叉树结点有 777 个。

输出

输出文只有一行，是二叉树遍历的序列。

AC代码

#include<iostream>
#include<bits/stdc++.h>
using namespace std;
int h, p, c, n;
//cnt：标记数组，标记数字是否出现
//8层的二叉树，最多有2^8-1个结点
int x, cnt[1010], a[300];
void dfs1(int x) {
    cout << a[x] << " ";
    if(x * 2 <= n && a[x * 2]) dfs1(x * 2);
    if(x * 2 + 1 <= n && a[x * 2 + 1]) dfs1(x * 2 + 1);
}
void dfs2(int x) {
    if(x * 2 <= n && a[x * 2]) dfs2(x * 2);
    cout << a[x] << " ";
    if(x * 2 + 1 <= n && a[x * 2 + 1]) dfs2(x * 2 + 1);
}
void dfs3(int x) {
    if(x * 2 <= n && a[x * 2]) dfs3(x * 2);
    if(x * 2 + 1 <= n && a[x * 2 + 1]) dfs3(x * 2 + 1);
    cout << a[x] << " ";
}
int main() {
    cin >> h >> p;
    // 满二叉树深度为h，最多有2^h-1个结点
    n = (1 << h) - 1; //最多的节点数量

    //还没读到n个结点
    while(c < n) {
        scanf("%d", &x);
        if(cnt[x]) continue;

        cnt[x] = 1;
        a[++c] = x;
    }

    if(p == 1) dfs1(1);
    if(p == 2) dfs2(1);
    if(p == 3) dfs3(1);
    return 0;
}

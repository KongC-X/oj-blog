1230 - 蝴蝶结

题目描述

请输出 nnn 行的蝴蝶结的形状，nnn 一定是一个奇数！

输入

一个整数 nnn ，代表图形的行数！

输出

nnn 行的图形。

AC代码

AC代码

#include<bits/stdc++.h>
#include<iostream>
using namespace std;

int n;
int main() {
cin>>n;
for(int i=1;i<=n/2+1;i++)
{
    
    for(int j=1;j<=n/2+2-i;j++)
    cout<<"*";
    cout<<endl;	
}
for(int i=n/2;i>=1;i--)
{
    
    for(int j=1;j<=n/2+2-i;j++)
    cout<<"*";
    cout<<endl;
    
}

return 0;
}

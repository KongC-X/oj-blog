B2137 判决素数个数

洛谷 | 入门 | C++14 | 提交时间: 2026-07-14

算法标签：素数判断、函数与递归

题目描述

求 $X$，$Y$ 之间的素数个数（包括 $X$ 和 $Y$）。

输入格式

两个整数 $X$ 和 $Y$（$1 \le X,Y \le 1.1 \times 10^6$）。

输出格式

输出一个整数，表示 $X, Y$ 之间的素数个数（包括 $X$ 和 $Y$）。

样例

样例输入 #1

1 100```

**样例输出 #1**

25```

AC 代码

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1100005;          // 1.1e6 多一点
bool isP[MAXN];                    // 标记数组，true 表示是素数

// 函数：埃拉托色尼筛法，标记 0..n 内的素数
void sieve(int n) {
    for (int i = 2; i <= n; i++) isP[i] = true;  // 先假设全是素数
    isP[0] = isP[1] = false;                     // 0和1不是素数
    for (int i = 2; i * i <= n; i++) {
        if (isP[i]) {
            for (int j = i * i; j <= n; j += i) {
                isP[j] = false;                  // i 的倍数不是素数
            }
        }
    }
}

int main() {
    int X, Y;
    cin >> X >> Y;
    if (X > Y) swap(X, Y);         // 保证 X 是左端点，Y 是右端点

    sieve(Y);                      // 筛出 0..Y 的素数

    int cnt = 0;
    for (int i = X; i <= Y; i++) {
        if (isP[i]) cnt++;
    }

    cout << cnt << endl;
    return 0;
}

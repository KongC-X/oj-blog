1685 - n个数的全排列

题目描述

从键盘读入 nnn 个整数（每个数都是 1∼91 \sim 91∼9 之间的数），输出这 nnn 个整数的全排列（数字不能重复）。

输入

第 111 行输入一个整数 nnn。（1≤n≤81 \le n \le 81≤n≤8）

第 222 行输入 nnn 个不相等的整数。（每个数在 [1,9][1,9][1,9] 的范围内）

输出

输出若干行，每行包括 nnn 个数据，表示一种排列方案，所有的排列按字典码从小到大排序输出。

AC代码

AC代码

#include <iostream>
#include <algorithm>
using namespace std;

const int MAXN = 10;
int n;
int nums[MAXN];
int current[MAXN];
bool visited[MAXN];

void dfs(int depth) {
    if (depth == n) {
        for (int i = 0; i < n; i++) {
            cout << current[i] << (i == n - 1 ? "\n" : " ");
        }
        return;
    }
    
    for (int i = 0; i < n; i++) {
        if (!visited[i]) {
            visited[i] = true;
            current[depth] = nums[i];
            dfs(depth + 1);
            visited[i] = false;
        }
    }
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    
    // 排序，确保字典序
    sort(nums, nums + n);
    
    dfs(0);
    
    return 0;
}

2293 - 比n小的最大质数

题目描述

对于给定的 nnn ，求比 nnn 小的质数中最大的一个。

质数是指一个大于 111 的自然数，除了 111 和它自身外，不能被其他自然数整除的数。

输入

一个整数 nnn 。（2≤n≤100002 \le n \le 100002≤n≤10000）

输出

一个整数，即题目要求的解。

AC代码

AC代码

#include <bits/stdc++.h>
using namespace std;

bool zhishu(int x){
    bool r = true;
    for(int i=2; i<=sqrt(x); i++){
        if(x%i==0){
            r = false;
        }
    }
    if(x<=1){
        r = false;
    }
    return r;
}

    int n,i,j;
    cin>>n;
    bool f = true;
    for(i=n-1; i>1; i--){
        if(zhishu(i)){
            cout<<i;
            break;
        }
    } 
    return 0;
}

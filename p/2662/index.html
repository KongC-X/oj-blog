2662 - 指定数出现的次数

题目描述

给定 $N$ 个整数，再给定 $M$ 次询问，每次询问给出一个整数 $X$ ，请求出针对每次询问，整数 $X$ 在 $N$ 个整数中出现的次数。

输入

第 $1$ 行读入一个整数 $N$ 和 $M$。

第 $2$ 行读入 $N$ 个整数，整数之间用空格隔开。

第 $3$ 行读入 $M$ 个整数，整数之间用空格隔开。

输出

输出 $M$ 个整数，用空格隔开，第 $i$ 个整数，代表针对第 $i$ 次询问的结果。

AC代码

AC代码

#include <iostream>
#include <algorithm>
using namespace std;

const int N=1e5+10;
int n,m,a[N];

// 找左边界
int leftx(int x){
    int l=1,r=n,mid;
    while(l<=r){
        mid = (l+r)/2;
        if(x>a[mid]){
            l = mid+1;
        }else{
            r = mid-1;
        }
    }
    if(x==a[l]){
        return l;
    }else{
        return 0;
    }
}

// 找右边界
int rightx(int x){
    int l=1,r=n,mid;
    while(l<=r){
        mid = (l+r)/2;
        if(x>=a[mid]){
            l = mid+1;
        }else{
            r = mid-1;
        }
    }
    if(x==a[l-1]){
        return l-1;
    }else{
        return 0;
    }
}


int main() {
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        scanf("%d",&a[i]);
    }
    
    sort(a+1,a+1+n);
    int x;
    while(m--){
        scanf("%d",&x);
        
        // 得到左右边界，再相减 r-x+1
        int lx = leftx(x);
        int rx = rightx(x);
        if(lx==0){
            // 不存在
            cout<<0<<" ";
        }else{
            cout<<rx-lx+1<<" ";
        }
    }
    
    return 0;
}

P2437 蜜蜂路线

洛谷 | 普及- | C++14 | 提交时间: 2025-11-20

算法标签：高精度、递推、Fibonacci 数列

题目描述

一只蜜蜂在下图所示的数字蜂房上爬动,已知它只能从标号小的蜂房爬到标号大的相邻蜂房,现在问你：蜜蜂从蜂房 mmm 开始爬到蜂房 nnn，m<nm<nm<n，有多少种爬行路线？（备注：题面有误，右上角应为 n−1n-1n−1）

输入格式

输入 m,nm,nm,n 的值。

样例

样例输入 #1

1 14```

**样例输出 #1**

377```

AC 代码

#include <iostream>
#include <algorithm>
using namespace std;

// 求两个高精度的整数之和
string add(string s1,string s2) {
    string r; // 存放总和
    int a[1000] = {0}, b[1000] = {0}, c[1000] = {0};
    int i;
    for(i = 0; i < s1.size(); i++) {
        a[i] = s1[s1.size() - i - 1] - '0';
    }
    for(i = 0; i < s2.size(); i++) {
        b[i] = s2[s2.size() - i - 1] - '0';
    }
    int len = max(s1.size(), s2.size());
    for(i = 0; i < len; i++) {
        c[i] += a[i] + b[i];
        if(c[i] >= 10) {
            c[i + 1] += c[i] / 10;
            c[i] %= 10;
        }
    }
    if(c[len] != 0) {
        len++;
    }
    for(i = len - 1; i >= 0; i--) {
        r += char(c[i] + '0');
    }
    return r;
}

int main() {
    int M, N;
    cin >> M >> N;
    
    int d = N - M;
    
    if (d == 0) {
        cout << "1";
        return 0;
    }
    
    string a = "1", b = "1";
    
    // 计算从f(0)=1, f(1)=1开始的斐波那契数列
    for (int i = 2; i <= d; i++) {
        string c = add(a, b);
        a = b;
        b = c;
    }
    
    cout << b;
    return 0;
}

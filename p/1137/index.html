1137 - 纯粹素数

题目描述

纯粹素数是这样定义的：一个素数，去掉最高位，剩下的数仍为素数，再去掉剩下的数的最高位，余下的数还是素数。这样下去一直到最后剩下的个位数也还是素数。

求出所有小于 300030003000 的四位的纯粹素数。

输入

无

输出

按从小到大的顺序输出若干个纯粹素数，每行一个。

AC代码

#include <iostream>
#include <cmath>
using namespace std;

// 判断是否是素数
bool isPrime(int n) {
    if (n <= 1) {
        return false;
    }
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) {
            return false;
        }
    }
    return true;
}

int main() {
    int i,n;
    for(i = 1000;i < 3000;i++) { // 1234
        n = i; // 不能修改i的值
        if(isPrime(n)) {
            n = n % 1000; // 234
            if(isPrime(n)) {
                n = n % 100; // 34
                if(isPrime(n)) {
                    n = n % 10; // 4
                    if(isPrime(n)) {
                        cout << i << endl;
                    }
                }
            }
        }
    }
}

1946 - 高精度减法3

题目描述

处理两个高精度数的减法。(1000位内)

输入

两行，每行一个整数。（可能为负数，负号用“-”表示）

输出

一个整数，即两个数相减的结果。

解题思路

AC代码

AC代码

#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

// 比较两个非负整数字符串大小：a > b 返回1，a == b 返回0，a < b 返回-1
int cmp(const string &a, const string &b) {
    if (a.length() != b.length()) return a.length() > b.length() ? 1 : -1;
    for (int i = 0; i < a.length(); i++) {
        if (a[i] != b[i]) return a[i] > b[i] ? 1 : -1;
    }
    return 0;
}

// 大数加法（a, b >= 0），返回 a + b
string add(const string &a, const string &b) {
    string res;
    int i = a.length() - 1, j = b.length() - 1, carry = 0;
    while (i >= 0 || j >= 0 || carry) {
        int sum = carry;
        if (i >= 0) sum += a[i--] - '0';
        if (j >= 0) sum += b[j--] - '0';
        res.push_back(sum % 10 + '0');
        carry = sum / 10;
    }
    reverse(res.begin(), res.end());
    return res;
}

// 大数减法（a >= b >= 0），返回 a - b
string sub(const string &a, const string &b) {
    string res;
    int i = a.length() - 1, j = b.length() - 1, borrow = 0;
    while (i >= 0) {
        int diff = (a[i] - '0') - borrow;
        if (j >= 0) diff -= (b[j] - '0');
        if (diff < 0) {
            diff += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        res.push_back(diff + '0');
        i--; j--;
    }
    // 去除前导零
    while (res.length() > 1 && res.back() == '0') res.pop_back();
    reverse(res.begin(), res.end());
    return res;
}

int main() {
    string s1, s2;
    cin >> s1 >> s2;

    // 提取符号和绝对值
    bool sign1 = true, sign2 = true;  // true为正
    string num1 = s1, num2 = s2;
    if (s1[0] == '-') { sign1 = false; num1 = s1.substr(1); }
    if (s2[0] == '-') { sign2 = false; num2 = s2.substr(1); }

    // 计算结果
    string ans;
    if (sign1 == sign2) {  // 同号，做减法
        int resCmp = cmp(num1, num2);
        if (resCmp == 0) {
            ans = "0";
        } else if (resCmp > 0) {
            ans = sub(num1, num2);
            if (!sign1) ans = "-" + ans;
        } else {
            ans = sub(num2, num1);
            if (sign1) ans = "-" + ans;  // 注意：若同为正且 num1<num2，结果应为负
            else ans = ans;              // 同为负且 num1<num2，结果为正
        }
    } else {  // 异号，做加法
        ans = add(num1, num2);
        if (!sign1) ans = "-" + ans;  // 结果符号与第一个数相同
    }

    cout << ans << endl;
    return 0;
}

3863 - 黑白方块

题目描述

小杨有一个 $n$ 行 $m$ 列的网格图，其中每个格子要么是白色，要么是黑色。

对于网格图中的一个子矩形，小杨认为它是平衡的当且仅当其中黑色格子与白色格子数量相同。

小杨想知道最大的平衡子矩形包含了多少个格子。

输入

第一行包含两个正整数 $n，m$ ，含义如题面所示。

之后 $n$ 行，每行一个长度为 $m$ 的 $01$ 串，代表网格图第 $i$ 行格子的颜色，如果为 $0$，则对应格子为白色，否则为黑色。

输出

输出一个整数，代表最大的平衡子矩形包含格子的数量，如果不存在则输出 $0$。

解题思路

AC代码

AC代码

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

char a[15][15];
int ans = 0;

// 检测矩阵是否平衡
bool check(int a2, int b2, int a1, int b1) {
    int w = 0, b = 0;
    for (int i = a2; i <= a1; i++) {
        for (int j = b2; j <= b1; j++) {
            if (a[i][j] == '0') {
                w++;
            } else {
                b++;
            }
        }
    }
    return w == b;
}

int main() {
    int n, m;
    cin >> n >> m;  // 输入矩阵的行数和列数
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> a[i][j]; // 输入矩阵的元素
        }
    }
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) { // (i,j)作为左上角坐标 
            for (int k = i; k <= n; k++) {
                for (int l = j; l <= m; l++) { // (k,l)作为右下角坐标 
                    if (check(i, j, k, l) == 1) // 检测当前矩阵是否平衡 
                        ans = max(ans, (k - i + 1) * (l - j + 1));
                }
            }
        }
    }
    
    cout << ans << endl;
    return 0;
}

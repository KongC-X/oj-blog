B4532 [信息与未来 2026] 折纸

洛谷 | 普及| C++14 | 提交时间: 2026-05-26

算法标签：数学、江苏、2026、信息与未来

题目描述

Dr. X 拿出了一张正方形纸，将其划分成 2n×2n2^n \times 2^n2n×2n 个大小相等的小格子。纸有正、反两面：初始时，纸正面朝上。Dr. X 将会按照指定的顺序折叠这张纸，每次折叠，都沿中线将纸对折，共分四种方向，如图所示：

:::align{center}

:::

  • D (Down)：上半部分向下折，盖在下半部分上。
  • U (Up)：下半部分向上折，盖在上半部分上。
  • R (Right)：左半部分向右折，盖在右半部分上。
  • L (Left)：右半部分向左折，盖在左半部分上。

对折时，被翻折的部分会整体翻转（被翻折的部分正面和反面互换，且原本在最底层的格子折叠后会翻到最上层），翻转后的部分作为一个整体，叠加到另一半的上方。经过 2n2n2n 次折叠后，纸最终被折成一个 1×11 \times 11×1 的方块，由 22n2^{2n}22n 层格子叠成。Dr. X 想知道：初始时位于左上角 (1,1)(1, 1)(1,1) 的那个格子，最终在方块中的第几层（从上往下数，最上面一层为第 111 层），以及它的正面是朝上还是朝下。

输入格式

输入共两行。第一行一个正整数 nnn，表示纸张的边长为 2n2^n2n。第二行一个由 D、U、R、L 组成的字符串，长度为 2n2n2n，描述折叠的顺序。保证其中恰好包含 nnn 个纵向折叠（D 或 U）和 nnn 个横向折叠（R 或 L）。

输出格式

输出两个值，用空格分隔：第一个是一个整数，表示左上角格子从上往下数所在的层数；第二个是一个字符，U 表示正面朝上，D 表示正面朝下。

样例

样例输入 #1

1
DR```

**样例输出 #1**

2 U```

样例输入 #2

2
DRUR```

**样例输出 #2**

11 D```

AC 代码

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n;                         // 边长指数，纸张大小为 2^n × 2^n
    string s;                      // 操作序列，长度 2n
    cin >> n >> s;

    // 当前纸张的行数 h、列数 w
    long long h = 1LL << n;
    long long w = 1LL << n;

    // (x, y)：初始左上角 (1,1) 在当前纸张中的坐标（1‑based）
    long long x = 1, y = 1;

    // face: 0 表示正面朝上，1 表示正面朝下
    int face = 0;

    // total: 当前每个坐标位置上堆叠的格子层数
    long long total = 1;

    // layer: (1,1) 在它当前所处单元格内从上往下的层号（1‑based）
    long long layer = 1;

    // 依次处理每一个折叠操作
    for (int i = 0; i < s.size(); i++) {
        char op = s[i];

        /*
         * 折叠规则：
         * - 操作 D / U 把高度 h 减半，操作 R / L 把宽度 w 减半。
         * - 每次折叠后，每个坐标位置的堆叠层数 total 翻倍。
         * - 被折部分整体会翻转（正反面颠倒，内部层序颠倒），然后盖在静止部分的上方。
         * - 若 (1,1) 在被折部分：坐标需要按对称轴翻转，朝向取反，
         *   层号在堆叠内颠倒：layer = total - layer + 1。
         * - 若 (1,1) 在静止部分：坐标只需平移，层号增加被折部分的总层数（total），
         *   因为被折部分整个压在了上面。
         */

        if (op == 'D') {                     // 上半部分向下折
            if (x <= h / 2) {                // (1,1) 在移动的上半部分
                // 坐标映射：上半第 x 行折到下侧，变成下半第 (h/2 - x + 1) 行
                x = h / 2 - x + 1;
                // 被折部分整体翻转：朝向反转，内部层顺序颠倒
                face = 1 - face;
                layer = total - layer + 1;
            } else {                         // (1,1) 在静止的下半部分
                // 行号平移：去掉上半部分所占的行
                x = x - h / 2;
                // 上半部分盖在上面 → 层号增加上半的所有层
                layer = layer + total;
                // 朝向不变
            }
            h = h / 2;                       // 高度减半
            total = total * 2;               // 单元格层数翻倍
        }
        else if (op == 'U') {                // 下半部分向上折
            if (x > h / 2) {                 // (1,1) 在移动的下半部分
                // 坐标映射：下半第 x 行翻转到上半，新行号为 h - x + 1
                x = h - x + 1;
                face = 1 - face;
                layer = total - layer + 1;
            } else {                         // 在静止的上半部分
                // x 不变
                // 下半部分盖在上面，层号增加下半的总层数
                layer = layer + total;
            }
            h = h / 2;
            total = total * 2;
        }
        else if (op == 'R') {                // 左半部分向右折
            if (y <= w / 2) {                // (1,1) 在移动的左半部分
                // 列坐标映射：左第 y 列折到右半，新列号为 w/2 - y + 1
                y = w / 2 - y + 1;
                face = 1 - face;
                layer = total - layer + 1;
            } else {                         // 在静止的右半部分
                // 列号平移
                y = y - w / 2;
                // 左半部分盖在上面，层号增加左半的总层数
                layer = layer + total;
            }
            w = w / 2;
            total = total * 2;
        }
        else if (op == 'L') {                // 右半部分向左折
            if (y > w / 2) {                 // (1,1) 在移动的右半部分
                // 列坐标映射：右第 y 列折到左半，新列号为 w - y + 1
                y = w - y + 1;
                face = 1 - face;
                layer = total - layer + 1;
            } else {                         // 在静止的左半部分
                // y 不变
                // 右半部分盖在上面，层号增加右半的总层数
                layer = layer + total;
            }
            w = w / 2;
            total = total * 2;
        }
    }

    // 折叠结束后纸张为 1×1，layer 即为全局从上往下的层号
    cout << layer << ' ' << (face == 0 ? 'U' : 'D') << '\n';

    return 0;
}

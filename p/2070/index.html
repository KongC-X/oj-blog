东方博宜 入门
并查集

2070 - 道路规划

题目描述

有 NNN 个村庄，编号从 111 到 NNN ，请你建立一些道路，使得任意两个村庄可以相连。我们说两个村 AAA 和 BBB 是相连的，当且仅当有一条公路在 AAA 与 BBB之间，或存在一个村庄 CCC，有一条的道路连接着 AAA 和 CCC 之间,一条连接着 BBB 和 CCC 。

我们知道，已经有一些道路在一些村庄之间，你的工作是再建一些道路，使得所有的村庄连接并且所有的道路建造总和最低。

输入

第一行是一个整数 NNN (3≤N≤1003 \le N \le 1003≤N≤100)，表示村庄的数量。

接下来 NNN 行，每行包含 NNN 个整数，在这 NNN 行中的第 iii 行的第 jjj 个整数 $A

ij$ 表示村庄 $i$ 到村庄 $j$ 建立道路的费用 ($1 \le A

ij
$≤1000\le 1000≤1000)。

然后有一个整数 QQQ (0≤Q≤N×(N+1)/20 \le Q \le N \times (N + 1)/20≤Q≤N×(N+1)/2)。

接下来再 QQQ 行，每一行包含两个整数 aaa 和 bbb (1≤a<b≤N1 \le a \lt b \le N1≤a<b≤N)，即村庄 aaa 和 bbb 之间已经建成道路。

输出

你应该输出一行包含一个整数，表示所有村庄都连接所要建设道路的最小费用。

AC代码

#include <bits/stdc++.h>
using namespace std;

int f[110];                      // 并查集父节点数组

// 边结构体
struct Node {
    int x, y, len;               // x,y:村庄编号, len:修建费用
} a[10010];                      // 最多约4950条边，开10010足够

int n, Q, k;                     // n:村庄数, Q:已有道路数, k:边的总数
int ans;                         // 最终的最小费用

// 排序比较函数：按边权从小到大
bool cmp(Node a, Node b) {
    return a.len < b.len;
}

// 并查集查找根节点（路径压缩）
int find(int x) {
    return x == f[x] ? x : f[x] = find(f[x]);
}

int main() {
    cin >> n;

    // 读入邻接矩阵，只存上三角（i < j），避免重复边
    int cost;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> cost;
            if (i < j) {
                k++;             // 边数加1
                a[k].x = i;
                a[k].y = j;
                a[k].len = cost;
            }
        }
    }

    // 初始化并查集
    for (int i = 1; i <= n; i++) f[i] = i;

    // 读入已有道路数
    cin >> Q;
    int c = 0;                   // 已经修好的道路数（包括已有道路和新修道路）
    // 将已有道路直接合并（费用为0）
    for (int i = 0; i < Q; i++) {
        int x, y;
        cin >> x >> y;
        int fx = find(x);
        int fy = find(y);
        if (fx != fy) {
            f[fx] = fy;
            c++;                 // 已经连通的道路数增加
        }
    }

    // 如果已有道路已经使所有村庄连通，直接输出0
    if (c == n - 1) {
        cout << 0 << endl;
        return 0;
    }

    // 将所有边按费用从小到大排序
    sort(a + 1, a + k + 1, cmp);

    int ans = 0;                 // 还需要花费的总费用
    // 遍历排序后的边，执行Kruskal
    for (int i = 1; i <= k; i++) {
        int fx = find(a[i].x);
        int fy = find(a[i].y);
        if (fx != fy) {
            f[fx] = fy;          // 合并两个集合
            ans += a[i].len;     // 累加修建费用
            c++;                 // 修路数加1
        }
        // 当连通的道路数达到 n-1 时，所有村庄已连通
        if (c == n - 1) {
            cout << ans << endl;
            return 0;
        }
    }

    return 0;
}

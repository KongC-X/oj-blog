B2156 最长单词 2

洛谷 | 入门 | C++14 | 提交时间: 2026-07-12

算法标签：字符串（入门）

题目描述

一个以 . 结尾的简单英文句子，单词之间用空格分隔，没有缩写形式和其它特殊形式，求句子中的最长单词。

输入格式

一个以 . 结尾的简单英文句子（长度不超过 500500500），单词之间用空格分隔，没有缩写形式和其它特殊形式。

输出格式

该句子中最长的单词。如果多于一个，则输出第一个。

样例

样例输入 #1

I am a student of Peking University.```

**样例输出 #1**

University```

AC 代码

#include <bits/stdc++.h>
using namespace std;

int main() {
    // 思路：
    // 1. 读入整行句子（以 '.' 结尾）。
    // 2. 逐个字符遍历，如果是字母则累加到当前单词 cur 中。
    // 3. 遇到非字母（空格或句号）时，说明一个单词结束。
    //    计算当前单词长度，如果比已知最长的还长，就更新最长单词和长度。
    //    然后清空 cur 准备接收下一个单词。
    // 4. 最后输出最长单词。

    string s;
    getline(cin, s);                     // 读入整行句子

    string ans = "";                     // ans: 当前找到的最长单词
    int max_len = -1;                    // max_len: 最长单词的长度，初始-1保证第一次必更新
    string cur = "";                     // cur: 正在构建的当前单词

    for (int i = 0; i < s.size(); i++) {
        char c = s[i];
        // 判断是否为英文字母（大小写）
        if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')) {
            cur += c;                    // 是字母，追加到当前单词
        } else {
            // 遇到分隔符（空格或最后的句号），表示一个单词结束
            if (!cur.empty()) {          // 有单词才处理
                int len = cur.size();
                if (len > max_len) {     // 找到更长的单词
                    max_len = len;
                    ans = cur;           // 更新答案
                }
                cur = "";                // 清空，准备接收下一个单词
            }
        }
    }

    // 输出结果
    cout << ans << endl;
    return 0;
}

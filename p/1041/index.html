1041 - 判断三个整数是否相邻

题目描述

判断三个整数是否相邻，是输出 TRUE，否则输出FALSE。

输入

输入只有一行，包括3个整数。

输出

输出只有一行。

解题思路

AC代码

AC代码

#include <iostream>
using namespace std;
int main(){
    int a,b,c;
    cin >>a>>b>>c;
    //先排序，再相加减判断
    //完成第一轮比较 
    if(a<b){
        int t=a;
        a = b;
        b = t;
    }
    if(b<c){
        int t=b;
        b = c;
        c = t;
    }
    //进行第二轮比较
    if(a<b){
        int t=a;
        a = b;
        b = t;
    }
    //如果a-1等于b ， 同时 b-1等于c ,那么三个数则相邻。
    if(a-1==b && b-1==c){
        cout << "TRUE";
    }else{
        cout << "FALSE";
    }
    return 0;
}

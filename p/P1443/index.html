P1443 马的遍历

洛谷 | 普及| C++14 | 提交时间: 2026-02-12

算法标签：搜索、Special Judge、广度优先搜索 BFS、队列

题目描述

有一个 n×mn \times mn×m 的棋盘，在某个点 (x,y)(x, y)(x,y) 上有一个马，要求你计算出马到达棋盘上任意一个点最少要走几步。

输入格式

输入只有一行四个整数，分别为 n,m,x,yn, m, x, yn,m,x,y。

输出格式

一个 n×mn \times mn×m 的矩阵，代表马到达某个点最少要走几步（不能到达则输出 −1-1−1）。

样例

样例输入 #1

3 3 1 1

样例输出 #1

0 3 2    
3 -1 1    
2 1 4    ```

## AC 代码

```cpp
#include <iostream>
#include <cstring>
using namespace std;

const int MAXN = 405;
const int MAXM = 405;

int n, m, x, y;
int dist[MAXN][MAXM];

// 马可以走的8个方向
int dx[8] = {1, 2, 2, 1, -1, -2, -2, -1};
int dy[8] = {2, 1, -1, -2, -2, -1, 1, 2};

// 手写队列
int qx[MAXN * MAXM];
int qy[MAXN * MAXM];
int front = 0, rear = 0;

void bfs(int sx, int sy) {
    // 初始化距离数组为-1
    memset(dist, -1, sizeof(dist));
    dist[sx][sy] = 0;
    
    // 起点入队
    qx[rear] = sx;
    qy[rear] = sy;
    rear++;
    
    while (front < rear) {
        int cx = qx[front];
        int cy = qy[front];
        front++;
        
        for (int i = 0; i < 8; i++) {
            int nx = cx + dx[i];
            int ny = cy + dy[i];
            if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && dist[nx][ny] == -1) {
                dist[nx][ny] = dist[cx][cy] + 1;
                qx[rear] = nx;
                qy[rear] = ny;
                rear++;
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    cin >> n >> m >> x >> y;
    
    bfs(x, y);
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cout << dist[i][j];
            if (j < m) cout << ' ';
        }
        if (i < n) cout << '\n';
    }
    
    return 0;
}

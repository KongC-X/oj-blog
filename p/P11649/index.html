P11649 [COCI 2024/2025 #4] 棋 / Šah

洛谷 | 普及- | C++14 | 提交时间: 2026-08-19

算法标签：2025、COCI（克罗地亚）

题目背景

译自 COCI 2024/2025 #4 T1。3s,0.5G\texttt{3s,0.5G}3s,0.5G。满分为 505050。

题目描述

有一块 n×nn\times nn×n 的国际象棋棋盘，上面摆放着 mmm 个棋子。棋子的类型只有骑士（knight），车（rook）和皇后（queen）。

已知

  • 骑士的攻击范围为：骑士所在的格子，以及骑士所在的格子走一列两行（或者一行两列）能走到的所有格子；
  • 车的攻击范围为：与车所在的格子同一行或者同一列的格子（包括车所在的格子）；
  • 皇后的攻击范围为：与皇后所在的格子同一行，同一列或者同一对角线的格子（包括皇后所在的格子）。

求出这个棋盘上有多少个格子被至少一个棋子攻击。

输入格式

第一行，两个正整数 n,mn,mn,m。

接下来 mmm 行，每行一个字符和两个正整数 C,x,yC,x,yC,x,y，表示有一枚类型 CCC 的棋子在 (x,y)(x,y)(x,y)。

  • C=NC=\texttt{N}C=N，表示骑士；
  • C=RC=\texttt{R}C=R，表示车；
  • C=QC=\texttt{Q}C=Q，表示皇后。

数据保证不会有两个棋子在一个格子里。

输出格式

输出一行一个正整数，表示答案。

样例

样例输入 #1

7 1
Q 4 4```

**样例输出 #1**

25```

样例输入 #2

5 1
N 3 3```

**样例输出 #2**

9```

样例输入 #3

6 3
R 1 4
Q 2 1
N 5 2```

**样例输出 #3**

25```

AC 代码

#include <bits/stdc++.h>
using namespace std;

int n, m;                // n: 棋盘边长, m: 棋子数量
bool vis[205][205];      // vis[i][j] 表示格子 (i,j) 是否被攻击

// 标记骑士攻击范围（含自身）
void markN(int x, int y) {
    int dx[8] = {-2, -2, -1, -1, 1, 1, 2, 2};
    int dy[8] = {-1, 1, -2, 2, -2, 2, -1, 1};
    vis[x][y] = true;
    for (int i = 0; i < 8; i++) {
        int nx = x + dx[i], ny = y + dy[i];
        if (nx >= 1 && nx <= n && ny >= 1 && ny <= n)
            vis[nx][ny] = true;
    }
}

// 标记车攻击范围（含自身）
void markR(int x, int y) {
    // 同一列上下
    for (int i = 1; i <= n; i++) vis[i][y] = true;
    // 同一行左右
    for (int j = 1; j <= n; j++) vis[x][j] = true;
}

// 标记皇后攻击范围（含自身）
void markQ(int x, int y) {
    markR(x, y);  // 先覆盖车能攻击的行列
    // 四条对角线
    for (int i = x, j = y; i >= 1 && j >= 1; i--, j--) vis[i][j] = true;
    for (int i = x, j = y; i <= n && j <= n; i++, j++) vis[i][j] = true;
    for (int i = x, j = y; i >= 1 && j <= n; i--, j++) vis[i][j] = true;
    for (int i = x, j = y; i <= n && j >= 1; i++, j--) vis[i][j] = true;
}

int main() {
    cin >> n >> m;
    char C; int x, y;
    for (int i = 0; i < m; i++) {
        cin >> C >> x >> y;
        if (C == 'N') markN(x, y);
        else if (C == 'R') markR(x, y);
        else if (C == 'Q') markQ(x, y);
    }

    int ans = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            if (vis[i][j]) ans++;

    cout << ans << endl;
    return 0;
}

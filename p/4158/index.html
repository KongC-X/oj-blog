4158 - 武器强化

题目描述

小杨有 nnn 种武器和 mmm 种强化材料。第 iii 种强化材料会适配第 pip_ipi​ 种武器，小杨可以花费 cic_ici​ 金币将该材料对应的适配武器修改为任意武器。

小杨最喜欢第 111 种武器，因此他希望适配该武器的强化材料种类数严格大于其他的武器，请你帮小杨计算为了满足该条件最少需要花费多少金币。

输入

第一行包含两个正整数 n,mn,mn,m，含义如题面所示。

之后 mmm行，每行包含两个正整数 pip_ipi​，cic_ici​代表第 iii 种强化材料的适配武器和修改花费。

输出

输出一个整数，代表能够使适配第 111 种武器的强化材料种类数严格大于其他的武器最少需要花费的金币。

解题思路

AC代码

AC代码

#include <iostream>
#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    
    // 存储武器1的材料数量
    int cnt1 = 0;
    
    // 存储非1武器的材料：weapons[i][0..len[i]-1]存储花费
    long long weapons[1005][1005] = {0};
    int len[1005] = {0}; // 每个武器存储的材料数量
    
    // 输入处理
    for (int i = 0; i < m; i++) {
        int p;
        long long c;
        cin >> p >> c;
        
        if (p == 1) {
            cnt1++; // 武器1的材料
        } else {
            // 存储到对应武器的数组
            weapons[p][len[p]] = c;
            len[p]++;
        }
    }
    
    // 对每个非1武器的材料进行排序
    for (int i = 2; i <= n; i++) {
        if (len[i] > 0) {
            sort(weapons[i], weapons[i] + len[i]);
        }
    }
    
    long long ans = 1e18; // 初始化为一个大数
    
    // 枚举目标值T：武器1最终的材料数
    for (int T = cnt1; T <= m; T++) {
        long long total_cost = 0; // 当前T的总花费
        long long D = 0;          // 必须移走的材料总数
        long long rest[1005];     // 存储剩余材料
        int rest_count = 0;       // 剩余材料数量
        
        // 处理每个非1武器
        for (int i = 2; i <= n; i++) {
            // 需要移走的材料数
            int d_i = len[i] - (T - 1);
            if (d_i < 0) d_i = 0;
            
            D += d_i;
            
            // 累加必须移走的材料花费（最小的d_i个）
            for (int j = 0; j < d_i; j++) {
                total_cost += weapons[i][j];
            }
            
            // 收集剩余材料
            for (int j = d_i; j < len[i]; j++) {
                rest[rest_count++] = weapons[i][j];
            }
        }
        
        // 计算还需要添加的材料数
        long long rem = T - cnt1 - D;
        if (rem < 0) rem = 0;  // 确保非负
        
        if (rest_count >= rem) {
            // 对剩余材料排序（只在需要时排序）
            if (rest_count > 0 && rem > 0) {
                sort(rest, rest + rest_count);
            }
            
            // 累加额外花费
            for (int j = 0; j < rem; j++) {
                total_cost += rest[j];
            }
            
            // 更新最小花费
            if (total_cost < ans) {
                ans = total_cost;
            }
        }
    }
    
    cout << ans << endl;
    return 0;
}

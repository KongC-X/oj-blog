1216 - 数塔问题

题目描述

有如下所示的数塔，要求从底层走到顶层，若每一步只能走到相邻的结点，则经过的结点的数字之和最大是多少？

输入

输入数据首先包括一个整数整数 NNN (1≤N≤1001 \le N \le 1001≤N≤100)，表示数塔的高度，接下来用 NNN 行数字表示数塔，其中第 iii 行有个 iii 个整数，且所有的整数均在区间 [0,99][0,99][0,99] 内。

输出

从底层走到顶层经过的数字的最大和是多少？

解题思路

AC代码

AC代码

#include <iostream>
#include <bits/stdc++.h>
using namespace std;

int n,a[110][110],ma;
int d[110][110]; 

// 深搜：从(1,1)开始，每次向下或向右走，求出所有路径和的最大值
void dfs(int x,int y,int sum) {
    ma = max(sum,ma);
    // 递归出口：当x > n时，说明已经遍历完所有点
    if(x <= n) {
        // cout << x << " " << y << " " << sum << endl;
        // 递归走：x+1,y 和 x+1,y+1
        dfs(x+1,y,sum+a[x+1][y]);
        dfs(x+1,y+1,sum+a[x+1][y+1]);
    }
}

// d(x,y): 表示x,y点到底层经过的数字和的最大值
// d(x,y) = a[x][y] + max(d(x+1,y),d(x+1,y+1))

// 动态规划：递推求解
void fun1() {
    int i,j;
    // 最后一层d[x][y] = a[x][y], 是问题的边界
    for(j = 1;j <= n;j++) {
        d[n][j] = a[n][j]; // 赋值最后一层
    }
    // 从倒数第二层开始递推
    for(i = n - 1;i >= 1;i--) {
        for(j = 1;j <= i;j++) {
            d[i][j] = a[i][j] + max(d[i+1][j],d[i+1][j+1]);
        }
    }
}

// d(x,y) = a[x][y] + max(d(x+1,y),d(x+1,y+1))
// 动态规划：递归求解
int fun2(int x,int y) {
    // 递归停止条件（出口）
    if(x < n) {
        return a[x][y] + max(fun2(x+1,y),fun2(x+1,y+1));
    } else {
        // x == n 最后一层的解就是自己
        return a[x][y];
    }
}

// 动态规划：递归求解+记忆化搜索
// 求解每个点的值，先判断该点的值是否已经计算过，如果计算过，直接拿曾经计算的结果
// 如果没有求解过，递归求解，并将结果保存到数组中
int fun3(int x,int y) {
    // 最后一行的值就是自己，不需要递归
    if(x == n) {
        return a[x][y];
    } else {
        // 如果该点的值已经计算过，直接拿
        if(d[x][y] != -1) {
            return d[x][y];
        } else {
            // 如果该点的值没有计算过，递归求解，并将结果保存到数组中
            d[x][y] = a[x][y] + max(fun3(x+1,y),fun3(x+1,y+1));
            return d[x][y];
        }
    }
}

int main() {
    cin >> n;
    int i,j;
    for(i = 1;i <= n;i++) {
        for(j = 1;j <= i;j++) {
            cin >> a[i][j];
        }
    }
    // 方法一：深搜
    // dfs(1,1,a[1][1]);
    // cout << ma << endl;

    // 方法二：动态规划(递推)
    // fun1();
    // cout << d[1][1] << endl;

    // 方法三：动态规划(递归)
    // cout << fun2(1,1) << endl;

    // 方法四：动态规划(递归+记忆化搜索)
    // 将d数组初始化为-1，方便判断是否已经计算过
    // memset(数组，初始化的值，数组大小)：
    // 将数组的值初始化，初始化的值只能是0或-1
    memset(d,-1,sizeof(d));
    // for(i = 1;i <= n;i++) {

}

1353 - 轴对称三角形

题目描述

在数学上，我们发现有一类图形是对称图形。我们对于左右一样的图形叫做沿 yyy 轴对称，对于上下一样的图形叫做沿 xxx 轴对称。

如下图所示的图形就是沿 yyy 轴对称的两个三角形：

而下图所示的图形就是沿 xxx 轴对称的两个三角形：

请从键盘读入一个整数 nnn ，代表三角形的行数，再读入一个字符，该字符为’ xxx ’或者’ yyy ’，代表是沿 xxx 轴对称还是沿 yyy 轴对称，输出对应的轴对称图形。（ 8.18.18.1 ）

输入

第一样是一个整数 n（n≤10n（n \le 10n（n≤10 ）

第二行是一个字符（字符’ xxx ’或者字符’ yyy ’）

输出

按照要求输出的轴对称图形。

解题思路

AC代码

AC代码

#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    char z;
    cin>>n>>z;
    int i,j;
    //沿x轴对称
    if(z=='x') {
        for(i=1; i<=n; i++) {
            //n-i个空格
            for(j = 1; j <= n - i; j++) {
                cout<<" ";
            }
            //2*i-1个星
            for(j=1; j<=2*i-1; j++) {
                cout<<"*";
            }
            cout<<endl;
        }

        //将上面的三角倒过来
        for(i=n; i>=1; i--) {
            //n-i个空格
            for(j = 1; j <= n - i; j++) {
                cout<<" ";
            }
            //2*i-1个星
            for(j=1; j<=2*i-1; j++) {
                cout<<"*";
            }
            cout<<endl;
        }
        //沿y轴对称
    } else {
        for(i=1; i<=n; i++) {
            //n-i个空格
            for(j = 1; j <= n - i; j++) {
                cout<<" ";
            }
            //2*i-1个星
            for(j=1; j<=2*i-1; j++) {
                cout<<"*";
            }
            //n-i个空格
            for(j = 1; j <= 2*(n - i); j++) {
                cout<<" ";
            }
            //2*i-1个星
            for(j=1; j<=2*i-1; j++) {
                cout<<"*";
            }
            cout<<endl;
        }
    }
    return 0;
}

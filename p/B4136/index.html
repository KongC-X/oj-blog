B4136 [信息与未来 2016] 数列求和

洛谷 | 普及- | C++14 | 提交时间: 2026-04-13

算法标签：数学、2016、江苏、信息与未来

题目描述

小明写出了一个数列，第 iii 项 aia_iai​ 的值为 i2i^2i2。数列从第一项 (i=1)(i=1)(i=1) 开始如下：1,4,9,16,25,…1,4,9,16,25,\dots1,4,9,16,25,…，编程求出这个数列前 nnn 项的和。

输入格式

一行，一个整数 nnn。

输出格式

一行，一个整数：a1+a2+⋯+ana_1+a_2+\dots+a_na1​+a2​+⋯+an​ 的值。

样例

样例输入 #1

6```

**样例输出 #1**

91```

样例输入 #2

987```

**样例输出 #2**

320988850```

AC 代码

#include <iostream>
using namespace std;

// 自定义函数：输出 __int128 类型的数字
void print128(__int128 x) {
    if (x == 0) {
        cout << 0;
        return;
    }
    // 如果 x 是负数，输出负号并转为正数
    if (x < 0) {
        cout << '-';
        x = -x;
    }
    // 把每一位数字存入字符数组（从低位到高位）
    char buf[50];
    int pos = 0;
    while (x > 0) {
        buf[pos++] = (x % 10) + '0';
        x /= 10;
    }
    // 倒序输出（从高位到低位）
    for (int i = pos - 1; i >= 0; i--) {
        cout << buf[i];
    }
}

int main() {
    long long n;          // 输入可以用 long long
    cin >> n;

    __int128 N = n;       // 转成 __int128 防止计算时溢出
    // 平方和公式：n*(n+1)*(2n+1)/6
    __int128 ans = N * (N + 1) * (2 * N + 1) / 6;

    print128(ans);        // 输出结果
    cout << endl;

    return 0;
}

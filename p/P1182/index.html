P1182 数列分段 Section II

洛谷 | 普及| C++14 | 提交时间: 2026-02-08

算法标签：贪心、二分、前缀和

题目描述

对于给定的一个长度为 $N$ 的正整数数列 $A_{1\sim N}$，现要将其分成 $M$（$M\leq N$）段，并要求每段连续，且每段和的最大值最小。

关于最大值最小：

例如一数列 $4\ 2\ 4\ 5\ 1$ 要分成 $3$ 段。

将其如下分段：

$$[4\ 2][4\ 5][1]$$

第一段和为 $6$，第 $2$ 段和为 $9$，第 $3$ 段和为 $1$，和最大值为 $9$。

将其如下分段：

$$[4][2\ 4][5\ 1]$$

第一段和为 $4$，第 $2$ 段和为 $6$，第 $3$ 段和为 $6$，和最大值为 $6$。

并且无论如何分段，最大值不会小于 $6$。

所以可以得到要将数列 $4\ 2\ 4\ 5\ 1$ 要分成 $3$ 段，每段和的最大值最小为 $6$。

输入格式

第 $1$ 行包含两个正整数 $N,M$。

第 $2$ 行包含 $N$ 个空格隔开的非负整数 $A_i$，含义如题目所述。

输出格式

一个正整数，即每段和最大值最小为多少。

样例

样例输入 #1

5 3
4 2 4 5 1```

**样例输出 #1**

6```

AC 代码

#include <iostream>
#include <bits/stdc++.h>
using namespace std;

// 定义全局数组存储数列，最大长度100010（根据题目N最大为10^5）
int arr[100010];
int n, m;  // n: 数列长度, m: 需要分成的段数

// 检查当每段和的最大值限制为limit时，能否将数列分成不超过m段
bool check(long long limit) {
    int cnt = 1;        // 当前段数，至少为1段（从第一个元素开始）
    long long s = 0;    // 当前段的和
    
    // 遍历数列中的每个元素
    for (int i = 0; i < n; ++i) {
        // 如果单个元素已经超过限制值，直接返回false（因为每个元素必须属于某一段）
        if (arr[i] > limit) return false;
        
        // 如果当前段加入当前元素后和不超过限制，则加入当前段
        if (s + arr[i] <= limit) {
            s += arr[i];
        } else {
            // 否则需要开启新的一段
            cnt++;
            s = arr[i];  // 新段的初始和就是当前元素
            
            // 如果当前段数已经超过m，说明limit太小，返回false
            if (cnt > m) return false;
        }
    }
    // 遍历完所有元素后，如果段数不超过m，则满足条件
    return true;
}

int main() {
    // 关闭同步流，提高cin/cout速度
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    // 输入数列长度n和需要分成的段数m
    cin >> n >> m;
    
    long long total = 0;  // 数列所有元素的总和
    int mx = 0;           // 数列中的最大值
    
    // 输入数列并计算总和和最大值
    for (int i = 0; i < n; ++i) {
        cin >> arr[i];
        total += arr[i];
        if (arr[i] > mx) mx = arr[i];
    }
    
    // 二分查找：左边界为数列最大值（每段至少包含一个元素），右边界为总和（只分成一段）
    long long l = mx, r = total, ans = total;
    
    // 二分查找模板：寻找满足条件的最小值
    while (l <= r) {
        long long mid = l + (r - l) / 2;  // 防止溢出的写法，等同于(l+r)/2
        
        // 检查mid是否满足条件
        if (check(mid)) {
            ans = mid;   // 记录当前可行解
            r = mid - 1; // 尝试寻找更小的解（减小每段和限制）
        } else {
            l = mid + 1; // 当前mid太小，需要增大每段和限制
        }
    }
    
    // 输出答案：每段和的最大值的最小可能值
    cout << ans << endl;
    return 0;
}

2191 - 树的重心（2）

题目描述

给定一棵树，树中有 $n$ 个结点（编号 $1 \sim n$）。
请求出，删除该重心后，剩余子树的最多结点数？

重心定义：重心是指树中的一个结点，如果将这个点删除后，剩余各个连通块中点数的最大值最小，那么这个结点被称为树的重心。

输入

第 $1$ 行读入一个整数 $n$ ，代表树的结点的数量（$1≤n≤10^5$）。

接下来 $n-1$ 行，每行读入两个整数 $x$ 和 $y$，表示结点 $x$ 和 $y$之间有一条边。（注意：不确定 $x$ 和 $y$ 的父子关系）

输出

输出一个整数，代表删除重心后，剩余子树的最多结点。

AC代码

#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;

// ---------- 链式前向星（邻接表） ----------
int pre[N];          // pre[x]：结点 x 的"最后一条出边"在 a 数组中的下标
struct node {
    int to;          // 这条边指向的目标结点
    int next;        // 上一条以相同起点出发的边的下标
} a[N * 2];          // 存边的数组，无向图每条边存两次

int n, k = 0;        // k：当前已使用的 a 数组下标

// 加一条从 u 到 v 的有向边
void add(int u, int v) {
    k++;
    a[k].to = v;
    a[k].next = pre[u]; // 新边的 next 指向 u 原来的"最后一条边"
    pre[u] = k;         // 更新 u 的"最后一条边"为当前新边
}
// -----------------------------------------

int fa[N];   // fa[x]：结点 x 的父结点
int si[N];   // si[x]：以 x 为根的子树大小（包含 x 本身）

/**
 * DFS 求子树大小和父子关系
 * @param x 当前结点
 * @param f 当前结点的父结点
 * @return 以 x 为根的子树大小
 */
int dfs(int x, int f) {
    si[x] = 1;          // 当前结点自己算 1 个
    fa[x] = f;          // 记录父结点

    // 遍历 x 的所有邻接点
    for (int i = pre[x]; i != 0; i = a[i].next) {
        int v = a[i].to;      // v 是 x 的邻居
        if (v != f) {          // 不走回头路（不能回到父结点）
            si[x] += dfs(v, x); // 递归求子树大小，并累加到 si[x]
        }
    }

    return si[x];  // 返回以 x 为根的子树大小
}

int main() {
    cin >> n;

    // 读入 n-1 条边，建图
    for (int i = 1; i <= n - 1; i++) {
        int x, y;
        cin >> x >> y;
        add(x, y);   // 加一条 x→y 的边
        add(y, x);   // 加一条 y→x 的边（双向建边）
    }

    // ========== 第一步：以 1 为根，求每个结点的子树大小 ==========
    dfs(1, 0);       // 1 的父结点设为 0（不存在）

    // ========== 第二步：计算重心对应的"最大连通块最小值" ==========
    int ans = INT_MAX;  // 记录最小的 r[i]，也就是最终答案

    for (int i = 1; i <= n; i++) {
        // 情况 1：父方向连通块大小 = 整棵树大小 - 以 i 为根的子树大小
        int maxPart = n - si[i];  // 这是"上面"那个连通块的大小

        // 情况 2：遍历 i 的所有孩子（子结点），取最大的子树大小
        for (int j = pre[i]; j != 0; j = a[j].next) {
            int v = a[j].to;       // v 是 i 的邻居
            // 判断 v 是不是 i 的孩子：v 的父结点是 i
            if (fa[v] == i) {
                maxPart = max(maxPart, si[v]);
            }
        }

        // 更新答案：记录所有 r[i] 中的最小值
        ans = min(ans, maxPart);
    }

    // 输出删除重心后，剩余子树的最多结点数
    cout << ans << endl;

    return 0;
}

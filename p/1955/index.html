1955 - 算24点

题目描述

给出 $n$ 组 $4$ 个整数，请问有多少组整数，在不改变顺序，且不加入括号的情况下，可以通过 $+$ $-$ $ \times $ 三种运算，得到 $24$ 。

比如 $1$ $2$ $3$ $4$ 四个数，可以通过如下的方法得到$24$：$1 \times 2 \times 3 \times 4=24$。

而 $20$ $30$ $40$ $50$ 四个数，在不改变顺序、不添加括号的情况下，通过 $+$ $-$ $ \times $ 三种运算是无法获得 $24$ 的。

输入

第1行有一个整数 $n$；（ $2 \le n \le 100$ ）

接下来 $n$ 行，每行有 $4$ 个整数 $a_i$ ；（ $1 \le a_i \le 100$ ）

输出

输出一个整数，代表有几组数能够通过题目的规则计算得到 $24$ 。

AC代码

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

/*
  24点计数 —— 枚举所有运算符组合
  
  关键：运算符优先级 × 优先于 + -
  计算时先处理所有 ×，再从左到右处理 + 和 -
  
  4个数，3个运算符位置，每个3种选择，共27种组合
*/

bool check(int a, int b, int c, int d, char op1, char op2, char op3) {
    // 用两个数组：vals存数值，ops存运算符（只有+和-，×已经算掉了）
    vector<int> vals;
    vector<char> ops;
    
    vals.push_back(a);  // 先放入第一个数
    
    // 处理 op1 和 b
    if(op1 == '*') {
        vals.back() *= b;  // 乘优先，直接算
    } else {
        vals.push_back(b);
        ops.push_back(op1);
    }
    
    // 处理 op2 和 c
    if(op2 == '*') {
        vals.back() *= c;
    } else {
        vals.push_back(c);
        ops.push_back(op2);
    }
    
    // 处理 op3 和 d
    if(op3 == '*') {
        vals.back() *= d;
    } else {
        vals.push_back(d);
        ops.push_back(op3);
    }
    
    // 现在 ops 里只有 + 和 -，从左到右计算
    int result = vals[0];
    for(int i = 0; i < (int)ops.size(); i++) {
        if(ops[i] == '+') {
            result += vals[i + 1];
        } else {
            result -= vals[i + 1];
        }
    }
    
    return result == 24;
}

int main() {
    int n;
    cin >> n;
    
    char op[] = {'+', '-', '*'};
    int ans = 0;  // 能算出24的组数
    
    for(int t = 1; t <= n; t++) {
        int a, b, c, d;
        cin >> a >> b >> c >> d;
        
        bool ok = false;
        // 枚举所有 27 种运算符组合
        for(int i = 0; i < 3 && !ok; i++) {
            for(int j = 0; j < 3 && !ok; j++) {
                for(int k = 0; k < 3 && !ok; k++) {
                    if(check(a, b, c, d, op[i], op[j], op[k])) {
                        ok = true;
                    }
                }
            }
        }
        if(ok) ans++;
    }
    
    cout << ans << endl;
    return 0;
}

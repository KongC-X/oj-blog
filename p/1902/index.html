1902 - 最少的修改次数

题目描述

现有整数 A1,A2,...AnA_1,A_2,...A_nA1​,A2​,...An​，修改最少的数字为实数（整数或者小数），使得数列严格单调递增。

输入

第一行，一个整数 nnn 。（n≤105n≤10^5n≤105）

第二行，nnn 个整数 AiA_iAi​ 。（Ai≤109A_i≤10^9Ai​≤109）

输出

111 个整数，表示最少修改的数字的数量。

解题思路

AC代码

AC代码

#include <iostream>
using namespace std;

const int MAX_N = 100010;
int a[MAX_N];
int dp[MAX_N]; // dp[i]表示长度为i的LIS的最小末尾元素

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    
    int len = 1; // LIS的初始长度
    dp[1] = a[1]; // 初始化第一个元素
    
    for (int i = 2; i <= n; i++) {
        if (a[i] > dp[len]) {
            // 如果当前元素大于dp数组最后一个元素，直接扩展
            dp[++len] = a[i];
        } else {
            // 二分查找左边界
            int l = 1, r = len, mid;
            while (l <= r) {
                mid = (l + r) / 2;
                if (a[i] <= dp[mid]) {
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            // 替换找到的位置
            dp[l] = a[i];
        }
    }
    
    cout << n - len << endl;
    return 0;
}

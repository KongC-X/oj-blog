东方博宜 入门 高精度运算
高精度算法

1296 - 求1!+2!+3!+4!+...+n!

题目描述

请求出 $1!+2!+3!+4!+...+n!$，请注意，$n \le 50$ 。

$n!=n*(n-1)(n-2)...1$ ，如：$5!=5432*1=120$ 。

输入

请输入一个整数 $n$（$n \le 50$）。

输出

输出求和的结果。

AC代码

AC代码

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

int a[1000] = {1};
// k代表a数组中元素的个数，代表2的n次方高精度的整数位数
// k2代表了高精度的总和的位数
int r[1000];
int n,i,j,k = 1,k2 = 1,len;
int main() {
    cin >> n;
    // 第一层循环是这个数要乘多少次2
    for(i = 1; i <= n; i++) {
        // 第二层是每次有几位要乘2
        for(j = 0;j < k;j++) {
            a[j] = a[j] * i;
        }
        // 处理进位
        for(j = 0;j < k + 2;j++) {
            if(a[j] >= 10) {
                a[j + 1] += a[j] / 10;
                a[j] = a[j] % 10;
            }
        }
        // 判断a数组是否多出一位还是两位
        if(a[k + 1] != 0) {
            k += 2;
        } else if(a[k] != 0) {
            k++;
        }
        
        // 求出了2的i次方，结果为k位
        // 将k位的2的i次方，加到k2位的总和r上
        len = k;
        if(k2 > k) len = k2;
        for(j = 0;j < len;j++) {
            r[j] += a[j];
            if(r[j] >= 10) {
                r[j + 1] += r[j] / 10;
                r[j] = r[j] % 10;
            }
        }
        // 判断r数组是否多了1位
        if(r[len] != 0) {
            len++;
        }
    }
    // 逆序输出
    for(i = len - 1;i >= 0;i--) {
        cout << r[i];
    }
}

2195 - 二叉排序树

题目描述

从键盘读入 $n$ 个不相同的整数，以每个整数作为结点的值，来创建一棵二叉排序树，假设读入的第 $1$ 个点是这棵树的根结点。

请求出这棵二叉排序树中序和后续遍历的结果？

输入

共两行，第一行为整数 $n$ ；

第二行为 $n$ 个不重复的整数 $a_i$ 。（，$1≤a_i≤10^5$，本题中$a_i$为随机生成的数值）

输出

共两行，第一行为中序遍历的结果，第二行为后序遍历的结果，同一行的输出用空格隔开。

AC代码

#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 10;
//孩子表示法
struct node {
    int data,lc,rc;
} a[N];

int n;

//中序遍历
void dfs1(int x) {
    if(a[x].lc) dfs1(a[x].lc);
    cout<<a[x].data<<" ";
    if(a[x].rc) dfs1(a[x].rc);
}

//后序遍历
void dfs2(int x) {
    if(a[x].lc) dfs2(a[x].lc);
    if(a[x].rc) dfs2(a[x].rc);
    cout<<a[x].data<<" ";
}


int main() {
    cin>>n;
    int x,p;
    for(int i = 1; i <= n; i++) {
        cin>>x;
        a[i].data = x;//将第i个结点存储
        //第1个结点只能是根
        if(i == 1) continue;

        //找一个结点p，作为当前结点的父
        p = 1;//从根开始比较
        while(1) {
            //如果在左子树
            if(x < a[p].data) {
                //如果有左子树
                if(a[p].lc) p = a[p].lc;
                else {
                    a[p].lc = i;//将编号为i的结点，作为p结点的左孩子
                    break;
                }
            } else {
                //在右子树
                if(a[p].rc) p = a[p].rc;
                else {
                    a[p].rc = i;
                    break;
                }
            }
        }
    }

    //中序，后序遍历
    dfs1(1);
    cout<<endl;
    dfs2(1);
    return 0;
}

1006 - 打印星号三角形

题目描述

打印星号三角形。

输入

输入只有一行，包括 111 个整数 nnn ， nnn 代表行数。

输出

输出 nnn 行。

解题思路

AC代码

AC代码

#include <bits/stdc++.h>
using namespace std;

int main(){
    //n代表输入的行数 
    int n;
    //输入n 
    cin>>n;
    int i,j,k;
    //循环行数 
    for(i=1;i<=n;i++){
        //循环3个三角形 
        for(j=1;j<=3;j++){
            //循环展示空格 
            for(k=1;k<=n-i;k++){
                cout<<" ";
            }
            //循环展示* 
            for(k=1;k<=2*i-1;k++){
                cout<<"*";
            }
            //循环展示空格 
            for(k=1;k<=n-i;k++){
                cout<<" ";
            }
        }
        //第i行结束，换行 
        cout<<endl;
    }
    return 0;
}

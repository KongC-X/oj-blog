# P1274 魔术数字游戏

> **洛谷** | 普及+/提高 | C++14 | 提交时间: 2026-07-28

**算法标签**：`搜索`

## 题目描述

填数字方格的游戏有很多种变化，如下图所示的 $4 \times 4$ 方格中，我们要选择从数字 $1$ 到 $16$ 来填满这十六个格子($A_{i,j}$ ，其中 $i=1 \cdots 4$ ，$j=1 \cdots 4$)。为了让游戏更有挑战性，我们要求下列六项中的每一项所指定的四个格子，其数字累加的和必须为 $34$ ：

| $A_{1,1}$ | $A_{1,2}$ | $A_{1,3}$ |  $A_{1,4}$|
| :----------- | :----------- | :----------- | :----------- |
| $A_{2,1}$ | $A_{2,2}$ | $A_{2,3}$ | $A_{2,4}$ |
| $A_{3,1}$ | $A_{3,2}$ | $A_{3,3}$ | $A_{3,4}$ |
| $A_{4,1}$ | $A_{4,2}$ | $A_{4,3}$ | $A_{4,4}$ |

- 四个角落上的数字，即 $A_{1,1}+A_{1,4}+A_{4,1}+A_{4,4}=34$ 。
- 每个角落上的 $2 \times 2$ 方格中的数字，例如左上角 $A_{1,1}+A_{1,2}+A_{2,1}+A_{2,2}=34$ 。
- 最中间的 $2 \times 2$ 方格中的数字，即 $A_{2,2}+A_{2,3}+A_{3,2}+A_{3,3}=34$ 。
- 每条水平线上四个格子中的数字，即 $A_{i,1}+A_{i,2}+A_{i,3}+A_{i,4}=34$，其中 $i=1 \cdots 4$ 。
- 每条垂直线上四个格子中的数字，即 $A_{1,j}+A_{2,j}+A_{3,j}+A_{4,j}=34$，其中 $j=1 \cdots 4$ 。
- 两条对角线上四个格子中的数字，例如左上角到右下角 $A_{1,1}+A_{2,2}+A_{3,3}+A_{4,4}=34$ 。
- 右上角到左下角：$A_{1,4}+A_{2,3}+A_{3,2}+A_{4,1}=34$ 。

特别的，我们会指定把数字 $1$ 先固定在某一格内。

## 输入格式

输入只有一行包含两个正数据 $i$ 和 $j$ ，表示第 $i$ 行和第 $j$ 列的格子放数字 $1$。剩下的十五个格子，请按照前述六项条件用数字 $2$ 到 $16$ 来填满。

## 输出格式

输出所有合法解，并且依序排好。相邻两组合法解之间用一个空行隔开。

对于每一组合法解，输出四行，每行四个数，相邻两数之间用一个空格隔开。

合法解排序的方式，是先从第一行的数字开始比较，每一行数字，由最左边的数字开始比，数字较小的解答必须先输出到文件中。

## 样例

**样例输入 #1**
```
1 1
```

**样例输出 #1**
```
1 4 13 16 
14 15 2 3 
8 5 12 9 
11 10 7 6 

1 4 13 16 
14 15 2 3 
12 9 8 5 
7 6 11 10 

……剩余214组合法解省略```

## AC 代码

```cpp
#include <bits/stdc++.h>
using namespace std;

int a[4][4];                     // 存储结果，0 表示未填
bool used[17];                   // 数字是否已使用

// 每条线已填入的数字个数以及当前和
int line_cnt[16];
int line_sum[16];

// 记录每个格子所属的线的编号列表
vector<int> lines_of_cell[4][4];

bool first_solution = true;      // 控制输出格式

// 输出找到的一组解
void print_solution() {
    if (!first_solution) {
        cout << "\n";             // 解之间用空行隔开
    }
    first_solution = false;
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            cout << a[i][j];
            if (j < 3) cout << " ";
        }
        cout << "\n";
    }
}

// 检查放置在 (r,c) 的数字是否违反任何约束
bool check(int r, int c) {
    for (int lid : lines_of_cell[r][c]) {
        int cnt = line_cnt[lid];
        int sum = line_sum[lid];
        if (cnt == 4) {
            if (sum != 34) return false;
        } else {
            if (sum >= 34) return false;                // 已经超过，后面至少加1
            if (sum + (4 - cnt) * 16 < 34) return false; // 即使全填16也达不到34
        }
    }
    return true;
}

// 在 (r,c) 放置 num 或 移除 num（由 add 控制）
void update(int r, int c, int num, bool add) {
    int delta = add ? 1 : -1;
    int val_delta = add ? num : -num;
    for (int lid : lines_of_cell[r][c]) {
        line_cnt[lid] += delta;
        line_sum[lid] += val_delta;
    }
}

// 深度优先搜索
void dfs(int r, int c) {
    if (r == 4) {
        print_solution();
        return;
    }
    int next_r = r, next_c = c + 1;
    if (next_c == 4) {
        next_r = r + 1;
        next_c = 0;
    }

    if (a[r][c] != 0) {          // 已经是固定的 1，跳过
        dfs(next_r, next_c);
        return;
    }

    for (int num = 2; num <= 16; num++) {
        if (!used[num]) {
            used[num] = true;
            a[r][c] = num;
            update(r, c, num, true);   // 加入

            if (check(r, c)) {
                dfs(next_r, next_c);
            }

            update(r, c, num, false);  // 移除
            a[r][c] = 0;
            used[num] = false;
        }
    }
}

// 初始化所有线的格子归属
void init_lines() {
    // 行 0~3
    for (int r = 0; r < 4; r++)
        for (int c = 0; c < 4; c++)
            lines_of_cell[r][c].push_back(r);
    // 列 4~7
    for (int c = 0; c < 4; c++)
        for (int r = 0; r < 4; r++)
            lines_of_cell[r][c].push_back(4 + c);
    // 主对角线 8
    for (int i = 0; i < 4; i++)
        lines_of_cell[i][i].push_back(8);
    // 副对角线 9
    for (int i = 0; i < 4; i++)
        lines_of_cell[i][3 - i].push_back(9);
    // 左上 2x2 10
    for (int r = 0; r < 2; r++)
        for (int c = 0; c < 2; c++)
            lines_of_cell[r][c].push_back(10);
    // 右上 2x2 11
    for (int r = 0; r < 2; r++)
        for (int c = 2; c < 4; c++)
            lines_of_cell[r][c].push_back(11);
    // 左下 2x2 12
    for (int r = 2; r < 4; r++)
        for (int c = 0; c < 2; c++)
            lines_of_cell[r][c].push_back(12);
    // 右下 2x2 13
    for (int r = 2; r < 4; r++)
        for (int c = 2; c < 4; c++)
            lines_of_cell[r][c].push_back(13);
    // 中心 2x2 14
    for (int r = 1; r <= 2; r++)
        for (int c = 1; c <= 2; c++)
            lines_of_cell[r][c].push_back(14);
    // 四个角 15
    lines_of_cell[0][0].push_back(15);
    lines_of_cell[0][3].push_back(15);
    lines_of_cell[3][0].push_back(15);
    lines_of_cell[3][3].push_back(15);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int r1, c1;
    cin >> r1 >> c1;               // 输入 1 所在的行列（1‑based）
    r1--; c1--;                    // 转为 0‑based

    init_lines();

    // 固定数字 1
    a[r1][c1] = 1;
    used[1] = true;
    update(r1, c1, 1, true);      // 更新相关线

    dfs(0, 0);

    return 0;
}
```

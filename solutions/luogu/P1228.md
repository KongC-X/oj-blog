# P1228 地毯填补问题

> **洛谷** | 普及/提高- | C++14 | 提交时间: 2026-04-29

**算法标签**：`最小环`、`SG函数`、`栈`

## 题目描述

相传在一个古老的阿拉伯国家里，有一座宫殿。宫殿里有个四四方方的格子迷宫，国王选择驸马的方法非常特殊，也非常简单：公主就站在其中一个方格子上，只要谁能用地毯将除公主站立的地方外的所有地方盖上，美丽漂亮聪慧的公主就是他的人了。公主这一个方格不能用地毯盖住，毯子的形状有所规定，只能有四种选择（如图）：

![](https://cdn.luogu.com.cn/upload/image_hosting/cik5fiow.png)

并且每一方格只能用一层地毯，迷宫的大小为 $2^k\times 2^k$ 的方形。当然，也不能让公主无限制的在那儿等，对吧？由于你使用的是计算机，所以实现时间为 $1$ 秒。

## 输入格式

输入文件共 $2$ 行。

第一行一个整数 $k$，即给定被填补迷宫的大小为 $2^k\times 2^k$（$0\lt k\leq 10$）；
第二行两个整数 $x,y$，即给出公主所在方格的坐标（$x$ 为行坐标，$y$ 为列坐标），$x$ 和 $y$ 之间有一个空格隔开。

## 输出格式

将迷宫填补完整的方案：每一补（行）为 $x\ y\ c$（$x,y$ 为毯子拐角的行坐标和列坐标，$c$ 为使用毯子的形状，具体见上面的图 $1$，毯子形状分别用 $1,2,3,4$ 表示，$x,y,c$ 之间用一个空格隔开）。

## 样例

**样例输入 #1**
```
3                          
3 3   ```

**样例输出 #1**
```

## AC 代码

```cpp
#include <iostream>
using namespace std;

// 根据公主所在象限，铺设L形地毯，并返回该块地毯的拐角坐标和形状
// (tx, ty) 是当前区域左上角的坐标，(px, py) 是公主坐标（缺失的格子），size 是当前区域边长
void solve(int tx, int ty, int px, int py, int size) {
    if (size == 1) return;                // 1x1 无需铺设
    int half = size / 2;
    // 判断公主在哪个象限
    // 左上象限
    if (px < tx + half && py < ty + half) {
        // 公主在左上方，在另外三个象限的中心交界处铺地毯：形状为 1（缺左上）
        cout << tx + half << " " << ty + half << " " << 1 << "\n";
        // 递归其他三个象限，缺口在各自靠近中心的角落
        solve(tx, ty, px, py, half);                          // 左上，缺口即公主
        solve(tx, ty + half, tx + half - 1, ty + half, half); // 右上，缺口在左下角
        solve(tx + half, ty, tx + half, ty + half - 1, half); // 左下，缺口在右上角
        solve(tx + half, ty + half, tx + half, ty + half, half);// 右下，缺口在左上角
    }
    // 右上象限
    else if (px < tx + half && py >= ty + half) {
        // 公主在右上，形状为 2（缺右上）
        cout << tx + half << " " << ty + half - 1 << " " << 2 << "\n";
        solve(tx, ty, tx + half - 1, ty + half - 1, half);      // 左上，缺口在右下角
        solve(tx, ty + half, px, py, half);                      // 右上，缺口即公主
        solve(tx + half, ty, tx + half, ty + half - 1, half);   // 左下，缺口在右上角
        solve(tx + half, ty + half, tx + half, ty + half, half);// 右下，缺口在左上角
    }
    // 左下象限
    else if (px >= tx + half && py < ty + half) {
        // 公主在左下，形状为 3（缺左下）
        cout << tx + half - 1 << " " << ty + half << " " << 3 << "\n";
        solve(tx, ty, tx + half - 1, ty + half - 1, half);      // 左上，缺口在右下角
        solve(tx, ty + half, tx + half - 1, ty + half, half);   // 右上，缺口在左下角
        solve(tx + half, ty, px, py, half);                      // 左下，缺口即公主
        solve(tx + half, ty + half, tx + half, ty + half, half);// 右下，缺口在左上角
    }
    // 右下象限
    else {
        // 公主在右下，形状为 4（缺右下）
        cout << tx + half - 1 << " " << ty + half - 1 << " " << 4 << "\n";
        solve(tx, ty, tx + half - 1, ty + half - 1, half);      // 左上，缺口在右下角
        solve(tx, ty + half, tx + half - 1, ty + half, half);   // 右上，缺口在左下角
        solve(tx + half, ty, tx + half, ty + half - 1, half);   // 左下，缺口在右上角
        solve(tx + half, ty + half, px, py, half);               // 右下，缺口即公主
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int k, x, y;
    cin >> k >> x >> y;
    int size = 1 << k;   // 2^k
    // 调用分治，左上角坐标为 (1,1)，公主坐标为 (x, y)
    solve(1, 1, x, y, size);
    return 0;
}
```

# 1140 - 亲密数对

## 题目描述

键盘输入 $N$ ，$N$ 在 $2$ 至 $2000$ 之间，求 $2$ 至 $N$ 中的亲密数对，就是 $A$ 的因子和等于 $B$ ，$B$ 的因子和等于 $A$ ，且 $A \neq B$ 。 

如 $48$ 和 $75$ 是亲密数对。$48$ 的因子和为 $2+3+4+6+8+12+16+24=75$ ，而 $75$ 的因子和为 $3+5+15+25=48$ 。

## 输入

只有一行，为一个整数 $N$ （ $2 \le N \le 2000$ ）

## 输出

输出若干行，每行两个整数（用一个空格隔开）。

## AC代码

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

int a[1010],b[1010];
int n,m,k,s,i,j;
int main() {
    cin >> m >> n;
    // 输入a 和 b 数组
    for(i = 1;i <= n;i++) {
        cin >> a[i] >> b[i];
    }
    // 联动排序，按照a数组从小到大排序
    for(i = 0;i <= n - 1;i++) {
        for(j = 1;j <= n - i - 1;j++) {
            if(a[j] > a[j + 1]) {
                swap(a[j],a[j + 1]);
                swap(b[j],b[j + 1]);
            }
        }
    }
    // 打印输出验证
    // for(i = 1;i <= n;i++) {
    //     cout << a[i] << " " << b[i] << endl;
    // }
    for(i = 1;i <= n;i++) {
        k += b[i];
        s += a[i] * b[i];
        if(k >= m) {
            s = s - (k - m) * a[i];
            break;
        }
    }
    cout << s << endl;
}



#include <iostream>
#include <bits/stdc++.h>
using namespace std;

struct Shop{
    int price; // 单价
    int num; // 数量
}a[1100];

bool cmp(Shop a,Shop b) {
    if(a.price > b.price) {
        return false;
    } else {
        return true;
    }
}

int main() {
    int i,m,n,k = 0,s = 0;
    cin >> m >> n;
    for(i = 0;i < n;i++) {
        cin >> a[i].price >> a[i].num;
    }
    // 按照单价从低到高排序
    sort(a,a+n,cmp);
    // 输出验证
    // for(i = 0;i < n;i++) {
    //     cout << a[i].price << " " << a[i].num << endl;
    // }
    for(i = 0;i < n;i++) {
        k += a[i].num;
        s += a[i].price * a[i].num;
        if(k >= m) {
            s = s - a[i].price * (k - m);
            break;
        }
    }
    cout << s << endl;
}

# 1186 - 数字走向III

## 题目描述

输入整数 $N$ ，输出相应方阵。

## 输入

一个整数 $N$ 。（ $0 \lt N \le 10$ )

## 输出

一个方阵，每个数字的场宽为 $3$ 。

## 解题思路

## AC代码

```cpp
#include<iostream>
#include<bits/stdc++.h>
using namespace std;

// 埃氏筛法
bool f[1000010]; // 标记素数，false表示素数，true表示合数
int k,n;
int main() {
	cin >> n;
	// 特判特殊的数
	f[0] = true;
	f[1] = true;
	k = n - 1; // 初始素数个数为n-1，因为1不是素数

	// 循环所有的数，逐个判断是否为素数
	for(int i = 2; i * i <= n; i++) {
		// 如果当前数是素数，则将其所有的倍数标记为合数
	    if(!f[i]) {
			// 将i的倍数标记为合数
			for(int j = i + i; j <= n;j += i) {
				// 如果没有标记过
				if(!f[j]) {
				    f[j] = true;
				    k--; // 素数个数减一
				}
			}
	    }
	}
	cout << k << endl;
}


// 线性筛
bool f[1000010]; // 标记素数，false表示素数，true表示合数
int prime[1000010]; // 存储素数
int k,n;
// 线性筛：每个合数只能被自己的最小质因子筛掉
// 比如：6 = 2 * 3，6只能被2筛掉，不能被3筛掉
int main() {
	cin >> n;
	// 特判特殊的数
	f[0] = true;
	f[1] = true;

	// 循环所有的数，逐个判断是否为素数
	for(int i = 2; i <= n; i++) {
		// 如果当前数是素数，则将其所有的倍数标记为合数
	    if(!f[i]) {
			k++;
			// 将素数存入数组中
			prime[k] = i;
		}
		// 每个合数只能被自己的最小质因子筛掉
		// j循环的是素数，prime[j]表示第j个素数
		for(int j = 1;i * prime[j] <= n;j++) {
			f[i * prime[j]] = true; // 将这个数筛掉
			// 如果i能被prime[j]整除
			// i * prime[j+1] % prime[j] == 0
			if(i % prime[j] == 0) {
				break; // 直接跳出循环
			}
		}
	}
	cout << k << endl;
}
```

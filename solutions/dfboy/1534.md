# 1534 - 小 X 与机器人 2 (sihuo)

## 题目描述

小 X 在解决了高大上的落子问题后， 开始着手训练 BetaGo 的深度学习 能力， 第一步要教 BetaGo识别各种各样的棋形，小 X 首先教 BetaGo 识别跟死活密切相关的棋型，在成功识别了一些简单的棋形后， BetaGo 在识别聚四棋形时遇到了困难，所谓聚四棋形就是连在一起的四颗棋子， 它们共有五种不同的形状， 五种棋形及其命名如下图所示：

注意： 每种形状经过旋转、翻折得到的图案属于同一种形状。 譬如只要四颗相邻的棋子排成一条直线则不管它们是横着排还是竖着排都称之为直四。

小 X 希望你编一个程序帮助 BetaGo 识别聚四棋形， 具体要求是：给出四个连在一起的围棋盘上的交叉点位置， 判断这四颗连在一起的棋子属于那种聚四？ 输入数据保证给出的棋形一定是五种聚四棋形中的一种。 棋盘上位置的表示方法与上一题相同。

## 输入

输入数据共有四行，每行包含两个用空格隔开的正整数 $x$ 和 $y$， 表示棋盘上的一颗棋子的位置。

## 输出

如果是直四 ， 则输出“ `zhisi`” ； 如果是曲四， 则输出“ `qusi`” ； 如果是弯四， 则输出“ `wansi`” ；

如果是丁四， 则输出 “`dingsi`” ； 如果是方四， 则输出 “`fangsi`” 。 注意引号不要输出。

#include <iostream>
using namespace std;

// 判断两个点是否相邻（上下左右）
bool fun(int x1, int y1, int x2, int y2) {
    if(x1 == x2 && (y1 == y2 + 1 || y1 == y2 - 1)) return true;
    if(y1 == y2 && (x1 == x2 + 1 || x1 == x2 - 1)) return true;
    return false;
}

int main() {
    int x[14], y[14];
    for (int i = 0; i < 4; ++i) {
        cin >> x[i] >> y[i];
    }

    // 计算每个点的度数（相邻棋子的个数）
    int deg[4] = {0};
    for (int i = 0; i < 4; ++i) {
        for (int j = 0; j < 4; ++j) {
            if (i != j && fun(x[i], y[i], x[j], y[j])) {
                deg[i]++;
            }
        }
    }

    // 找出最大度数
    int maxdeg = 0;
    for (int i = 0; i < 4; ++i) {
        if (deg[i] > maxdeg) maxdeg = deg[i];
    }

    // 根据度数分类
    if (maxdeg == 3) {
        cout << "dingsi" << endl;
        return 0;
    }
    if (maxdeg == 2) {
        // 判断是否所有点度数都是2（方四）
        bool all_two = true;
        for (int i = 0; i < 4; ++i) {
            if (deg[i] != 2) all_two = false;
        }
        if (all_two) {
            cout << "fangsi" << endl;
            return 0;
        }
    }

    // 剩下的是路径形（度数为1,2,2,1），需要判断直四、曲四、弯四
    // 找到两个端点（度数为1的点）
    int start = -1, end = -1;
    for (int i = 0; i < 4; ++i) {
        if (deg[i] == 1) {
            if (start == -1) start = i;
            else end = i;
        }
    }

    // 从 start 走到 end，记录路径
    int path[4];      // 路径上的点索引
    int dir[4];       // 每一步的方向编码（0=右,1=左,2=上,3=下，实际可简化）
    path[0] = start;
    int cur = start;
    int prev = -1;
    int idx = 1;      // 路径下一个位置
    while (cur != end) {
        // 找 cur 的相邻点中不是 prev 的那个
        int nxt = -1;
        for (int i = 0; i < 4; ++i) {
            if (i != prev && fun(x[cur], y[cur], x[i], y[i])) {
                nxt = i;
                break;
            }
        }
        // 记录方向（相对于 cur 的增量）
        int dx = x[nxt] - x[cur];
        int dy = y[nxt] - y[cur];
        // 方向编码：dx,dy 只有 (0,1),(0,-1),(1,0),(-1,0)
        int dir_code;
        if (dx == 1 && dy == 0) dir_code = 0;      // 下
        else if (dx == -1 && dy == 0) dir_code = 1; // 上
        else if (dx == 0 && dy == 1) dir_code = 2;  // 右
        else dir_code = 3;                          // 左

        dir[idx-1] = dir_code;   // 记录从 path[idx-1] 到 path[idx] 的方向
        path[idx] = nxt;
        prev = cur;
        cur = nxt;
        ++idx;
    }

    // 计算方向变化次数
    int changes = 0;
    for (int i = 1; i < idx - 1; ++i) { // 比较相邻两个方向
        if (dir[i] != dir[i-1]) ++changes;
    }

    if (changes == 0) cout << "zhisi" << endl;
    else if (changes == 1) cout << "qusi" << endl;
    else cout << "wansi" << endl;

    return 0;
}

# 2279 - 数字变换

## 题目描述

给定一个包含 $5$ 个数字（$0 \sim 9$）的字符串，例如 $02943$ ，请将 $12345$ 变换到它。 你可以采取 $3$ 种操作进行变换 

- 交换相邻的两个数字。

- 将一个数字加 $1$ 。如果加 $1$ 后大于 $9$ ，则变为 $0$ 。 

- 将一个数字加倍。如果加倍后大于 $9$ ,则将其变为加倍后的结果除以 $10$ 的余数。 

最多只能用第 $2$ 种操作 $3$ 次，第 $3$ 种操作 $2$ 次 求最少经过多少次操作可以完成变换。

## 输入

有最多 $100000$ 组数据，每组数据就是包含 $5$ 个数字的字符串。

## 输出

对每组数据，输出将 $12345$ 变换到给定字符串所需要的最少操作步数。如果无法变换成功，输出 $-1$。
## AC代码

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXS = 100000;          // 状态总数 00000~99999
const int MAXA = 4;               // 操作2最多使用3次，数组维0..3
const int MAXB = 3;               // 操作3最多使用2次，数组维0..2
int dist[MAXS][MAXA][MAXB];       // -1 表示不可达

// 五个位置的权值：万位到个位
const int p[5] = {10000, 1000, 100, 10, 1};

// BFS队列（可用三个数组或结构体）
struct Node {
    int s;     // 数字状态 0~99999
    int a;     // 已使用操作2的次数
    int b;     // 已使用操作3的次数
};
Node q[MAXS * MAXA * MAXB];       // 队列大小足够
int head, tail;

// 辅助函数：获取数 s 的第 i 位数字（i=0..4 万位到个位）
inline int get_digit(int s, int i) {
    return (s / p[i]) % 10;
}

// 将新状态尝试加入队列
void try_push(int s, int a, int b, int d) {
    if (dist[s][a][b] == -1) {
        dist[s][a][b] = d;
        q[tail++] = {s, a, b};
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    // 初始化 dist 为 -1
    memset(dist, -1, sizeof(dist));

    // 初始状态：12345，未使用操作2和3
    int start = 12345;
    dist[start][0][0] = 0;
    head = tail = 0;
    q[tail++] = {start, 0, 0};

    // BFS
    while (head < tail) {
        Node cur = q[head++];
        int s = cur.s, a = cur.a, b = cur.b;
        int d = dist[s][a][b];

        // 操作1：交换相邻两位（i 和 i+1，i=0..3）
        for (int i = 0; i < 4; i++) {
            int d1 = get_digit(s, i);
            int d2 = get_digit(s, i + 1);
            // 计算新状态：交换两个数字
            int s2 = s - d1 * p[i] - d2 * p[i + 1] + d1 * p[i + 1] + d2 * p[i];
            try_push(s2, a, b, d + 1);
        }

        // 操作2：某一位数字加1模10，最多还能用3-a次
        if (a < 3) {
            for (int i = 0; i < 5; i++) {
                int cur_d = get_digit(s, i);
                int nxt_d = (cur_d + 1) % 10;
                int s2 = s - cur_d * p[i] + nxt_d * p[i];
                try_push(s2, a + 1, b, d + 1);
            }
        }

        // 操作3：某一位数字加倍模10，最多还能用2-b次
        if (b < 2) {
            for (int i = 0; i < 5; i++) {
                int cur_d = get_digit(s, i);
                int nxt_d = (cur_d * 2) % 10;
                int s2 = s - cur_d * p[i] + nxt_d * p[i];
                try_push(s2, a, b + 1, d + 1);
            }
        }
    }

    // 处理查询
    string target;
    while (cin >> target) {
        // 将目标字符串转为整数（自动忽略前导零，与状态表示兼容）
        int t = stoi(target);
        int ans = -1;
        for (int a = 0; a <= 3; a++) {
            for (int b = 0; b <= 2; b++) {
                if (dist[t][a][b] != -1) {
                    if (ans == -1 || dist[t][a][b] < ans) {
                        ans = dist[t][a][b];
                    }
                }
            }
        }
        cout << ans << '\n';
    }

    return 0;
}
```

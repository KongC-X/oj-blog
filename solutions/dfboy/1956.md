# 1956 - 算24点（2）

## 题目描述

给出 $n$ 个整数，请问这 $n$ 个数字在不改变顺序且不加入括号的情况下，仅通过 $+- \times $ 三种运算符，有多少种运算能得到 $24$ 。

比如：$4$ $10$ $2$ $4$ $8$，有如下 $3$ 种运算能够得到$24$ 。

```
`4+10-2+4+8=24
4-10-2+4*8=24
4*10-2*4-8=24
`
```

## 输入

第 $1$ 行有一个整数 $n$ ；（$2 \le n \le 10$）

第 $2$ 行有 $n$ 个整数 $a_i$ ；（$1 \le a_i \le 20$）

## 输出

一个整数，代表能计算出 $24$ 点的方案数；
## AC代码

```cpp
#include <bits/stdc++.h>
#include <iostream>
using namespace std;

/*
  24点计数（n个数版）
  
  n个数，n-1个运算符位置，每个位置3种选择（+ - *）
  总方案数 = 3^(n-1)，n≤10 时最多 19683 种，全枚举
  
  计算时：× 优先于 + -，同优先级从左到右
*/

int n;           // 数字个数
int a[15];       // 存放 n 个数字
char op[15];     // 存放 n-1 个运算符
int ans = 0;     // 能算出 24 的方案数

// 检查当前 op 组合能否算出 24
bool check() {
    vector<int> vals;
    vector<char> ops;
    
    vals.push_back(a[1]);  // 先放第一个数
    
    // 第一遍：处理所有 ×
    for(int i = 1; i < n; i++) {
        if(op[i] == '*') {
            vals.back() *= a[i + 1];  // 乘优先，直接算
        } else {
            vals.push_back(a[i + 1]);
            ops.push_back(op[i]);
        }
    }
    
    // 第二遍：剩下的 + - 从左到右
    int result = vals[0];
    for(int i = 0; i < (int)ops.size(); i++) {
        if(ops[i] == '+') {
            result += vals[i + 1];
        } else {
            result -= vals[i + 1];
        }
    }
    
    return result == 24;
}

// DFS：枚举第 pos 个运算符（从 1 到 n-1）
void dfs(int pos) {
    if(pos == n) {
        // 所有运算符都填好了，检查结果
        if(check()) ans++;
        return;
    }
    
    // 每个位置可以填 + - *
    op[pos] = '+';
    dfs(pos + 1);
    
    op[pos] = '-';
    dfs(pos + 1);
    
    op[pos] = '*';
    dfs(pos + 1);
}

int main() {
    cin >> n;
    for(int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    
    ans = 0;       // 方案数清零
    dfs(1);        // 从第1个运算符开始枚举
    
    cout << ans << endl;
    
    return 0;
}
```

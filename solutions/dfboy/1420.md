# 1420 - 已知正方形的周长求面积

## 题目描述

已知一个正方形的周长为整数 $n$ ，请问该正方形的面积是多少？

## 输入

一个整数 $n$ 代表正方形的周长。

## 输出

一个小数，代表正方形的面积（保留 $2$ 位小数）。

## 解题思路

## AC代码

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,x;
struct Num
{
    int num,loc;//num是当前数字,loc是该数字的位置
} a[100010];
bool cmp(Num a,Num b)//sort判断函数
{
    if(a.num<b.num)return true;
    else return false;
}
int find(int x)//二分找数函数
{
    int l=1,r=n,mid;
    while(l<=r)
    {
        mid=l+r>>1;
        if(a[mid].num<x)//通过分治来缩小二分左右边界
        {
            l=mid+1;
        }
        else if(a[mid].num>x)
        {
            r=mid-1;
        }
        else
        {
            return a[mid].loc;//找到了直接返回它的位置
        }
    }
    return 0;//否则返回0
}
int main()
{
    scanf("%d",&n);//scanf运行效率较高,cin可能会超时
    for(int i = 1; i <= n; i++)
    {
        scanf("%d",&a[i].num);
        a[i].loc=i;
    }
    sort(a+1,a+n+1,cmp);//自定义排序
    scanf("%d",&m);
    for(int i = 1; i <= m; i++)
    {
        scanf("%d",&x);
        printf("%d\n",find(x));
    }
    return 0;
}
```

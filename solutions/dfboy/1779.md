# 1779 - 装箱问题

## 题目描述

有一个箱子容量为 $V$（正整数，$0 \le V \le 200000$ ），同时有 $n$ 个物品（ $0 \lt n \le 30$ )，每个物品有一个体积（正整数）。

要求 $n$ 个物品中，任取若干个装入箱内，使箱子的剩余空间为最小。

## 输入

$1$ 个整数，表示箱子容量；

$1$ 个整数，表示有 $n$ 个物品；

接下来 $n$ 行，分别表示这 $n$ 个物品的各自体积；

## 输出

$1$ 个整数，表示箱子剩余空间。

// T3大模拟：非常考察思维的缜密性及基本功。
#include <cmath>
#include <cstdio>
#include <iostream>
using namespace std;
int t, m, a, b, c;

int gcd(int p, int q) {
    int u;
    while (q != 0) {
        u = p % q;
        p = q;
        q = u;
    }
    return p;
}

void fun1(int p, int q) {
    if (p * q < 0)
        cout << '-';
    if (p < 0)
        p = 0 - p;
    if (q < 0)
        q = 0 - q;
    int g = gcd(p, q);
    if (g == q)
        cout << p / g;
    else
        cout << p / g << '/' << q / g;
    return;
}
void fun2(int p, int q) {
    if (p < 0)
        p = -p;
    if (q < 0)
        q = -q;
    int u = 1;
    for (int i = sqrt(p); i >= 2; i--) {
        if (p % (i * i) == 0) {
            p = p / (i * i);
            u = u * i;
            break;
        }
    }
    int g = gcd(u, q);
    u = u / g;
    q = q / g;
    if (u != 1)
        cout << u << "*";
    if (p != 1)
        cout << "sqrt(" << p << ")";
    if (q != 1)
        cout << "/" << q;
}
int main() {
    cin >> t >> m;
    while (t--) {
        cin >> a >> b >> c;
        int d = b * b - 4 * a * c;
        if (d < 0) {
            cout << "NO";
        } else {
            int q = sqrt(d);
            if (q * q == d) {
                double ans1 = 1.0 * (-b + q) / (2 * a);
                double ans2 = 1.0 * (-b - q) / (2 * a);
                if (ans1 > ans2) {
                    if (-b + q == 0)
                        cout << 0;
                    else
                        fun1(-b + q, 2 * a);
                } else {
                    if (-b - q == 0)
                        cout << 0;
                    else
                        fun1(-b - q, 2 * a);
                }
            } else {
                if (b != 0)
                    fun1(-b, 2 * a);
                if (b != 0)
                    cout << "+";
                fun2(d, 2 * a);
            }
        }
        cout << endl;
    }
    return 0;
}

# 1135 - 歌德巴赫猜想

## 题目描述

任一个大于等于 $4$ 的偶数都可以拆分为两个素数之和。

从键盘读入一个整数 $n$，请将小于等于 $n$ 的偶数拆分为 $2$ 个质数之和，列出所有方案。

## 输入

一个整数 $n$ ($4 \le n \le 200$）；

## 输出

将小于等于 $n$ 的偶数拆分为 $2$ 个质数之和，列出所有方案，输出格式请参考样例输出。

## 解题思路

## AC代码

```cpp
// 贪心策略：优先选择结束时间尽可能早的活动，因为结束时间越早，剩余的时间就越多，那么剩余的时间可以排更多的活动
// 解法：按照结束时间升序排序，然后逐个统计哪个活动的开始时间 >= 上一个被选中活动的结束时间  

#include <bits/stdc++.h>
#include <iostream>
using namespace std;

int n; //活动数量
int s[110],e[110]; //活动起止时间
int i,j,c,etime; 
int main() {
	cin>>n;
	for(i = 0;i < n;i++){
		cin>>s[i]>>e[i];
	}
	//按照结束时间升序
	for(i = 1;i <= n - 1;i++){
		for(j = 0;j <= n - i - 1;j++){
			if(e[j] > e[j+1]){
				swap(e[j],e[j+1]);
				swap(s[j],s[j+1]);
			}
		}
	} 
	
	//第1个活动必选
	etime = e[0];//存储第1个活动的结束时间
	c = 1;//已经有1个选中的活动了

	//在剩余活动中继续选 
	for(i = 1;i < n;i++){
		//如果有一个活动的开始时间>=上一个选中活动的结束时间 
		if(s[i] >= etime){
			c++;
			etime = e[i];
		}
	} 
	cout<<c;
	return 0;
}



#include <bits/stdc++.h>
#include <iostream>
#include <iomanip>
using namespace std;

struct Node {
    int begin;
    int end;
}a[110];

bool cmp(Node a, Node b) {
    if(a.end < b.end) {
        return true;
    } else {
        return false;
    }
}

int main() {
    int i,j,n,c = 0,e;
    cin >> n;
    for (i = 0; i < n; i++) {
        cin >> a[i].begin >> a[i].end;
    }
    sort(a,a + n,cmp);
    c = 1; // 第一个活动必须选
    e = a[0].end;
    for(i = 1;i < n;i++) {
        if(a[i].begin >= e) {
            c++;
            e = a[i].end;
        }
    }
    cout << c << endl;
}
```

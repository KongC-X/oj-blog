# 1924 - 集合

## 题目描述

现在给你一些连续的整数，它们是从A到B的整数。一开始每个整数都属于各自的集合，然后你需要进行以下的操作：

每次选择两个属于不同集合的整数，如果这两个整数拥有大于等于P的公共质因数，那么把它们所在的集合合并。

反复如上操作，直到没有可以合并的集合为止。

现在请编程计算，最后有多少个集合。

## 输入

一行，三个整数A,B,P。

A≤B≤100000；

2≤P≤B。

## 输出

一个数，表示最终集合的个数。
## AC代码

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXB = 100010;             // B 最大 100000，多开10个
int fa[MAXB];                        // 并查集数组

// 查找根节点，路径压缩
int find(int x) {
    return x == fa[x] ? x : fa[x] = find(fa[x]);
}

// 合并两个集合
void merge(int x, int y) {
    int fx = find(x);
    int fy = find(y);
    if (fx != fy) {
        fa[fx] = fy;
    }
}

int main() {
    int A, B, P;                     // A,B:整数区间，P:公共质因数下限
    cin >> A >> B >> P;

    // 初始化并查集，编号从 1 到 B（包括不在区间内的也没关系）
    for (int i = 1; i <= B; i++) {
        fa[i] = i;
    }

    // 线性筛找出 2~B 的所有质数
    vector<int> primes;              // 存放质数
    bool isComp[MAXB] = {false};     // 标记合数

    for (int i = 2; i <= B; i++) {
        if (!isComp[i]) {
            primes.push_back(i);     // i 是质数
        }
        for (int j = 0; j < primes.size() && (long long)i * primes[j] <= B; j++) {
            isComp[i * primes[j]] = true;
            if (i % primes[j] == 0) break;
        }
    }

    // 遍历每个质数，如果 >= P，则将它的倍数合并
    for (int i = 0; i < primes.size(); i++) {
        int p = primes[i];           // p: 当前质数
        if (p < P) continue;         // 小于 P 的质数不参与合并

        // 找到区间 [A, B] 内第一个 p 的倍数
        int start = (A + p - 1) / p * p;
        if (start > B) continue;     // 没有倍数在区间内

        // 将 start 与其它的 p 的倍数合并
        for (int j = start + p; j <= B; j += p) {
            merge(start, j);
        }
    }

    // 统计 [A, B] 中不同集合的数量
    int ans = 0;
    for (int i = A; i <= B; i++) {
        if (find(i) == i) {          // i 是所在集合的根
            ans++;
        }
    }

    cout << ans << endl;
    return 0;
}
```

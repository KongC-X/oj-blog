# 1161 - 元素插入有序数组

## 题目描述

给你一个整数 $n$ 和一个数列(数列个数不超过 $1000$ )，这个数列保证从小到大排列，现要求将这个整数 $n$ 插入到数列中，使新的数列仍然从小到大排列。

## 输入

第一行一个整数 $n$ 表示等待插入的数 ；

第二行一个整数 $m$ 表示数列中数的个数；

第三行 $m$ 个整数（空格隔开）。

## 输出

一行整数：新的数列（空格隔开）。

// 这里先不考虑高精度数的情况，只处理递推的逻辑
#include <bits/stdc++.h>
#include <iostream>
using namespace std;

int main() {
    int n,i,a,b,c;
    cin >> n;
    a = 1;
    b = 2;
    for(i = 3;i <= n;i++) {
        c = b * 2 + a;
        a = b;
        b = c;
    }
    cout << c << endl;
}



// 高精度处理
#include<iostream>
using namespace std;

// An = A(n - 1) * 2 + A(n - 2)

// 求两个高精度的整数之和
string he(string s1,string s2) {
    string r; // 存放总和
    int a[1000] = {0}, b[1000] = {0}, c[1000] = {0};
    int i;
    for(i = 0; i < s1.size(); i++) {
        a[i] = s1[s1.size() - i - 1] - '0';
    }
    for(i = 0; i < s2.size(); i++) {
        b[i] = s2[s2.size() - i - 1] - '0';
    }
    int len = max(s1.size(), s2.size());
    for(i = 0; i < len; i++) {
        c[i] += a[i] + b[i];
        if(c[i] >= 10) {
            c[i + 1] += c[i] / 10;
            c[i] %= 10;
        }
    }
    if(c[len] != 0) {
        len++;
    }
    for(i = len - 1; i >= 0; i--) {
        r += char(c[i] + '0');
    }
    return r;
}

// 求一个高精度的整数 * 2 的积
string cheng(string s) {
    string r;
    int a[1000] = {0};
    int i;
    for(i = 0; i < s.size(); i++) {
        a[i] = s[s.size() - i - 1] - '0';
    }
    for(i = 0; i < s.size(); i++) {
        a[i] = a[i] * 2;
    }
    for(i = 0; i < s.size(); i++) {
        if(a[i] >= 10) {
            a[i + 1] += a[i] / 10;
            a[i] %= 10;
        }
    }
    int len = s.size();
    if(a[len] != 0) {
        len++;
    }
    for(i = len - 1; i >= 0; i--) {
        r += char(a[i] + '0');
    }
    return r;
}

    int i,n;
    // z代表计算结果，就是当前项，x代表前前一项，y代表前一项
    string x,y,z;

    if(n == 1) {
        cout << x;
    } else if(n == 2) {
        cout << y;
    } else {
        // 从第三项开始递推
        for(i = 3;i <= n;i++) {
            z = he(cheng(y),x);
            x = y;
            y = z;
        }
        cout << z;
    }
}

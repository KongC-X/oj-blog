# 6055 - 大数比较大小

## 题目描述

输入两个很大的十进制正实数（长度超过 $20$ 位），输出代表数值较大的数。
说明：

- 1）两个数都大于 $10000000000000000000$；

- 2）实数包含有小数点的数，一个数里只可能有一个小数点，且不在数的最前或最后。

## 输入

输入两行，每行一个数字。

## 输出

数值较大的数。
如果两数相等，则输出“`equation`”。
如果输入的数不符合数字规则，则输出“`error`”。
## AC代码

```cpp
#include <iostream>
#include <string>
#include <cctype>
#include <algorithm>
using namespace std;

// 检查字符串 s 是否为合法的正实数（或正整数）
// 合法：非空，仅含数字和小数点；若有点则恰有一个，且不在首尾；若无点也可。
bool check(const string &s) {
    if (s.empty()) return false;
    int dot = 0;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] == '.') {
            dot++;
            if (dot > 1) return false;        // 多个点
            if (i == 0 || i == s.size() - 1)  // 点在首或尾非法
                return false;
        } else if (!isdigit(s[i])) {
            return false;
        }
    }
    return true; // 0或1个点都合法
}

// 比较两个合法实数 a 和 b
// 返回 1:a>b  0:相等  -1:a<b
int cmp(const string &a, const string &b) {
    // 分离整数部分和小数部分
    int da = a.find('.'), db = b.find('.');
    string intA = (da == string::npos) ? a : a.substr(0, da);
    string intB = (db == string::npos) ? b : b.substr(0, db);
    string fracA = (da == string::npos) ? "" : a.substr(da + 1);
    string fracB = (db == string::npos) ? "" : b.substr(db + 1);
    
    // 去除整数部分前导零（保留至少一位）
    auto strip = [](string &s) {
        int pos = 0;
        while (pos + 1 < s.size() && s[pos] == '0') pos++;
        s = s.substr(pos);
    };
    strip(intA);
    strip(intB);
    
    // 比较整数部分
    if (intA.size() != intB.size())
        return intA.size() > intB.size() ? 1 : -1;
    if (intA != intB)
        return intA > intB ? 1 : -1;
    
    // 整数部分相等，比较小数部分：短补零后字典序比较
    int len = max(fracA.size(), fracB.size());
    while (fracA.size() < len) fracA += '0';
    while (fracB.size() < len) fracB += '0';
    if (fracA == fracB) return 0;
    return fracA > fracB ? 1 : -1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    
    string a, b;
    cin >> a >> b;
    
    if (!check(a) || !check(b)) {
        cout << "error\n";
        return 0;
    }
    
    int res = cmp(a, b);
    if (res == 0)
        cout << "equation\n";
    else if (res > 0)
        cout << a << '\n';
    else
        cout << b << '\n';
    
    return 0;
}
```

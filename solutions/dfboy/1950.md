# 1950 - 蓝桥杯赛迷宫

## 题目描述

把一个 $n$ 行 $m$ 列的字符阵列看做一个迷宫，迷宫仅包含 `L`、`Q`、`B`、`S` 中的大写字母（蓝桥杯赛的汉语拼音首字母）。

初始时，你可以从任意一个 `L` 字母开始，移向相邻的 `Q` 字母，然后从此 `Q` 字母出发，移向相邻的 `B` 字母，然后从此 `B` 字母出发，移向相邻的 `S` 字母……。这样，你就算是走过了一个 `LQBS` 字符序列。

接下来，仍然可以从此 `S` 字母出发，移向相邻的 `L` 字母……，重复上述的动作，你就可以不断地走过 `LQBS` 序列。

请注意，所谓相邻仅包含上、下、左、右 $4$ 个方向，且只能从 `L->Q`，从 `Q->B`，从 `B->S`，从 `S->L`。

可以想像，由于选择的出发点不同，我们有可能在迷宫中走过无数次的 `LQBS` ，或者是有限次的 `LQBS` ，或者一次也走不了。

编程实现：请你编写程序，求出在给定的迷宫中，我们最多可以走过多少次 `LQBS`?

## 输入

第一行：正整数 $n,m$，表示迷宫的规模为 $n$ 行 $m$ 列，。

接下来的 $n$ 行：每行 $m$ 个符合题意的字母，字母间无空格。

## 输出

一个整数。

即：如果在迷宫中可以无限次的走过 `LQBS`，输出 $-1$，否则，输出可以走过 `LQBS` 的最多次数。
## AC代码

```cpp
#include <bits/stdc++.h>
#include <iostream>
using namespace std;

/*
  LQBS 迷宫 - 正确做法
  
  关键：只能从 L 开始走！
  如果图中存在从某个 L 可达的环，输出 -1（能无限走）
  否则，答案是 max(dp[L]) / 4 over all L
  
  用 DFS + 三色标记检测环
  vis[u] = 0(白), 1(灰,在递归栈中), 2(黑,已算完)
  dp[u] = 从u出发最多能经过的节点数（包括u）
*/

const int MAXN = 110;  // n,m < 100
char g[MAXN][MAXN];
int dp[MAXN*MAXN];
int vis[MAXN*MAXN];
int n, m;
int dx[4] = {0,0,1,-1};
int dy[4] = {1,-1,0,0};

// 获取当前字符的下一个应该走的字符
char getNext(char c) {
    if(c == 'L') return 'Q';
    if(c == 'Q') return 'B';
    if(c == 'B') return 'S';
    if(c == 'S') return 'L';
    return '?';  // 不会执行到这里
}

int getId(int x, int y) {
    return x * m + y;
}

// DFS返回值：
//   >= 0: 从u出发最多能经过的节点数
//   -1: 检测到环（从当前路径可达）
int dfs(int u) {
    if(vis[u] == 1) {
        // 遇到灰色节点 = 发现环！
        return -1;
    }
    if(vis[u] == 2) {
        // 已计算过
        return dp[u];
    }
    
    vis[u] = 1;  // 标灰
    dp[u] = 1;  // 至少包含自己
    
    int x = u / m, y = u % m;
    char nextChar = getNext(g[x][y]);
    
    for(int d = 0; d < 4; d++) {
        int nx = x + dx[d], ny = y + dy[d];
        if(nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
        if(g[nx][ny] != nextChar) continue;
        
        int v = getId(nx, ny);
        int res = dfs(v);
        if(res == -1) {
            // 从u出发能走到环，向上传播
            vis[u] = 2;
            dp[u] = -1;
            return -1;
        }
        if(1 + res > dp[u]) {
            dp[u] = 1 + res;
        }
    }
    
    vis[u] = 2;  // 标黑
    return dp[u];
}

int main() {
    cin >> n >> m;
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            cin >> g[i][j];
        }
    }
    
    // 初始化
    for(int i = 0; i < n*m; i++) {
        vis[i] = 0;
        dp[i] = 0;
    }
    
    bool hasCycle = false;
    int ans = 0;
    
    // 只对 L 调用 DFS（题目说可以从任意 L 开始）
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            if(g[i][j] != 'L') continue;  // 只从 L 开始
            int id = getId(i, j);
            if(vis[id] == 0) {
                int res = dfs(id);
                if(res == -1) {
                    hasCycle = true;
                    break;
                }
                // 更新答案：从当前 L 出发能完成的最多 LQBS 序列数
                if(res / 4 > ans) {
                    ans = res / 4;
                }
            } else if(vis[id] == 2) {
                // 已计算过，直接取结果
                if(dp[id] / 4 > ans) {
                    ans = dp[id] / 4;
                }
            }
        }
        if(hasCycle) break;
    }
    
    if(hasCycle) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    
    return 0;
}
```

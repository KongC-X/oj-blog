# 1369 - Pell数列

## 题目描述

有一种数列，它的前 $10$ 项的值分别为：$1$ $2$ $5$ $12$ $29$ $70$ $169$ $408$ $985$ $2378$，这个数列被称为 Pell 数列，请问该数列的第 $n$ 项的值是多少？（$n \le 1000$）

## 输入

一个整数 $n$ 。

## 输出

第 $n$ 项的值。

#include<iostream>
using namespace std;

// An = A(n - 1) * 2 + A(n - 2)

// 求两个高精度的整数之和
string he(string s1,string s2) {
    string r; // 存放总和
    int a[1000] = {0}, b[1000] = {0}, c[1000] = {0};
    int i;
    for(i = 0; i < s1.size(); i++) {
        a[i] = s1[s1.size() - i - 1] - '0';
    }
    for(i = 0; i < s2.size(); i++) {
        b[i] = s2[s2.size() - i - 1] - '0';
    }
    int len = max(s1.size(), s2.size());
    for(i = 0; i < len; i++) {
        c[i] += a[i] + b[i];
        if(c[i] >= 10) {
            c[i + 1] += c[i] / 10;
            c[i] %= 10;
        }
    }
    if(c[len] != 0) {
        len++;
    }
    for(i = len - 1; i >= 0; i--) {
        r += char(c[i] + '0');
    }
    return r;
}

// 求一个高精度的整数 * 2 的积
string cheng(string s) {
    string r;
    int a[1000] = {0};
    int i;
    for(i = 0; i < s.size(); i++) {
        a[i] = s[s.size() - i - 1] - '0';
    }
    for(i = 0; i < s.size(); i++) {
        a[i] = a[i] * 2;
    }
    for(i = 0; i < s.size(); i++) {
        if(a[i] >= 10) {
            a[i + 1] += a[i] / 10;
            a[i] %= 10;
        }
    }
    int len = s.size();
    if(a[len] != 0) {
        len++;
    }
    for(i = len - 1; i >= 0; i--) {
        r += char(a[i] + '0');
    }
    return r;
}

int main() {
    // cout << he("9999","1");
    // cout << cheng("512");
    int i,n;
    // z代表计算结果，就是当前项，x代表前前一项，y代表前一项
    string x,y,z; 
    cin >> n;
    x = "1";
    y = "2";
    if(n == 1) {
        cout << x;
    } else if(n == 2) {
        cout << y;
    } else {
        // 从第三项开始递推
        for(i = 3;i <= n;i++) {
            z = he(cheng(y),x);
            x = y;
            y = z;
        }
        cout << z;
    }
}

# 2280 - 表达式求值

## 题目描述

求一个可能包含加、减、乘、除、乘方运算的中缀表达式的值。
在计算机中，我们常用栈来解决这一问题。

首先将中缀表达式转换到后缀表达式，然后对后缀表达式求值。

加、减、乘、除、乘方分别用 `+`，`-`，`*`, `/`, `^` 来表示。表达式可以有圆括号()。

## 输入

第一行为测试数据的组数 $N$ 。

接下来的 $N$ 行，每行是一个中缀表达式。 每个表达式中，圆括号、运算符和运算数相互之间都用空格分隔，运算数是整数。一般运算数可正可负（负数的符号和数字之间无空格），指数一定为自然数($0$ 和正整数)。

不必考虑除 $0$ 的情况。每个运算数均可由 $int$ 放下。不必考虑溢出。中缀表达式的字符串长度不超过 $600$ 。乘方的优先级比乘除都高，结合性是向左结合，如 `2 ^ 3 ^ 4` 表示 `( 2 ^ 3 ) ^ 4 = 4096`。 除法的商向下取整。

## 输出

对每一组测试数据输出一行，为表达式的值
## AC代码

```cpp
#include <bits/stdc++.h>
using namespace std;

int n;
string s;
//运算符栈，用于辅助中缀转后缀
stack<char> op;
//运算数栈，用于辅助后缀表达式计算
stack<int> st;
vector<string> v;//存储后缀表达式
string t1 = "*/^";//优先级高的运算符
string t2 = "+-*/^";//所有运算符

//计算y的x次方
int cf(int y,int x) {
	int r = 1;
	for(int i = 1; i <= x; i++) r *= y;
	return r;
}

int main() {
	getline(cin,s);
	n = stoi(s);
	while(n--) {
		getline(cin,s);

		//将中缀转后缀
		int len = s.size();
		string w = "";//存储运算数
		string tt = " ";//运算符字符串
		v.clear();
		char f = '+';//标记负数，默认不是负数
		for(int i = 0; i < len; i++) {
			//遇到运算数存入后缀表达式
			if(isdigit(s[i])) {
				//如果是第一个数字，且上一个是负号
				if(i != 0 && s[i-1] == '-') f = '-';
				w = w + s[i];
				//到运算数结尾
				if(i==len-1 || !isdigit(s[i+1])) {
					if(f == '-') w = "-" + w;
					v.push_back(w);
					w = "";
					f = '+';
				}
			} else if(s[i] == '(') op.push(s[i]); //左括号入栈
			else if(s[i] == ')') {
				//右括号弹出栈中元素直到遇到左括号
				while(!op.empty() && op.top() != '(') {
					tt[0] = op.top();
					v.push_back(tt);
					op.pop();
				}
				op.pop();//左括号弹出
			} else if(s[i] == '^') { //^ 优先级最高
				//弹出栈中运算符直到遇到更低优先级的元素
				while(!op.empty()&&(op.top() == '^')) {
					tt[0] = op.top();
					v.push_back(tt);
					op.pop();
				}
				op.push(s[i]);//当前运算符入栈
			} else if(t1.find(s[i])!=-1) { //*/
				//弹出栈中运算符直到遇到更低优先级的元素
				while(!op.empty()&&(t1.find(op.top())!=-1)) {
					tt[0] = op.top();
					v.push_back(tt);
					op.pop();
				}
				op.push(s[i]);//当前运算符入栈
			} else if(t2.find(s[i])!=-1) { //优先级较低的运算符
				//紧挨着数字的-是负号，不是-
				if(i != len -1 && s[i] == '-' && isdigit(s[i+1])) continue;
				//弹出栈中所有运算符
				while(!op.empty()&&(t2.find(op.top())!=-1)) {
					tt[0] = op.top();
					v.push_back(tt);
					op.pop();
				}
				op.push(s[i]);
			}
		}

		//弹出运算符栈中的剩余元素
		while(!op.empty()) {
			tt[0] = op.top();
			v.push_back(tt);
			op.pop();
		}

		//后缀表达式计算
		len = v.size();
		int x,y;
		for(int i = 0; i < len; i++) {
			//运算数入栈
			if(t2.find(v[i]) == -1) {
				st.push(stoi(v[i]));
			} else {
				//运算符：取出次顶和栈顶运算，结果入栈
				x = st.top();
				st.pop();
				y = st.top();
				st.pop();
//                cout<<y<<v[i]<<x<<endl;
				if(v[i] == "+") st.push(y+x);
				else if(v[i] == "-") st.push(y-x);
				else if(v[i] == "*") st.push(y*x);
				else if(v[i] == "/") st.push(y/x);
				else if(v[i] == "^") st.push(cf(y,x));
			}
		}

		cout<<st.top()<<endl;
	}
	return 0;
}
```

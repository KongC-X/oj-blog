## AC代码

```cpp
#include<bits/stdc++.h>
using namespace std;

int main() {
    int a[10][10];
    int i, j, n;
    cin >> n;
    //赋初值
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            a[i][j] = 1;
        }
    }

    //赋0
    for(i = 0; i <= n / 2; i++) {
        a[i][n / 2 - i] = 0;//上半部分左边的0
        a[i][n / 2 + i] = 0;//上半部分右边的0

        a[n - 1 - i][n / 2 - i] = 0;//下半部分左边的0
        a[n - 1 - i][n / 2 + i] = 0;//下半部分右边的0
    }
    a[n / 2][n / 2] = 0;//最中间的0

    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            cout << setw(3) << a[i][j];
        }
        cout << endl;
    }

    return 0;
}
```

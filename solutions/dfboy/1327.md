## 解题思路

## AC代码

```cpp
#include<iostream>
using namespace std;

int a[110],n,s = 0;

int main() {
	cin>>n;
	for(int i = 0;i < n;i++){
		cin>>a[i];
    }
    int max = a[0];
    for(int i = 0;i < n;i++){
        if(a[i] > max){
            max = a[i];
        }
        s += a[i];
    }
    if(s > 100) {
        cout << "Yes" << endl;
        cout << s - max;
    } else {
        cout << "No" << endl;
        cout << s;
    }
}
```

# 1713 - 输出满足条件的整数3

## 题目描述

有一个数列，该数列的前 $4$ 个数是： $1$ $4$ $7$ $10$ ；

请从键盘读入一个正整数 $n$ ，请通过观察前 $4$ 项的规律，输出 $1 \sim n$ 之间所有满足该规律的数字。

## 输入

在一行输入一个整数 $n$ 。

## 输出

输出满足题目要求的整数，每行 $1$ 个。

## 解题思路

## AC代码

```cpp
#include<bits/stdc++.h> 
#include<iostream> 
using namespace std;

// 第一种方法：更符合快速排序的经典描述（分治 + 双指针交换），交换逻辑更直接，容易可视化
// 分区函数：将数组分为两部分，左边小于基准值，右边大于基准值
int partition(int arr[], int left, int right) {
    int pivot = arr[left]; // 选择第一个元素作为基准值
	int i = left; // 左指针
	int j = right; // 右指针
    while(i < j) {
        // 从右向左找到第一个小于基准值的元素
        while(i < j && arr[j] >= pivot) {
            j--;
        }
        // 从左向右找到第一个大于基准值的元素
        while(i < j && arr[i] <= pivot) {
            i++;
        }
        // 如果i < j, 交换两个元素
		if(i < j) {
			swap(arr[i], arr[j]);
		}
    }
    // 将基准值放到正确的位置，即排序后的最终位置
    swap(arr[left], arr[i]);
    return i; // 返回基准值的位置
}


// 第二种方法：单方向扫描，维护一个"小于基准"的边界，代码结构更简单，但是交换次数会变多
// 分区函数
int partition(int arr[], int left, int right) {
	int pivot = arr[right]; // 选择最右边的元素作为基准值
	int i = left - 1; // i指向小于基准值的最后一个元素
	for(int j = left; j < right; j++) {
		if(arr[j] <= pivot) { // 如果当前元素小于等于基准值
			i++; // i指针右移,扩大小于基准的区域
			swap(arr[i], arr[j]); // 把当前元素放到小于区域
		}
	}
	swap(arr[i + 1], arr[right]); // 将基准值放到正确的位置
	return i + 1; // 返回基准值的位置
}


// 快速排序函数，使用递归实现
void quickSort(int arr[],int left,int right) {
	if(left < right) { // 如果子数组长度大于1
		int pivotIndex = partition(arr, left, right); // 找到基准值的位置
		quickSort(arr, left, pivotIndex - 1); // 对基准值左边的子数组进行快速排序
		quickSort(arr, pivotIndex + 1, right); // 对基准值右边的子数组进行快速排序

	}
}

int main() {
	int a[9] = {52,144,42,134,63,186,18,143,167};
	int n = sizeof(a)/sizeof(a[0]); // 获取数组长度
	quickSort(a, 0, n - 1); // 调用快速排序函数
	for(int i=0; i<n; i++) {
		cout << a[i] << " ";
	}
}
```

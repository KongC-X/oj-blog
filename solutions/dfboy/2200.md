# 2200 - 树的公共祖先(LCA)(3)

## 题目描述

给定一棵 $n$ 个结点的树（结点标号 $1 \dots n$ ）以及树中结点的边，结点 $s$ 为树的根。

有 $m$ 次询问，请求出每次询问的两个结点 $x$ 和 $y$ 的最近的公共祖先结点。

## 输入

第 $1$ 行输入 $3$ 个整数 $n$、$m$、$s$（$n≤500000$，$m≤500000$，$1≤s≤n$）； 

接下来 $n-1$ 行，每行两个整数 $a$ 和 $b$ ，结点 $a$ 和 $b$ 是父子关系，但**不保证 $a$ 是 $b$ 的父**，数据保证一定能构成树；

接下来 $m$ 行，每行两个整数 $x$，$y$，表示要求出 $x$ 和 $y$ 结点的公共祖先。

## 输出

输出 $m$ 行，每行一个整数，表示 $m$ 次询问求出的结果。
## AC代码

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 5e5 + 10;   // 最大结点数
const int L = 20;          // 2^20 = 1,048,576 > 5e5，所以 20 就够了

// ---------- 链式前向星（邻接表） ----------
int pre[N];          // pre[x]：结点 x 的最后一条出边在 a 数组中的下标
struct node {
    int to;          // 这条边指向的目标结点
    int next;        // 上一条以相同起点出发的边的下标
} a[N * 2];          // 存边数组，无向图每条边存两次

int k = 0;           // 当前已使用的 a 数组下标

// 加一条从 u 到 v 的有向边
void add(int u, int v) {
    k++;
    a[k].to = v;
    a[k].next = pre[u]; // 新边的 next 指向 u 原来的"最后一条边"
    pre[u] = k;         // 更新 u 的最后一条边为当前新边
}
// -----------------------------------------

int n, m, s;         // n:结点数, m:询问数, s:根结点编号
int f[N][L];         // f[x][j]：结点 x 的 2^j 级祖先
int dep[N];          // dep[x]：结点 x 的深度（根深度为 1）
int lg[N];           // lg[i]：⌊log₂(i)⌋，即 i 的二进制位数减 1

/**
 * DFS 预处理深度和一级祖先（f[x][0]）
 * @param x 当前结点
 * @param fath 当前结点的父结点
 */
void dfs(int x, int fath) {
    dep[x] = dep[fath] + 1;      // 深度 = 父结点深度 + 1
    f[x][0] = fath;              // 1 级祖先（2^0）就是父结点

    // 遍历 x 的所有邻居
    for (int i = pre[x]; i != 0; i = a[i].next) {
        int to = a[i].to;
        if (to != fath) {        // 不走回头路
            dfs(to, x);
        }
    }
}

/**
 * 求 x 和 y 的最近公共祖先
 */
int lca(int x, int y) {
    // ===== 第一步：将 x 和 y 提升到同一深度 =====
    // 保证 x 是较深的那个
    if (dep[x] < dep[y]) swap(x, y);

    // 将 x 向上跳 (dep[x] - dep[y]) 步，使 x 和 y 深度相同
    // 利用二进制拆分：dep[x] - dep[y] 这个差值可以拆成若干个 2^j 的和
    // lg[dep[x]-dep[y]] 就是最大的 j 满足 2^j <= 差值
    while (dep[x] > dep[y]) {
        // 每次跳 2^(lg[差值]) 步，直到深度相等
        x = f[x][lg[dep[x] - dep[y]]];
    }

    // 如果此时 x == y，说明 y 本来就是 x 的祖先
    if (x == y) return x;

    // ===== 第二步：x 和 y 一起向上跳，直到它们的父结点相同 =====
    // 从高位到低位枚举，保证不跳过头
    for (int i = L - 1; i >= 0; i--) {
        // 如果跳 2^i 步后，x 和 y 还不相等，说明 LCA 还在上面
        if (f[x][i] != f[y][i]) {
            x = f[x][i];   // 一起往上跳
            y = f[y][i];
        }
    }
    // 循环结束后，x 和 y 的父结点就是 LCA
    return f[x][0];
}

int main() {
    // 输入 n, m, s
    scanf("%d%d%d", &n, &m, &s);

    // 读入 n-1 条边，建无向图
    int x, y;
    for (int i = 1; i <= n - 1; i++) {
        scanf("%d%d", &x, &y);
        add(x, y);   // 加 x→y
        add(y, x);   // 加 y→x（双向）
    }

    // ===== 第一步：DFS 预处理深度和一级祖先 =====
    // 从根 s 开始，父结点设为 0（不存在）
    dfs(s, 0);       // dep[0] = 0，所以 dep[s] = 1

    // ===== 第二步：预处理倍增表 f[x][j] =====
    // f[x][j] = f[ f[x][j-1] ][j-1]
    // 即：x 的 2^j 级祖先 = (x 的 2^(j-1) 级祖先) 的 2^(j-1) 级祖先
    for (int j = 1; j < L; j++) {
        for (int i = 1; i <= n; i++) {
            f[i][j] = f[f[i][j-1]][j-1];
        }
    }

    // ===== 第三步：预计算 log2 向下取整 =====
    // lg[i] = ⌊log₂(i)⌋
    // 用于 lca 函数中快速确定每次跳多少步
    lg[1] = 0;
    for (int i = 2; i <= n; i++) {
        lg[i] = lg[i / 2] + 1;  // 等价于 floor(log2(i))
    }

    // ===== 第四步：处理 m 次询问 =====
    while (m--) {
        scanf("%d%d", &x, &y);
        printf("%d\n", lca(x, y));
    }

    return 0;
}
```
